<script define:vars={{ albums, r2Base }}>
  let currentAlbumIndex = 0;
  let currentSongIndex = 0;

  /*
   * This number changes whenever a new lyrics request begins.
   * It prevents a slower, older request from replacing the lyrics
   * for an album selected more recently.
   */
  let lyricsRequestNumber = 0;

  const albumsEl = document.getElementById('albums');
  const songsEl = document.getElementById('songs');
  const albumTitleEl = document.getElementById('albumTitle');
  const nowPlayingEl = document.getElementById('nowPlaying');
  const albumLyricsEl = document.getElementById('albumLyrics');
  const audio = document.getElementById('audioPlayer');
  const prevBtn = document.getElementById('prevBtn');
  const nextBtn = document.getElementById('nextBtn');

  function getSongUrl(song) {
    if (song.file.startsWith('http')) {
      return song.file;
    }

    return r2Base + song.file;
  }

  async function loadAlbumLyrics(album) {
    const requestNumber = ++lyricsRequestNumber;

    if (!album.lyricsFile) {
      albumLyricsEl.textContent = 'Lyrics Unknown';
      return;
    }

    albumLyricsEl.textContent = 'Loading lyrics...';

    try {
      const response = await fetch(album.lyricsFile);

      if (!response.ok) {
        throw new Error(
          `Lyrics request failed with status ${response.status}`
        );
      }

      const lyricsText = await response.text();

      /*
       * Ignore this result if another album was selected while
       * this text file was still loading.
       */
      if (requestNumber !== lyricsRequestNumber) {
        return;
      }

      albumLyricsEl.textContent =
        lyricsText.trim().length > 0
          ? lyricsText
          : 'Lyrics Unknown';
    } catch (error) {
      console.error(
        `Could not load lyrics for "${album.title}":`,
        error
      );

      if (requestNumber === lyricsRequestNumber) {
        albumLyricsEl.textContent = 'Lyrics Unknown';
      }
    }
  }

  function updateSelectedAlbumButton() {
    const albumButtons =
      albumsEl.querySelectorAll('.album-button');

    albumButtons.forEach((button, index) => {
      const isSelected = index === currentAlbumIndex;

      button.classList.toggle('selected', isSelected);
      button.setAttribute(
        'aria-pressed',
        isSelected ? 'true' : 'false'
      );
    });
  }

  function renderAlbums() {
    albumsEl.innerHTML = '';

    albums.forEach((album, index) => {
      const button = document.createElement('button');

      button.type = 'button';
      button.textContent = album.title;
      button.className = 'album-button';
      button.setAttribute('aria-pressed', 'false');

      button.addEventListener('click', () => {
        currentAlbumIndex = index;
        currentSongIndex = 0;

        renderSongs();
        updateSelectedAlbumButton();
      });

      albumsEl.appendChild(button);
    });

    updateSelectedAlbumButton();
  }

  function renderSongs() {
    const album = albums[currentAlbumIndex];

    if (!album) {
      albumTitleEl.textContent = 'No albums available';
      songsEl.innerHTML = '';
      albumLyricsEl.textContent = 'Lyrics Unknown';
      return;
    }

    albumTitleEl.textContent = album.title;
    songsEl.innerHTML = '';

    /*
     * Selecting another album does not automatically start music.
     * It also resets the Now Playing message until a song is chosen.
     */
    nowPlayingEl.textContent = 'Nothing playing';

    loadAlbumLyrics(album);

    if (!album.songs || album.songs.length === 0) {
      const message = document.createElement('p');
      message.textContent = 'No songs available.';
      message.className = 'empty-message';
      songsEl.appendChild(message);
      return;
    }

    album.songs.forEach((song, index) => {
      const button = document.createElement('button');

      button.type = 'button';
      button.textContent = song.title;
      button.className = 'song-button';

      button.addEventListener('click', () => {
        playSong(index);
      });

      songsEl.appendChild(button);
    });
  }

  function updateSelectedSongButton() {
    const songButtons =
      songsEl.querySelectorAll('.song-button');

    songButtons.forEach((button, index) => {
      const isSelected = index === currentSongIndex;

      button.classList.toggle('selected', isSelected);
      button.setAttribute(
        'aria-current',
        isSelected ? 'true' : 'false'
      );
    });
  }

  function playSong(index) {
    const album = albums[currentAlbumIndex];

    if (!album?.songs?.length) {
      return;
    }

    const song = album.songs[index];

    if (!song) {
      return;
    }

    currentSongIndex = index;
    audio.src = getSongUrl(song);
    nowPlayingEl.textContent = `Now Playing: ${song.title}`;

    updateSelectedSongButton();

    audio.play().catch((error) => {
      console.error('Audio playback could not begin:', error);
    });
  }

  function nextSong() {
    const album = albums[currentAlbumIndex];

    if (!album?.songs?.length) {
      return;
    }

    currentSongIndex =
      (currentSongIndex + 1) % album.songs.length;

    playSong(currentSongIndex);
  }

  function prevSong() {
    const album = albums[currentAlbumIndex];

    if (!album?.songs?.length) {
      return;
    }

    currentSongIndex =
      (currentSongIndex - 1 + album.songs.length) %
      album.songs.length;

    playSong(currentSongIndex);
  }

  nextBtn.addEventListener('click', nextSong);
  prevBtn.addEventListener('click', prevSong);
  audio.addEventListener('ended', nextSong);

  renderAlbums();
  renderSongs();
</script>
