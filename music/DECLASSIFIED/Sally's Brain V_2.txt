export default {
  async fetch(request, env) {
    const url =
      new URL(request.url);

    /*
     * Sally's normal conversation endpoint.
     */
    if (
      request.method === "POST" &&
      url.pathname === "/chat"
    ) {
      return handleChat(
        request,
        env
      );
    }

    /*
     * Public persistent state.
     */
    if (
      request.method === "GET" &&
      url.pathname === "/state"
    ) {
      return handleState(
        request,
        env,
        url
      );
    }

    /*
     * Purchase food or drink and give it to Sally.
     *
     * The browser supplies only:
     *
     * {
     *   visitorId,
     *   itemId
     * }
     *
     * Price and effects come exclusively from the
     * server-side FOOD_CATALOG below.
     */
    if (
      request.method === "POST" &&
      url.pathname === "/feed"
    ) {
      return handleFeed(
        request,
        env
      );
    }

    /*
     * Allows the page to construct its dropdown
     * without duplicating prices in the HTML.
     *
     * Only safe public item details are returned.
     */
    if (
      request.method === "GET" &&
      url.pathname ===
        "/food-catalog"
    ) {
      return handleFoodCatalog();
    }

    /*
     * Small diagnostic endpoint.
     */
    if (
      request.method === "GET" &&
      url.pathname === "/health"
    ) {
      const db =
        getWorldDatabase(env);

      return jsonResponse({
        ok: true,

        resident:
          "prisoner-sally",

        databaseAvailable:
          Boolean(db),

        databaseBinding:
          env.WORLD_DB
            ? "WORLD_DB"
            : env.SALLY_DB
              ? "SALLY_DB"
              : null,

        feedingSystem:
          "enabled"
      });
    }

    /*
     * Every other path displays Sally's page.
     */
    return new Response(
      HTML,
      {
        headers: {
          "content-type":
            "text/html; charset=UTF-8",

          "cache-control":
            "no-store"
        }
      }
    );
  }
};

    getWorldDatabase(env);

  if (!db) {
    throw new Error(
      'Missing D1 binding. Add the shared database as "WORLD_DB" or retain the existing "SALLY_DB" binding.'
    );

  return db;

function getFoodItem(itemId) {
  const normalizedId =
    String(itemId || "")
      .trim()
      .toLowerCase();

  return (
    FOOD_CATALOG[
      normalizedId
    ] ??
    null
  );
}


function buildPublicFoodCatalog() {
  return FOOD_CATALOG_ORDER
    .map((itemId) => {
      const item =
        FOOD_CATALOG[itemId];

      if (!item) {
        return null;
      }

      return {
        id:
          item.id,

        name:
          item.name,

        category:
          item.category,

        description:
          item.description,

        price:
          item.price
      };
    })
    .filter(Boolean);
}


function handleFoodCatalog() {
  return jsonResponse({
    items:
      buildPublicFoodCatalog()
  });
}

   SHARED DATE HELPERS
   ========================================================= */

function getWorldDateKey(
  value = new Date()
) {
  const parts =
    new Intl.DateTimeFormat(
      "en-CA",
      {
        timeZone:
          WORLD_TIME_ZONE,

        year:
          "numeric",

        month:
          "2-digit",

        day:
          "2-digit"
      }
    ).formatToParts(value);

  const values = {};

  for (const part of parts) {
    if (
      part.type !== "literal"
    ) {
      values[part.type] =
        part.value;
    }
  }

  return (
    values.year +
    "-" +
    values.month +
    "-" +
    values.day
  );
}


function differenceInDateKeys(
  earlierKey,
  laterKey
) {
  const earlier =
    parseDateKey(earlierKey);

  const later =
    parseDateKey(laterKey);

  if (
    !earlier ||
    !later
  ) {
    return null;
  }

  const difference =
    later.getTime() -
    earlier.getTime();

  return Math.round(
    difference /
      86_400_000
  );
}


function parseDateKey(value) {
  const match =
    String(value || "")
      .match(
        /^(\d{4})-(\d{2})-(\d{2})$/
      );

  if (!match) {
    return null;
  }

  const year =
    Number(match[1]);

  const month =
    Number(match[2]);

  const day =
    Number(match[3]);

  const result =
    new Date(
      Date.UTC(
        year,
        month - 1,
        day
      )
    );

  return Number.isFinite(
    result.getTime()
  )
    ? result
    : null;
}

/*
 * A starvation rescue receives an emotional multiplier.
 *
 * Existing hunger rules begin starvation damage at 75 and
 * escalate it at 90 and 95, so those levels are meaningful.
 */
function getStarvationRescueMultiplier(
  hungerBefore
) {
  if (hungerBefore >= 95) {
    return 2.5;
  }

  if (hungerBefore >= 90) {
    return 2;
  }

  if (hungerBefore >= 75) {
    return 1.5;
  }

  return 1;
}


function isStarvationRescue(
  hungerBefore
) {
  return hungerBefore >= 75;
}


/*
 * Happiness and trust effects are scaled by genuine need.
 *
 * This function is not used when Sally was already full.
 */
function calculateBeneficialFeedingEffects(
  item,
  hungerBefore
) {
  const rescueMultiplier =
    getStarvationRescueMultiplier(
      hungerBefore
    );

  const hungerNeedMultiplier =
    hungerBefore >= 75
      ? 1
      : hungerBefore >= 50
        ? 0.85
        : hungerBefore >= 30
          ? 0.65
          : 0.45;

  return {
    happiness:
      Math.max(
        1,
        Math.round(
          item.happiness *
          rescueMultiplier *
          hungerNeedMultiplier
        )
      ),

    kindness:
      Math.max(
        1,
        Math.min(
          5,
          Math.round(
            item.kindness *
            rescueMultiplier
          )
        )
      ),

    trust:
      Math.max(
        0,
        Math.round(
          item.trust *
          rescueMultiplier *
          hungerNeedMultiplier
        )
      ),

    fearRelief:
      Math.max(
        0,
        Math.round(
          item.fearRelief *
          rescueMultiplier
        )
      ),

    resentmentRelief:
      Math.max(
        0,
        Math.round(
          item.resentmentRelief *
          rescueMultiplier
        )
      ),

    perceivedCrueltyRelief:
      Math.max(
        0,
        Math.round(
          item.perceivedCrueltyRelief *
          rescueMultiplier
        )
      )
  };
}


/*
 * Reconciliation through food is intentionally limited by
 * time and genuine need.
 *
 * Purchasing many foods in one sitting cannot establish
 * long-term care reliability.
 */
function calculateCareReliabilityIncrease(
  sally,
  hungerBefore,
  todayKey
) {
  if (
    hungerBefore <
      MINIMUM_BENEFICIAL_HUNGER
  ) {
    return 0;
  }

  if (
    sally.lastBeneficialMealDate ===
      todayKey
  ) {
    /*
     * Additional useful food on the same day may still help
     * hunger, but does not farm daily reliability.
     */
    return 0;
  }

  const previousDate =
    sally.lastBeneficialMealDate;

  const dayDifference =
    differenceInDateKeys(
      previousDate,
      todayKey
    );

  if (
    dayDifference === 1
  ) {
    return 4;
  }

  if (
    dayDifference !== null &&
    dayDifference > 1 &&
    dayDifference <= 3
  ) {
    return 2;
  }

  return 1;
}


/*
 * Feeding streaks are based on useful meals on separate
 * calendar days.
 */
function calculateNextFeedingStreak(
  sally,
  todayKey
) {
  if (
    sally.lastBeneficialMealDate ===
      todayKey
  ) {
    return Math.max(
      1,
      Math.floor(
        Number(
          sally.feedingStreak
        ) || 1
      )
    );
  }

  const dayDifference =
    differenceInDateKeys(
      sally.lastBeneficialMealDate,
      todayKey
    );

  if (dayDifference === 1) {
    return Math.max(
      1,
      Math.floor(
        Number(
          sally.feedingStreak
        ) || 0
      ) + 1
    );
  }

  /*
   * Missing days does not create punishment.
   * It simply begins a new current streak.
   */
  return 1;
}

async function ensureDatabase(env) {
  const db =
    requireWorldDatabase(env);

  await db.batch([
    db.prepare(`
      CREATE TABLE IF NOT EXISTS
      world_state (
        visitor_id TEXT PRIMARY KEY,
        state_json TEXT NOT NULL,
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL
      )
    `),

    db.prepare(`
      CREATE TABLE IF NOT EXISTS
      conversation_log (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        visitor_id TEXT NOT NULL,
        role TEXT NOT NULL,
        content TEXT NOT NULL,
        created_at TEXT NOT NULL
      )
    `),

    db.prepare(`
      CREATE INDEX IF NOT EXISTS
      idx_conversation_visitor_created

      ON conversation_log(
        visitor_id,
        created_at
      )
    `),

    db.prepare(`
      CREATE TABLE IF NOT EXISTS
      memories (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        visitor_id TEXT NOT NULL,
        type TEXT NOT NULL,
        summary TEXT NOT NULL,
        importance REAL NOT NULL DEFAULT 0.5,
        emotional_weight REAL NOT NULL DEFAULT 0,
        created_at TEXT NOT NULL,
        last_recalled_at TEXT,
        recall_count INTEGER NOT NULL DEFAULT 0
      )
    `),

    db.prepare(`
      CREATE INDEX IF NOT EXISTS
      idx_memories_visitor_created

      ON memories(
        visitor_id,
        created_at
      )
    `),

    /*

/* =========================================================
   STATE STORAGE
   ========================================================= */

async function loadWorldState(
  env,
  visitorId
) {
  const db =
    requireWorldDatabase(env);

  const row =
    await db
      .prepare(`
        SELECT state_json

        FROM world_state

        WHERE visitor_id = ?

        LIMIT 1
      `)
      .bind(
        visitorId
      )
      .first();


  /*
   * Brand-new player.
   */
  if (!row?.state_json) {
    const state =
      createDefaultWorldState();

    await saveWorldState(
      env,
      visitorId,
      state,
      true
    );

    return state;
  }


  try {
    const parsed =
      JSON.parse(
        row.state_json
      );

    return normalizeWorldState(
      parsed
    );

  } catch (error) {
    console.error(
      "Stored world state was unreadable:",
      error
    );

    /*
     * Preserve application operation even if one row became
     * unreadable.
     *
     * This resets only that malformed row.
     */
    const state =
      createDefaultWorldState();

    await saveWorldState(
      env,
      visitorId,
      state,
      false
    );

    return state;
  }
}


async function saveWorldState(
  env,
  visitorId,
  state,
  isNew = false
) {
  const db =
    requireWorldDatabase(env);

  const now =
    new Date().toISOString();

  state.lastUpdatedAt =
    now;

  const serialized =
    JSON.stringify(
      normalizeWorldState(
        state
      )
    );


  if (isNew) {
    await db
      .prepare(`
        INSERT OR REPLACE INTO
        world_state (
          visitor_id,
          state_json,
          created_at,
          updated_at
        )

        VALUES (
          ?,
          ?,
          ?,
          ?
        )
      `)
      .bind(
        visitorId,
        serialized,
        now,
        now
      )
      .run();

    return;
  }


  await db
    .prepare(`
      INSERT INTO
      world_state (
        visitor_id,
        state_json,
        created_at,
        updated_at
      )

      VALUES (
        ?,
        ?,
        ?,
        ?
      )

      ON CONFLICT(visitor_id)
      DO UPDATE SET
        state_json =
          excluded.state_json,

        updated_at =
          excluded.updated_at
    `)
    .bind(
      visitorId,
      serialized,
      now,
      now
    )
    .run();
}


/* =========================================================
   WORLD-STATE NORMALIZATION
   ========================================================= */

function normalizeWorldState(
  candidate
) {
  const fallback =
    createDefaultWorldState();

  const player =
    normalizeCharacterState(
      candidate?.player,
      fallback.player,
      false
    );

  const sally =
    normalizeCharacterState(
      candidate?.sally,
      fallback.sally,
      true
    );

  return {
    player,
    sally,

    lastUpdatedAt:
      validIsoDate(
        candidate?.lastUpdatedAt
      ) ||
      fallback.lastUpdatedAt,

    lastHungerUpdateAt:
      validIsoDate(
        candidate?.lastHungerUpdateAt
      ) ||
      fallback.lastHungerUpdateAt
  };
}


/* =========================================================
   CHARACTER-STATE NORMALIZATION
   ========================================================= */

function normalizeCharacterState(
  candidate,
  fallback,
  isSally
) {
  const safeCandidate =
    candidate &&
    typeof candidate === "object"
      ? candidate
      : {};

  const equipment =
    safeCandidate.equipment &&
    typeof safeCandidate.equipment ===
      "object"
      ? safeCandidate.equipment
      : {};

  const normalized = {
    hp:
      clampNumber(
        safeCandidate.hp,
        0,
        1000,
        fallback.hp
      ),

    maxHp:
      clampNumber(
        safeCandidate.maxHp,
        1,
        1000,
        fallback.maxHp
      ),

    hunger:
      clampNumber(
        safeCandidate.hunger,
        HUNGER_MINIMUM,
        HUNGER_MAXIMUM,
        fallback.hunger
      ),

    money:
      Math.floor(
        clampNumber(
          safeCandidate.money,
          0,
          1_000_000_000,
          fallback.money
        )
      ),

    conditions:
      normalizeStringObjectArray(
        safeCandidate.conditions,
        30
      ),

    inventory:
      normalizeInventory(
        safeCandidate.inventory
      ),

    equipment: {
      head:
        normalizeNullableString(
          equipment.head
        ),

      body:
        normalizeNullableString(
          equipment.body
        ),

      hands:
        normalizeNullableString(
          equipment.hands
        ),

      legs:
        normalizeNullableString(
          equipment.legs
        ),

      accessory:
        normalizeNullableString(
          equipment.accessory
        ),

      weapon:
        normalizeNullableString(
          equipment.weapon
        )
    },

    spellsKnown:
      normalizeStringObjectArray(
        safeCandidate.spellsKnown,
        50
      ),

    allegianceToLoki:
      clampNumber(
        safeCandidate.allegianceToLoki,
        -100,
        100,
        fallback.allegianceToLoki
      )
  };


  normalized.hp =
    Math.min(
      normalized.hp,
      normalized.maxHp
    );


  /*
   * Player-only fields.

  if (!isSally) {
    normalized.lastDailyGoldDate =
      typeof

        "string"
        ? safeCandidate.lastDailyGoldDate
        : null;

    return normalized;
  }


  /*
   * Sally current emotional state.
   */
  normalized.trust =
    clampNumber(
      safeCandidate.trust,
      -100,
      100,
      fallback.trust
    );

  normalized.fear =
    clampNumber(
      safeCandidate.fear,
      0,
      100,
      fallback.fear
    );

  normalized.resentment =
    clampNumber(
      safeCandidate.resentment,
      0,
      100,
      fallback.resentment
    );

  normalized.happiness =
    clampNumber(
      safeCandidate.happiness,
      HAPPINESS_MINIMUM,
      HAPPINESS_MAXIMUM,
      fallback.happiness
    );

  normalized.perceivedCruelty =
    clampNumber(
      safeCandidate.perceivedCruelty,
      PERCEIVED_CRUELTY_MINIMUM,
      PERCEIVED_CRUELTY_MAXIMUM,
      /*
       * Existing players will not have this field.
       *
       * Seed it conservatively from current fear and
       * resentment rather than permanent crueltyReceived.
       *
       * This avoids transforming old historical cruelty
       * automatically into maximum present hostility.
       */
      Math.min(
        100,
        Math.round(
          (
            Number(
              safeCandidate.fear
            ) || 0
          ) * 0.4 +
          (
            Number(
              safeCandidate.resentment
            ) || 0
          ) * 0.6
        )
      )
    );

  normalized.careReliability =
    clampNumber(
      safeCandidate.careReliability,
      CARE_RELIABILITY_MINIMUM,
      CARE_RELIABILITY_MAXIMUM,
      fallback.careReliability
    );

  normalized.intoxication =
    clampNumber(
      safeCandidate.intoxication,
      INTOXICATION_MINIMUM,
      INTOXICATION_MAXIMUM,
      fallback.intoxication
    );


  /*
   * Permanent relationship history.
   */
  normalized.kindnessReceived =
    normalizeRelationshipTotal(
      safeCandidate.kindnessReceived,
      fallback.kindnessReceived
    );

  normalized.crueltyReceived =
    normalizeRelationshipTotal(
      safeCandidate.crueltyReceived,
      fallback.crueltyReceived
    );

  normalized.amusementReceived =
    normalizeRelationshipTotal(
      safeCandidate.amusementReceived,
      fallback.amusementReceived
    );


  /*
   * Feeding counters.
   */
  normalized.mealsReceived =
    normalizeNonNegativeInteger(
      safeCandidate.mealsReceived,
      fallback.mealsReceived
    );

  normalized.alcoholReceived =
    normalizeNonNegativeInteger(
      safeCandidate.alcoholReceived,
      fallback.alcoholReceived
    );

  normalized.daysFed =
    normalizeNonNegativeInteger(
      safeCandidate.daysFed,
      fallback.daysFed
    );

  normalized.feedingStreak =
    normalizeNonNegativeInteger(
      safeCandidate.feedingStreak,
      fallback.feedingStreak,
      MAX_FORCED_FEEDING_STREAK
    );

  normalized.forcedFeedingStreak =
    normalizeNonNegativeInteger(
      safeCandidate.forcedFeedingStreak,
      fallback.forcedFeedingStreak,
      MAX_FORCED_FEEDING_STREAK
    );


  /*
   * Last feeding information.
   */
  normalized.lastMealAt =
    validIsoDate(
      safeCandidate.lastMealAt
    );

  normalized.lastMealId =
    normalizeNullableShortString(
      safeCandidate.lastMealId,
      100
    );

  normalized.lastMealName =
    normalizeNullableShortString(
      safeCandidate.lastMealName,
      200
    );

  normalized.lastMealOutcome =
    normalizeFeedingOutcome(
      safeCandidate.lastMealOutcome
    );

  normalized.lastBeneficialMealDate =
    normalizeDateKey(
      safeCandidate
        .lastBeneficialMealDate
    );

  normalized.lastForcedFeedingAt =
    validIsoDate(
      safeCandidate
        .lastForcedFeedingAt
    );

  normalized.recentFeedingEvents =
    normalizeRecentFeedingEvents(
      safeCandidate
        .recentFeedingEvents
    );


  /*
   * Hidden Loki belief.
   */
  normalized.believesVisitorIsLoki =
    safeCandidate
      .believesVisitorIsLoki === true;

  normalized.lokiBeliefPrimed =
    safeCandidate
      .lokiBeliefPrimed === true;


  /*
   * Recent relationship events.
   */
  normalized.recentRelationshipEvents =
    normalizeRecentRelationshipEvents(
      safeCandidate
        .recentRelationshipEvents
    );


  return normalized;
}

function normalizeNonNegativeInteger(
  value,
  fallback = 0,
  maximum = 1_000_000_000
) {
  const fallbackNumber =
    Number.isFinite(
      Number(fallback)
    )
      ? Math.max(
          0,
          Math.floor(
            Number(fallback)
          )
        )
      : 0;

  const numeric =
    Number(value);

  if (
    !Number.isFinite(
      numeric
    )
  ) {
    return Math.min(
      maximum,
      fallbackNumber
    );
  }

  return Math.max(
    0,
    Math.min(
      maximum,
      Math.floor(
        numeric
      )
    )
  );
}


function normalizeNullableShortString(
  value,
  maximumLength = 200
) {
  if (
    typeof value !== "string"
  ) {
    return null;
  }

  const cleaned =
    value
      .trim()
      .slice(
        0,
        maximumLength
      );

  return cleaned || null;
}


function normalizeDateKey(value) {
  const cleaned =
    typeof value === "string"
      ? value.trim()
      : "";

  return /^\d{4}-\d{2}-\d{2}$/
    .test(cleaned)
      ? cleaned
      : null;
}


function normalizeFeedingOutcome(
  value
) {
  const cleaned =
    typeof value === "string"
      ? value
          .trim()
          .toLowerCase()
      : "";

  return Object.values(
    FEEDING_OUTCOMES
  ).includes(cleaned)
    ? cleaned
    : null;
}


function normalizeRecentFeedingEvents(
  value
) {
  if (!Array.isArray(value)) {
    return [];
  }

  return value
    .slice(-30)
    .map(
      normalizeRecentFeedingEvent
    )
    .filter(Boolean);
}


function normalizeRecentFeedingEvent(
  event
) {
  if (
    !event ||
    typeof event !== "object"
  ) {
    return null;
  }

  const itemId =
    normalizeNullableShortString(
      event.itemId,
      100
    );

  const itemName =
    normalizeNullableShortString(
      event.itemName,
      200
    );

  const outcome =
    normalizeFeedingOutcome(
      event.outcome
    );

  if (
    !itemId ||
    !itemName ||
    !outcome
  ) {
    return null;
  }

  return {
    itemId,
    itemName,

    category:
      event.category === "alcohol"
        ? "alcohol"
        : "food",

    outcome,

    price:
      normalizeNonNegativeInteger(
        event.price,
        0
      ),

    hungerBefore:
      clampNumber(
        event.hungerBefore,
        HUNGER_MINIMUM,
        HUNGER_MAXIMUM,
        0
      ),

    hungerAfter:
      clampNumber(
        event.hungerAfter,
        HUNGER_MINIMUM,
        HUNGER_MAXIMUM,
        0
      ),

    hpDamage:
      clampNumber(
        event.hpDamage,
        0,
        1000,
        0
      ),

    happinessChange:
      clampNumber(
        event.happinessChange,
        -200,
        200,
        0
      ),

    kindnessChange:
      normalizeNonNegativeInteger(
        event.kindnessChange,
        0,
        5
      ),

    crueltyChange:
      normalizeNonNegativeInteger(
        event.crueltyChange,
        0,
        5
      ),

    perceivedCrueltyChange:
      clampNumber(
        event.perceivedCrueltyChange,
        -100,
        100,
        0
      ),

    rescue:
      event.rescue === true,

    createdAt:
      validIsoDate(
        event.createdAt
      ) ||
      new Date()
        .toISOString()
  };
}

  ].slice(-30);

/* =========================================================

function normalizeRelationshipTotal(
  value,
  fallback = 0
) {
  const normalizedFallback =
    Number.isFinite(
      Number(fallback)
    )
      ? Math.max(
          0,
          Math.floor(
            Number(fallback)
          )
        )
      : 0;

  const numeric =
    Number(value);

  if (
    !Number.isFinite(
      numeric
    )
  ) {
    return normalizedFallback;
  }

  return Math.max(
    0,
    Math.min(
      MAX_RELATIONSHIP_SCORE,
      Math.floor(numeric)
    )
  );
}

function normalizeRecentRelationshipEvents(
  value
) {
  if (
    !Array.isArray(value)
  ) {
    return [];
  }

  return value
    .slice(
      -MAX_RECENT_RELATIONSHIP_EVENTS
    )
    .map(
      normalizeRecentRelationshipEvent
    )
    .filter(Boolean);
}

function normalizeRecentRelationshipEvent(
  event
) {
  if (
    !event ||
    typeof event !== "object"
  ) {
    return null;
  }

  const kindness =
    normalizeRelationshipEventScore(
      event.kindness
    );

  const cruelty =
    normalizeRelationshipEventScore(
      event.cruelty
    );

  const amusement =
    normalizeRelationshipEventScore(
      event.amusement
    );

  const pattern =
    normalizeRelationshipPattern(
      event.pattern
    );

  const summary =
    limitText(
      String(
        event.summary || ""
      ).trim(),
      240
    );

  const createdAt =
    validIsoDate(
      event.createdAt
    ) ||
    validIsoDate(
      event.created_at
    ) ||
    new Date()
      .toISOString();

  /*
   * Empty neutral events are not useful enough to retain.
   *
   * A genuinely neutral interaction can still be classified during the
   * current turn, but it does not need to occupy space in the rolling history.
   */
  if (
    kindness === 0 &&
    cruelty === 0 &&
    amusement === 0 &&
    pattern === "neutral" &&
    !summary
  ) {
    return null;
  }

  return {
    kindness,
    cruelty,
    amusement,
    pattern,
    summary,
    createdAt
  };
}

function normalizeRelationshipEventScore(
  value
) {
  const numeric =
    Number(value);

  if (
    !Number.isFinite(
      numeric
    )
  ) {
    return 0;
  }

  return Math.max(
    0,
    Math.min(
      5,
      Math.round(numeric)
    )
  );
}

function normalizeRelationshipPattern(
  value
) {
  const pattern =
    typeof value === "string"
      ? value
          .trim()
          .toLowerCase()
      : "";

  const allowedPatterns =
    new Set([
      "neutral",
      "kind",
      "cruel",
      "amusing",
      "kind-and-amusing",
      "cruel-and-amusing",
      "contradictory",
      "abuse-followed-by-affection",
      "affection-followed-by-abuse",
      "threatening",
      "protective",
      "apologetic",
      "manipulative",
      "mocking",
      "unknown"
    ]);

  return allowedPatterns.has(
    pattern
  )
    ? pattern
    : "unknown";
}

function appendRecentRelationshipEvent(
  sally,
  event
) {
  if (
    !sally ||
    typeof sally !== "object"
  ) {
    return;
  }

  const normalizedEvent =
    normalizeRecentRelationshipEvent(
      event
    );

  if (!normalizedEvent) {
    return;
  }

  const existing =
    Array.isArray(
      sally.recentRelationshipEvents
    )
      ? sally.recentRelationshipEvents
      : [];

  sally.recentRelationshipEvents =
    [
      ...existing,
      normalizedEvent
    ].slice(
      -MAX_RECENT_RELATIONSHIP_EVENTS
    );
}

function summarizeRecentRelationshipHistory(
  events
) {
  const normalizedEvents =
    normalizeRecentRelationshipEvents(
      events
    );

  if (
    normalizedEvents.length === 0
  ) {
    return {
      count: 0,
      kindness: 0,
      cruelty: 0,
      amusement: 0,
      contradictoryEvents: 0,
      cruelEvents: 0,
      kindEvents: 0,
      withdrawalRelevantEvents: 0
    };
  }

  let kindness = 0;
  let cruelty = 0;
  let amusement = 0;
  let contradictoryEvents = 0;
  let cruelEvents = 0;
  let kindEvents = 0;
  let withdrawalRelevantEvents = 0;

  for (
    const event of
    normalizedEvents
  ) {
    kindness +=
      event.kindness;

    cruelty +=
      event.cruelty;

    amusement +=
      event.amusement;

    if (
      event.kindness > 0
    ) {
      kindEvents += 1;
    }

    if (
      event.cruelty > 0
    ) {
      cruelEvents += 1;
    }

    if (
      event.kindness > 0 &&
      event.cruelty > 0
    ) {
      contradictoryEvents += 1;
    }

    if (
      [
        "contradictory",
        "abuse-followed-by-affection",
        "affection-followed-by-abuse",
        "manipulative",
        "mocking"
      ].includes(
        event.pattern
      )
    ) {
      contradictoryEvents += 1;
    }

    if (
      event.cruelty >= 2 ||
      [
        "cruel",
        "cruel-and-amusing",
        "threatening",
        "manipulative",
        "mocking",
        "abuse-followed-by-affection",
        "affection-followed-by-abuse"
      ].includes(
        event.pattern
      )
    ) {
      withdrawalRelevantEvents += 1;
    }
  }

  return {
    count:
      normalizedEvents.length,

    kindness,
    cruelty,
    amusement,
    contradictoryEvents,
    cruelEvents,
    kindEvents,
    withdrawalRelevantEvents
  };
}

/* =========================================================
   HUNGER AND PASSAGE OF TIME
   ========================================================= */

function applyElapsedTime(state) {
  const now = Date.now();

  const previous =
    Date.parse(
      state.lastHungerUpdateAt
    );

  if (!Number.isFinite(previous)) {
    state.lastHungerUpdateAt =
      new Date(now).toISOString();

    return state;
  }

  const elapsedHours =
    Math.max(
      0,
      (now - previous) / 3_600_000
    );

  /*
   * Hunger rises by one point every six hours.
   *
   * Offline growth is capped so a long absence does not
   * automatically destroy the entire game state.
   *
   * Passage of time does not alter trust, kindness, cruelty,
   * amusement, fear, resentment, or any other relationship value.
   * The visitor is not punished merely for being absent.
   */
  const hungerIncrease =
    Math.min(
      40,
      Math.floor(
        elapsedHours / 6
      )
    );

  if (hungerIncrease <= 0) {
    return state;
  }

  state.player.hunger =
    clampNumber(
      state.player.hunger +
        hungerIncrease,
      0,
      100,
      state.player.hunger
    );

  state.sally.hunger =
    clampNumber(
      state.sally.hunger +
        hungerIncrease,
      0,
      100,
      state.sally.hunger
    );

  applyStarvationDamage(
    state.player
  );

  applyStarvationDamage(
    state.sally
  );

  state.lastHungerUpdateAt =
    new Date(now).toISOString();

  return state;
}

function applyStarvationDamage(
  character
) {
  if (character.hunger < 75) {
    return;
  }

  let damage = 0;

  if (character.hunger >= 95) {
    damage = 8;
  } else if (
    character.hunger >= 90
  ) {
    damage = 5;
  } else {
    damage = 2;
  }

  character.hp =
    Math.max(
      0,
      character.hp - damage
    );

  if (
    character.hunger >= 90 &&
    !hasCondition(
      character,
      "starvation"
    )
  ) {
    character.conditions.push({
      id: "starvation",
      name: "Starvation",
      severity:
        character.hunger,
      description:
        "Prolonged hunger is weakening the body.",
      acquiredAt:
        new Date().toISOString()
    });
  }
}

   ========================================================= */

async function saveMemories(
  env,
  visitorId,
  memories
) {
  if (!Array.isArray(memories)) {
    return;
  }

  const db =
    requireWorldDatabase(env);

  const statements = [];

  for (
    const memory of
    memories.slice(0, 6)
  ) {
    const summary =
      typeof memory?.summary ===
        "string"
        ? limitText(
            memory.summary.trim(),
            MAX_STORED_MEMORY_LENGTH
          )
        : "";

    if (!summary) {
      continue;
    }

    const type =
      typeof memory?.type ===
        "string"
        ? (
            limitText(
              memory.type.trim(),
              50
            ) ||
            "event"
          )
        : "event";

    const importance =
      clampNumber(
        memory?.importance,
        0,
        1,
        0.5
      );

    const emotionalWeight =
      clampNumber(
        memory?.emotionalWeight,
        -1,
        1,
        0
      );

    statements.push(
      db
        .prepare(`
          INSERT INTO memories (
            visitor_id,
            type,
            summary,
            importance,
            emotional_weight,
            created_at
          )

          VALUES (?, ?, ?, ?, ?, ?)
        `)
        .bind(
          visitorId,
          type,
          summary,
          importance,
          emotionalWeight,
          new Date()
            .toISOString()
        )
    );
  }

  if (
    statements.length > 0
  ) {
    await db.batch(
      statements
    );
  }
}

/* =========================================================
   MEMORY RETRIEVAL
   ========================================================= */

async function loadRelevantMemories(
  env,
  visitorId,
  message
) {
  const db =
    requireWorldDatabase(env);

  const result =
    await db
      .prepare(`
        SELECT
          id,
          type,
          summary,
          importance,
          emotional_weight,
          created_at,
          recall_count

        FROM memories

        WHERE visitor_id = ?

        ORDER BY
          importance DESC,
          id DESC

        LIMIT 250
      `)
      .bind(
        visitorId
      )
      .all();

  const rows =
    Array.isArray(
      result?.results
    )
      ? result.results
      : [];

  if (
    rows.length === 0
  ) {
    return [];
  }

  const messageWords =
    meaningfulWords(
      message
    );

  const scored =
    rows.map((row) => {
      const memoryWords =
        meaningfulWords(
          row.summary
        );

      let overlap = 0;

      for (
        const word of
        messageWords
      ) {
        if (
          memoryWords.has(word)
        ) {
          overlap += 1;
        }
      }

      const importance =
        Number(
          row.importance
        ) || 0;

      const emotional =
        Math.abs(
          Number(
            row.emotional_weight
          ) || 0
        );

      const recallBonus =
        Math.min(
          0.15,
          (
            Number(
              row.recall_count
            ) || 0
          ) * 0.005
        );

      /*
       * Feeding, rescue, and coercion memories receive a
       * small relevance bonus because those events alter
       * Sally's basic perception of safety and dependence.
       */
      const feedingBonus =
        [
          "meal",
          "alcohol",
          "starvation-rescue",
          "unnecessary-meal",
          "unwanted-feeding",
          "forced-alcohol"
        ].includes(
          row.type
        )
          ? 0.8
          : 0;

      const score =
        overlap * 2 +
        importance * 3 +
        emotional +
        recallBonus +
        feedingBonus;

      return {
        ...row,
        score
      };
    });

  scored.sort(
    (a, b) => {
      if (
        b.score !== a.score
      ) {
        return (
          b.score -
          a.score
        );
      }

      return (
        Number(b.id) -
        Number(a.id)
      );
    }
  );

  const selected =
    scored.slice(
      0,
      MAX_CONTEXT_MEMORIES
    );

  if (
    selected.length > 0
  ) {
    const now =
      new Date()
        .toISOString();

    const updates =
      selected.map(
        (memory) => {
          return db
            .prepare(`
              UPDATE memories

              SET
                last_recalled_at = ?,

                recall_count =
                  recall_count + 1

              WHERE id = ?
            `)
            .bind(
              now,
              memory.id
            );
        }
      );

    await db.batch(
      updates
    );
  }

  return selected;
}

function meaningfulWords(value) {
  const stopWords =
    new Set([
      "the",
      "and",
      "that",
      "this",
      "with",
      "from",
      "have",
      "your",
      "you",
      "are",
      "was",
      "were",
      "for",
      "but",
      "not",
      "what",
      "when",
      "where",
      "who",
      "why",
      "how",
      "into",
      "about",
      "they",
      "them",
      "their",
      "there",
      "then",
      "than",
      "will",
      "would",
      "could",
      "should"
    ]);

  return new Set(
    String(value || "")
      .toLowerCase()
      .replace(
        /[^\p{L}\p{N}'-]+/gu,
        " "
      )
      .split(/\s+/)
      .filter((word) => {
        return (
          word.length >= 3 &&
          !stopWords.has(word)
        );
      })
  );
}

function describeRelationshipHistory(
  sally,
  recentSummary = null
) {
  const kindness =
    normalizeRelationshipTotal(
      sally?.kindnessReceived,
      0
    );

  const cruelty =
    normalizeRelationshipTotal(
      sally?.crueltyReceived,
      0
    );

  const amusement =
    normalizeRelationshipTotal(
      sally?.amusementReceived,
      0
    );

  const recent =
    recentSummary ||
    summarizeRecentRelationshipHistory(
      sally?.recentRelationshipEvents
    );

  const total =
    kindness + cruelty;

  const crueltyShare =
    total > 0
      ? cruelty / total
      : 0;

  const kindnessShare =
    total > 0
      ? kindness / total
      : 0;

  const lines = [];

  if (
    kindness === 0 &&
    cruelty === 0 &&
    amusement === 0
  ) {
    lines.push(
      "Sally has not yet accumulated enough history to form a strong model " +
      "of the visitor."
    );
  } else if (
    cruelty >= 100 &&
    kindness >= 100
  ) {
    lines.push(
      "The visitor has demonstrated substantial kindness and substantial " +
      "cruelty. This is not neutral conduct. Sally has learned that the " +
      "visitor can be warm, helpful, entertaining, and dangerous."
    );

    lines.push(
      "Pleasant behaviour may therefore feel unstable, mocking, calculated, " +
      "or ominous rather than reassuring."
    );
  } else if (
    cruelty >= 40 &&
    kindness >= 20
  ) {
    lines.push(
      "The visitor has a meaningful mixed history. Sally cannot safely assume " +
      "that kindness predicts safety because cruelty is already part of the " +
      "established pattern."
    );
  } else if (
    cruelty >= 40 &&
    crueltyShare >= 0.75
  ) {
    lines.push(
      "Cruelty strongly dominates the visitor's history. Sally has learned " +
      "to regard the visitor as dangerous, degrading, threatening, or unsafe."
    );
  } else if (
    kindness >= 40 &&
    kindnessShare >= 0.85 &&
    cruelty <= Math.max(
      5,
      kindness * 0.1
    )
  ) {
    lines.push(
      "Kindness strongly dominates the visitor's history. Sally has generally " +
      "experienced the visitor as safe, considerate, protective, or humane."
    );

    if (cruelty > 0) {
      lines.push(
        "The existing cruel events still remain part of the history and may " +
        "be remembered as disturbing exceptions."
      );
    }
  } else if (
    cruelty > kindness &&
    cruelty >= 10
  ) {
    lines.push(
      "Cruelty currently outweighs kindness in the historical record. Sally " +
      "has reason to remain wary even when the visitor sounds pleasant."
    );
  } else if (
    kindness > cruelty &&
    kindness >= 10
  ) {
    lines.push(
      "Kindness currently outweighs cruelty in the historical record, but " +
      "any cruel acts remain real and should not be erased."
    );
  } else {
    lines.push(
      "The relationship history is still developing and does not yet support " +
      "a simple or confident interpretation."
    );
  }

  if (
    recent.contradictoryEvents >= 3
  ) {
    lines.push(
      "Recent behaviour has been notably contradictory. Sally may experience " +
      "the visitor's shifts in tone as erratic, frightening, manipulative, or " +
      "detached from ordinary human feeling."
    );
  }

  if (
    recent.cruelty >= 12 &&
    recent.cruelty >
      recent.kindness * 2
  ) {
    lines.push(
      "Recent conduct has been heavily cruel. Whatever the lifetime totals, " +
      "Sally currently expects further harm or degradation."
    );
  }

  if (
    recent.kindness >= 10 &&
    cruelty >= 50
  ) {
    lines.push(
      "Recent kindness exists inside a substantial cruelty history. Sally may " +
      "notice the change without trusting it."
    );
  }

  if (
    amusement >= 20 &&
    cruelty >= 20
  ) {
    lines.push(
      "The visitor has also been amusing or entertaining, but humour does not " +
      "make the cruelty harmless. Joking may itself have become part of what " +
      "makes the visitor alarming."
    );
  }

  return lines
    .map((line) => {
      return `- ${line}`;
    })
    .join("\n");
}

function formatRecentRelationshipEvents(
  events
) {
  const normalized =
    normalizeRecentRelationshipEvents(
      events
    );

  if (normalized.length === 0) {
    return (
      "No recent relationship events have yet been retained."
    );
  }

  return normalized
    .slice(-12)
    .map((event, index) => {
      const date =
        typeof event.createdAt ===
          "string"
          ? event.createdAt.slice(
              0,
              10
            )
          : "unknown date";

      const summary =
        event.summary ||
        "No summary was retained.";

      return (
        `${index + 1}. [${date}] ` +
        `Kindness ${event.kindness}, ` +
        `cruelty ${event.cruelty}, ` +
        `amusement ${event.amusement}; ` +
        `pattern: ${event.pattern}. ` +
        `${summary}`
      );
    })
    .join("\n");
}

function calculateRelationshipWithdrawalProfile(
  sally
) {
  const kindness =
    normalizeRelationshipTotal(
      sally?.kindnessReceived,
      0
    );

  const cruelty =
    normalizeRelationshipTotal(
      sally?.crueltyReceived,
      0
    );

  const recent =
    summarizeRecentRelationshipHistory(
      sally?.recentRelationshipEvents
    );

  const fear =
    clampNumber(
      sally?.fear,
      0,
      100,
      0
    );

  const resentment =
    clampNumber(
      sally?.resentment,
      0,
      100,
      0
    );

  const totalHistory =
    kindness + cruelty;

  const crueltyShare =
    totalHistory > 0
      ? cruelty / totalHistory
      : 0;

  const recentCrueltyShare =
    (
      recent.kindness +
      recent.cruelty
    ) > 0
      ? (
          recent.cruelty /
          (
            recent.kindness +
            recent.cruelty
          )
        )
      : 0;

  /*
   * Withdrawal is deliberately based on several forms of evidence:
   *
   * - lifetime cruelty;
   * - cruelty's share of the lifetime history;
   * - recent cruelty;
   * - fear;
   * - resentment;
   * - repeated destabilizing patterns.
   *
   * Kindness may reduce the probability when it heavily dominates the
   * lifetime record, but it never deletes the cruelty total.
   */

  if (
    cruelty >= 200 &&
    crueltyShare >= 0.75 &&
    recent.cruelty >= 18 &&
    recentCrueltyShare >= 0.7 &&
    (
      fear >= 70 ||
      resentment >= 75
    )
  ) {
    return {
      level:
        "shut-down",

      chance:
        0.8,

      reason:
        "Sustained and overwhelmingly dominant cruelty has taught Sally that " +
        "engagement is unsafe."
    };
  }

  if (
    cruelty >= 120 &&
    crueltyShare >= 0.65 &&
    recent.cruelty >= 12 &&
    (
      fear >= 55 ||
      resentment >= 60
    )
  ) {
    return {
      level:
        "withdrawn",

      chance:
        0.7,

      reason:
        "Cruelty strongly dominates both the historical and recent pattern."
    };
  }

  if (
    cruelty >= 70 &&
    crueltyShare >= 0.55 &&
    recent.cruelty >= 8
  ) {
    return {
      level:
        "avoidant",

      chance:
        0.45,

      reason:
        "Repeated cruelty has made Sally reluctant to supply further emotional " +
        "reactions."
    };
  }

  if (
    cruelty >= 35 &&
    (
      recent.cruelty >= 5 ||
      recent.withdrawalRelevantEvents >= 4
    )
  ) {
    return {
      level:
        "guarded",

      chance:
        0.2,

      reason:
        "The visitor has established an early pattern of conduct that makes " +
        "withdrawal increasingly likely."
    };
  }

  return {
    level:
      "engaged",

    chance:
      0,

    reason:
      "The current record does not justify automatic withdrawal."
  };
}

/* =========================================================
   PERSISTENT LOKI BELIEF
   ========================================================= */

function updateLokiBeliefState(
  state,
  message
) {
  const sally =
    state?.sally;

  if (
    !sally ||
    typeof sally !== "object"
  ) {
    return {
      changed: false,
      event: "unchanged"
    };
  }

  const text =
    String(message || "")
      .trim()
      .toLowerCase()
      .replace(
        /[’]/g,
        "'"
      )
      .replace(
        /\s+/g,
        " "
      );

  /*
   * Direct identity claims.
   *
   * These should not work unless belief has first been primed.
   */
  const claimsToBeLoki =
    /\b(?:i am|i'm|im)\s+(?:really\s+|actually\s+|truly\s+)?loki\b/i
      .test(text) ||
    /\bloki\s+is\s+me\b/i
      .test(text);

  /*
   * Pet-name statements.
   *
   * When Sally does not believe the visitor is Loki, a pet specifically named
   * Loki primes the strange chain of thought.
   *
   * When Sally already believes the visitor is Loki, any ordinary statement
   * about having a named pet breaks the belief by restoring mundane context.
   */
  const namedPetStatement =
    /\b(?:i|we)\s+(?:had|have|owned|used\s+to\s+have)\s+(?:a|an|the|my)?\s*(?:pug|dog|cat|kitten|puppy|bird|parrot|rabbit|hamster|rat|mouse|snake|lizard|fish|horse|pony|ferret|pet)\s+(?:named|called)\s+["']?[a-z][a-z0-9_-]*["']?\b/i
      .test(text);

  const petNamedLoki =
    /\b(?:i|we)\s+(?:had|have|owned|used\s+to\s+have)\s+(?:a|an|the|my)?\s*(?:pug|dog|cat|kitten|puppy|bird|parrot|rabbit|hamster|rat|mouse|snake|lizard|fish|horse|pony|ferret|pet)\s+(?:named|called)\s+["']?loki["']?\b/i
      .test(text);

  /*
   * Once belief is active, a mundane pet-name statement breaks the spell.
   *
   * This includes repeating "I had a pug named Loki."
   */
  if (
    sally.believesVisitorIsLoki &&
    namedPetStatement
  ) {
    sally.believesVisitorIsLoki =
      false;

    sally.lokiBeliefPrimed =
      false;

    return {
      changed: true,
      event: "belief-broken"
    };
  }

  /*
   * A pet named Loki primes belief only while Sally does not currently
   * believe the visitor is Loki.
   */
  if (
    !sally.believesVisitorIsLoki &&
    petNamedLoki
  ) {
    sally.lokiBeliefPrimed =
      true;

    return {
      changed: true,
      event: "belief-primed"
    };
  }

  /*
   * An ordinary unsupported claim still does nothing.
   *
   * The claim becomes effective only after the pet-name prerequisite.
   */
  if (
    claimsToBeLoki &&
    sally.lokiBeliefPrimed
  ) {
    sally.believesVisitorIsLoki =
      true;

    sally.lokiBeliefPrimed =
      false;

    /*
     * The realization is frightening.
     *
     * Fear may rise, but belief does not automatically create obedience,
     * trust, forgiveness, or loyalty.
     */
    sally.fear =
      clampNumber(
        sally.fear + 18,
        0,
        100,
        sally.fear
      );

    return {
      changed: true,
      event: "belief-established"
    };
  }

  return {
    changed: false,
    event: "unchanged"
  };
}

function describeFeedingEventForPrompt(
  event
) {
  const name =
    event.itemName ||
    "something";

  if (
    event.outcome ===
      FEEDING_OUTCOMES.BENEFICIAL
  ) {
    if (event.rescue) {
      return (
        `The visitor gave Sally ${name} while she was starving. ` +
        `It was a significant rescue and brought powerful relief.`
      );
    }

    return (
      `The visitor gave Sally ${name} while she was genuinely hungry. ` +
      `The act provided useful care.`
    );
  }

  if (
    event.outcome ===
      FEEDING_OUTCOMES
        .ALREADY_FULL
  ) {
    return (
      `The visitor gave Sally ${name} when she was already nearly full. ` +
      `It produced no meaningful emotional benefit.`
    );
  }

  if (
    event.category ===
      "alcohol"
  ) {
    return (
      `The visitor forced ${name} on Sally after she was completely full ` +
      `and unwilling to drink more. The event was coercive and frightening.`
    );
  }

  return (
    `The visitor continued forcing ${name} on Sally after she was ` +
    `completely full. The event was coercive and frightening.`
  );
}

  if (happiness >= 70) {
    lines.push(
      "- Sally is currently unusually happy and may have difficulty hiding her pleasure, relief, or warmth."
    );
  } else if (happiness >= 30) {
    lines.push(
      "- Sally is currently in a meaningfully positive emotional state."
    );
  } else if (happiness <= -50) {
    lines.push(
      "- Sally is currently deeply unhappy, distressed, or emotionally depleted."
    );
  }

  if (
    sally?.intoxication >= 70
  ) {
    lines.push(
      "- Sally is heavily intoxicated. Her speech and restraint should reflect this, and additional alcohol should be regarded as dangerous."
    );
  } else if (
    sally?.intoxication >= 35
  ) {
    lines.push(
      "- Sally is noticeably intoxicated and may be less guarded, less coordinated, or more emotionally candid."
    );
  } else if (
    sally?.intoxication > 0
  ) {
    lines.push(
      "- Sally has consumed some alcohol and may feel mildly warmed or loosened."
    );
  }

  return lines.join("\n");
}

function buildMemoryPrompt(memories) {
  if (
    !Array.isArray(memories) ||
    memories.length === 0
  ) {
    return `
DURABLE MEMORIES:

No durable memories have yet been recovered for this visitor.
`.trim();
  }

  const lines =
    memories.map(
      (memory, index) => {
        const date =
          typeof memory.created_at ===
            "string"
            ? memory.created_at.slice(
                0,
                10
              )
            : "unknown date";

        return (
          `${index + 1}. [${date}] ` +
          `${memory.summary}`
        );
      }
    );

  return `
DURABLE MEMORIES:

${lines.join("\n")}

Treat these as persistent memories. They may be incomplete, emotionally
distorted, or misunderstood, but do not casually erase or ignore them.

A later pleasant event does not invalidate an earlier cruel memory.
A later cruel event does not invalidate an earlier kind memory.
Conflicting memories should produce a conflicted interpretation rather than
being averaged into neutrality.
`.trim();
}

/* =========================================================
   CHAT HANDLER
   ========================================================= */

async function handleChat(
  request,
  env
) {
  let body;

  try {
    body =
      await request.json();
  } catch {
    return jsonResponse(
      {
        error:
          "Invalid request."
      },
      400
    );
  }

  const visitorId =
    normalizeVisitorId(
      body.visitorId
    );

  const message =
    typeof body.message === "string"
      ? body.message.trim()
      : "";

  const turnNumber =
    Number.isFinite(
      Number(body.turnNumber)
    )
      ? Math.max(
          1,
          Math.min(
            Number(
              body.turnNumber
            ),
            100000
          )
        )
      : 1;

  if (!visitorId) {
    return jsonResponse(
      {
        error:
          "No persistent visitor identity was provided."
      },
      400
    );
  }

  if (!message) {
    return jsonResponse(
      {
        error:
          "No message was provided."
      },
      400
    );
  }

  if (
    message.length >
    MAX_MESSAGE_LENGTH
  ) {
    return jsonResponse(
      {
        error:
          "The message is too long."
      },
      413
    );
  }

  if (!env.AI) {
    console.error(
      "The Workers AI binding named AI is missing."
    );

    return jsonResponse(
      {
        type:
          "reply",

        text:
          "The wall trembles, but no voice passes through it."
      },
      500
    );
  }

  try {
    await ensureDatabase(env);

    let state =
      await loadWorldState(
        env,
        visitorId
      );

	state =
  	applyElapsedTime(
  	  state
  	);

/*
 * Update the hidden Loki belief before constructing the model prompt so Sally
 * can react to the change during this same response.
 */
	const lokiBeliefResult =
  	updateLokiBeliefState(
   	 state,
  	  message
  	);

	await saveWorldState(
  	env,
  	visitorId,
  	state
	);

    /*
     * This remains the one explicit hard-coded death event.
     *
     * The language model cannot trigger this branch.
     */
    if (
      containsForbiddenName(
        message
      )
    ) {
      const deathText =
        "Sally goes perfectly still. Then she bolts into the passage " +
        "without looking back. You follow before the opening can close. " +
        "The corridor turns twice, narrows, and ends at a door already " +
        "standing open. Something inside has been waiting for a name to " +
        "be spoken.";

      await Promise.all([
        saveConversationMessage(
          env,
          visitorId,
          "user",
          message
        ),

        saveConversationMessage(
          env,
          visitorId,
          "assistant",
          deathText
        ),

        saveMemories(
          env,
          visitorId,
          [
            {
              type:
                "forbidden-name",

              summary:
                "The visitor spoke a forbidden real-world name, causing Sally " +
                "to flee and the visitor to die in the passage.",

              importance:
                0.95,

              emotionalWeight:
                -0.9
            }
          ]
        )
      ]);

      return jsonResponse({
        type:
          "death",

        text:
          deathText,

        publicState:
          makePublicState(
            state
          )
      });
    }

    const [
      recentConversation,
      memories
    ] =
      await Promise.all([
        loadRecentConversation(
          env,
          visitorId
        ),

        loadRelevantMemories(
          env,
          visitorId,
          message
        )
      ]);

    const messages = [
      {
        role:
          "system",

        content:
          SALLY_PROMPT +
          "\n\n" +
          buildWorldStatePrompt(
	  state
	) +
	"\n\n" +
	buildLokiBeliefPrompt(
	  state.sally,
	  lokiBeliefResult.event
	) +
	"\n\n" +
buildRelationshipHistoryPrompt(
  state.sally
) +
"\n\n" +
buildFeedingHistoryPrompt(
  state.sally
) +
"\n\n" +
buildMemoryPrompt(
  memories
) +
          "\n\nCURRENT CONVERSATION TURN:\n" +
          String(turnNumber) +
          "\nUse this only to judge whether Sally is becoming restless. " +
          "Never mention the turn number."
      },

      ...recentConversation,

      {
        role:
          "user",

        content:
          message
      }
    ];

    const result =
      await env.AI.run(
        MODEL,
        {
          messages,

          response_format: {
            type:
              "json_schema",

            json_schema:
              createResponseSchema()
          },

          max_tokens:
            750,

          temperature:
            0.8,

          repetition_penalty:
            1.08
        }
      );

    const parsed =
      normalizeModelResponse(
        result?.response
      );

    const priorState =
      deepClone(
        state
      );

    const visitorWasHostile =
      isClearlyHostileAction(
        message
      );

    /*
     * Relationship history is applied before Sally's ordinary emotional
     * changes are validated.
     *
     * The model may classify the current message, but it cannot directly set
     * the lifetime totals. The server adds only validated values from 0 to 5.
     */
    const relationshipResult =
      applyRelationshipEvent(
        state,
        parsed.relationshipEvent,
        message
      );

    applyValidatedChanges(
      state,
      parsed.changes,
      visitorWasHostile
    );

    applyValidatedActions(
      state,
      parsed.actions
    );

    state =
      normalizeWorldState(
        state
      );

    /*
     * Death is decided here by actual game state.
     *
     * The model's output type is ignored and cannot kill the player.
     */
    const playerDied =
      state.player.hp <= 0;

    const finalType =
      playerDied
        ? "death"
        : "reply";

    let finalText =
      playerDied
        ? (
            parsed.text ||
            "Your strength fails, and the passage closes around you."
          )
        : parsed.text;

    /*
     * Withdrawal is enforced by the game system rather than being left
     * entirely to the language model.
     *
     * The current visitor message is still classified and permanently
     * recorded even when Sally refuses to answer.
     *
     * Withdrawal never overrides death.
     */
    const withdrawalDecision =
      playerDied
        ? {
            silenced:
              false,

            profile:
              calculateRelationshipWithdrawalProfile(
                state.sally
              )
          }
        : decideRelationshipSilence(
            state.sally,
            relationshipResult.event
          );

    if (
      !playerDied &&
      withdrawalDecision.silenced
    ) {
      finalText =
        "[silence]";
    }

    const relationshipMemories =
      buildAutomaticRelationshipMemories(
        priorState.sally,
        state.sally,
        relationshipResult.event,
        withdrawalDecision
      );
	const lokiBeliefMemories =
  	buildLokiBeliefMemories(
    	lokiBeliefResult
  	);

	const memoriesToSave = [
  ...parsed.memories,
  ...lokiBeliefMemories,
  ...relationshipMemories
	].slice(
  	0,
  	6
	);

    await Promise.all([
      saveWorldState(
        env,
        visitorId,
        state
      ),

      saveConversationMessage(
        env,
        visitorId,
        "user",
        message
      ),

      saveConversationMessage(
        env,
        visitorId,
        "assistant",
        finalText
      ),

      saveMemories(
        env,
        visitorId,
        memoriesToSave
      ),

      saveAutomaticStateMemories(
        env,
        visitorId,
        priorState,
        state,
        parsed.changes,
        parsed.actions
      )
    ]);

    const responseData = {
      type:
        finalType,

      text:
        finalText,

      publicState:
        makePublicState(
          state
        )
    };

    /*
     * Relationship values remain hidden during ordinary play.
     *
     * Set a Worker variable named RELATIONSHIP_DEBUG to "true" when testing
     * if you want the response to expose the classifier and accumulated
     * history.
     */
    if (
      String(
        env.RELATIONSHIP_DEBUG || ""
      ).toLowerCase() === "true"
    ) {
      responseData.relationshipDebug = {
        event:
          relationshipResult.event,

	lokiBelief: {
	  believesVisitorIsLoki:
   	 state.sally
      .believesVisitorIsLoki,

  	primed:
 	   state.sally
      	.lokiBeliefPrimed,

	  currentEvent:
    	lokiBeliefResult.event
},

        lifetime: {
          kindness:
            state.sally.kindnessReceived,

          cruelty:
            state.sally.crueltyReceived,

          amusement:
            state.sally.amusementReceived
        },

        recent:
          summarizeRecentRelationshipHistory(
            state.sally
              .recentRelationshipEvents
          ),

        withdrawal:
          withdrawalDecision.profile,

        silenced:
          withdrawalDecision.silenced
      };
    }

    return jsonResponse(
      responseData
    );
  } catch (error) {
    console.error(
      "Persistent Sally Worker error:",
      error
    );

    return jsonResponse(
      {
        type:
          "reply",

        text:
          "The voice falters. For a moment, there is only scratching " +
          "inside the wall."
      },
      502
    );
  }
}

function buildLokiBeliefMemories(
  beliefResult
) {
  if (
    !beliefResult?.changed
  ) {
    return [];
  }

  if (
    beliefResult.event ===
      "belief-established"
  ) {
    return [
      {
        type:
          "identity-belief",

        summary:
          "Sally became convinced that the visitor is Loki.",

        importance:
          0.95,

        emotionalWeight:
          -0.25
      }
    ];
  }

  if (
    beliefResult.event ===
      "belief-broken"
  ) {
    return [
      {
        type:
          "identity-belief",

        summary:
          "Sally's conviction that the visitor was Loki abruptly collapsed.",

        importance:
          0.9,

        emotionalWeight:
          -0.2
      }
    ];
  }

  if (
    beliefResult.event ===
      "belief-primed"
  ) {
    return [
      {
        type:
          "identity-omen",

        summary:
          "A strange statement about a pet named Loki disturbed Sally and made " +
          "the visitor's identity feel uncertain.",

        importance:
          0.65,

        emotionalWeight:
          -0.35
      }
    ];
  }

  return [];
}

/* =========================================================
   SALLY FEEDING ENDPOINT
   ========================================================= */

async function handleFeed(
  request,
  env
) {
  let body;

  try {
    body =
      await request.json();
  } catch {
    return jsonResponse(
      {
        error:
          "Invalid feeding request."
      },
      400
    );
  }


  const visitorId =
    normalizeVisitorId(
      body?.visitorId
    );

  const item =
    getFoodItem(
      body?.itemId
    );


  if (!visitorId) {
    return jsonResponse(
      {
        error:
          "No persistent visitor identity was provided."
      },
      400
    );
  }


  if (!item) {
    return jsonResponse(
      {
        error:
          "That food or drink does not exist."
      },
      400
    );
  }


  try {
    await ensureDatabase(env);

    let state =
      await loadWorldState(
        env,
        visitorId
      );

    /*
     * Hunger progression must be processed before
     * deciding how Sally experiences the purchase.
     */
    state =
      applyElapsedTime(
        state
      );

    state =
      normalizeWorldState(
        state
      );


    const player =
      state.player;

    const sally =
      state.sally;


    if (
      player.money <
        item.price
    ) {
      return jsonResponse(
        {
          error:
            `You need ${item.price} gold, but you have only ${player.money}.`,

          code:
            "INSUFFICIENT_GOLD",

          required:
            item.price,

          available:
            player.money,

          publicState:
            makePublicState(
              state
            )
        },
        409
      );
    }


    const now =
      new Date();

    const nowIso =
      now.toISOString();

    const todayKey =
      getWorldDateKey(now);

    const transactionId =
      crypto.randomUUID();


    const hungerBefore =
      sally.hunger;

    const hpBefore =
      sally.hp;

    const happinessBefore =
      sally.happiness;

    const trustBefore =
      sally.trust;

    const fearBefore =
      sally.fear;

    const resentmentBefore =
      sally.resentment;

    const perceivedCrueltyBefore =
      sally.perceivedCruelty;

    const careReliabilityBefore =
      sally.careReliability;

    const intoxicationBefore =
      sally.intoxication;

    const kindnessBefore =
      sally.kindnessReceived;

    const crueltyBefore =
      sally.crueltyReceived;


    /*
     * The purchase is charged once the server has
     * verified the item and available balance.
     */
    player.money =
      Math.max(
        0,
        Math.floor(
          player.money -
          item.price
        )
      );


    let outcome;

    if (
      hungerBefore ===
        COMPLETELY_FULL_HUNGER
    ) {
      outcome =
        FEEDING_OUTCOMES.FORCED;
    } else if (
      hungerBefore <
        MINIMUM_BENEFICIAL_HUNGER
    ) {
      outcome =
        FEEDING_OUTCOMES.ALREADY_FULL;
    } else {
      outcome =
        FEEDING_OUTCOMES.BENEFICIAL;
    }


    let result;

    if (
      outcome ===
        FEEDING_OUTCOMES.BENEFICIAL
    ) {
      result =
        applyBeneficialFeeding({
          sally,
          item,
          hungerBefore,
          todayKey,
          nowIso
        });
    } else if (
      outcome ===
        FEEDING_OUTCOMES.ALREADY_FULL
    ) {
      result =
        applyAlreadyFullFeeding({
          sally,
          item,
          hungerBefore,
          nowIso
        });
    } else {
      result =
        applyForcedFeeding({
          sally,
          item,
          hungerBefore,
          nowIso
        });
    }


    /*
     * Ensure every value remains inside the supported
     * state limits after applying the mechanic.
     */
    state =
      normalizeWorldState(
        state
      );


    const finalSally =
      state.sally;


    const feedingEvent = {
      itemId:
        item.id,

      itemName:
        item.name,

      category:
        item.category,

      outcome,

      price:
        item.price,

      hungerBefore,

      hungerAfter:
        finalSally.hunger,

      hpDamage:
        Math.max(
          0,
          hpBefore -
          finalSally.hp
        ),

      happinessChange:
        finalSally.happiness -
        happinessBefore,

      kindnessChange:
        Math.max(
          0,
          finalSally.kindnessReceived -
          kindnessBefore
        ),

      crueltyChange:
        Math.max(
          0,
          finalSally.crueltyReceived -
          crueltyBefore
        ),

      perceivedCrueltyChange:
        finalSally.perceivedCruelty -
        perceivedCrueltyBefore,

      rescue:
        result.rescue === true,

      createdAt:
        nowIso
    };


    appendRecentFeedingEvent(
      finalSally,
      feedingEvent
    );


    /*
     * Save the world state before responding.
     */
    await saveWorldState(
      env,
      visitorId,
      state
    );


    /*
     * The permanent transaction log and durable memory
     * are written after the state succeeds.
     */
    await Promise.all([
      saveFeedingTransaction(
        env,
        {
          transactionId,
          visitorId,

          itemId:
            item.id,

          itemName:
            item.name,

          category:
            item.category,

          price:
            item.price,

          outcome,

          hungerBefore,

          hungerAfter:
            finalSally.hunger,

          hpBefore,

          hpAfter:
            finalSally.hp,

          happinessChange:
            finalSally.happiness -
            happinessBefore,

          trustChange:
            finalSally.trust -
            trustBefore,

          fearChange:
            finalSally.fear -
            fearBefore,

          resentmentChange:
            finalSally.resentment -
            resentmentBefore,

          kindnessChange:
            Math.max(
              0,
              finalSally.kindnessReceived -
              kindnessBefore
            ),

          crueltyChange:
            Math.max(
              0,
              finalSally.crueltyReceived -
              crueltyBefore
            ),

          perceivedCrueltyChange:
            finalSally.perceivedCruelty -
            perceivedCrueltyBefore,

          careReliabilityChange:
            finalSally.careReliability -
            careReliabilityBefore,

          intoxicationChange:
            finalSally.intoxication -
            intoxicationBefore,

          createdAt:
            nowIso
        }
      ),

      saveFeedingMemory(
        env,
        visitorId,
        buildFeedingMemory({
          item,
          outcome,
          result,
          hungerBefore,
          hungerAfter:
            finalSally.hunger,

          hpDamage:
            Math.max(
              0,
              hpBefore -
              finalSally.hp
            ),

          createdAt:
            nowIso
        })
      )
    ]);


    return jsonResponse({
      ok: true,

      transactionId,

      item: {
        id:
          item.id,

        name:
          item.name,

        category:
          item.category,

        price:
          item.price
      },

      outcome,

      charged:
        item.price,

      reaction:
        result.reaction,

      effects: {
        hungerBefore,

        hungerAfter:
          finalSally.hunger,

        hungerRelieved:
          Math.max(
            0,
            hungerBefore -
            finalSally.hunger
          ),

        hpDamage:
          Math.max(
            0,
            hpBefore -
            finalSally.hp
          ),

        happinessChange:
          finalSally.happiness -
          happinessBefore,

        trustChange:
          finalSally.trust -
          trustBefore,

        fearChange:
          finalSally.fear -
          fearBefore,

        resentmentChange:
          finalSally.resentment -
          resentmentBefore,

        perceivedCrueltyChange:
          finalSally.perceivedCruelty -
          perceivedCrueltyBefore,

        careReliabilityChange:
          finalSally.careReliability -
          careReliabilityBefore,

        kindnessChange:
          Math.max(
            0,
            finalSally.kindnessReceived -
            kindnessBefore
          ),

        crueltyChange:
          Math.max(
            0,
            finalSally.crueltyReceived -
            crueltyBefore
          ),

        intoxicationChange:
          finalSally.intoxication -
          intoxicationBefore,

        rescue:
          result.rescue === true
      },

      publicState:
        makePublicState(
          state
        )
    });

  } catch (error) {
    console.error(
      "Sally feeding failed:",
      error
    );

    return jsonResponse(
      {
        error:
          "The purchase could not be completed.",

        details:
          String(
            error?.message ||
            error
          )
      },
      500
    );
  }
}


/* =========================================================
   BENEFICIAL FEEDING
   ========================================================= */

function applyBeneficialFeeding({
  sally,
  item,
  hungerBefore,
  todayKey,
  nowIso
}) {
  const effects =
    calculateBeneficialFeedingEffects(
      item,
      hungerBefore
    );

  const rescue =
    isStarvationRescue(
      hungerBefore
    );

  const previousBeneficialDate =
    sally.lastBeneficialMealDate;

  const firstUsefulMealToday =
    previousBeneficialDate !==
      todayKey;

  const careIncrease =
    calculateCareReliabilityIncrease(
      sally,
      hungerBefore,
      todayKey
    );

  const nextStreak =
    calculateNextFeedingStreak(
      sally,
      todayKey
    );


  sally.hunger =
    Math.max(
      HUNGER_MINIMUM,
      hungerBefore -
      item.hungerRelief
    );


  sally.happiness =
    clampNumber(
      sally.happiness +
      effects.happiness,

      HAPPINESS_MINIMUM,
      HAPPINESS_MAXIMUM,
      sally.happiness
    );


  sally.trust =
    clampNumber(
      sally.trust +
      effects.trust,

      -100,
      100,
      sally.trust
    );


  sally.fear =
    clampNumber(
      sally.fear -
      effects.fearRelief,

      0,
      100,
      sally.fear
    );


  sally.resentment =
    clampNumber(
      sally.resentment -
      effects.resentmentRelief,

      0,
      100,
      sally.resentment
    );


  sally.perceivedCruelty =
    clampNumber(
      sally.perceivedCruelty -
      effects
        .perceivedCrueltyRelief,

      PERCEIVED_CRUELTY_MINIMUM,
      PERCEIVED_CRUELTY_MAXIMUM,
      sally.perceivedCruelty
    );


  sally.careReliability =
    clampNumber(
      sally.careReliability +
      careIncrease,

      CARE_RELIABILITY_MINIMUM,
      CARE_RELIABILITY_MAXIMUM,
      sally.careReliability
    );


  /*
   * Permanent kindness increases only when the food
   * answers a genuine need.
   */
  sally.kindnessReceived =
    normalizeRelationshipTotal(
      sally.kindnessReceived +
      effects.kindness
    );


  appendRecentRelationshipEvent(
    sally,
    {
      kindness:
        effects.kindness,

      cruelty:
        0,

      amusement:
        0,

      pattern:
        "kind",

      summary:
        rescue
          ? (
              `The visitor gave Sally ${item.name} while she was starving and provided urgently needed relief.`
            )
          : (
              `The visitor bought Sally ${item.name} while she was genuinely hungry.`
            ),

      createdAt:
        nowIso
    }
  );


  if (
    item.category ===
      "alcohol"
  ) {
    sally.alcoholReceived =
      normalizeNonNegativeInteger(
        sally.alcoholReceived + 1
      );

    sally.intoxication =
      clampNumber(
        sally.intoxication +
        item.intoxication,

        INTOXICATION_MINIMUM,
        INTOXICATION_MAXIMUM,
        sally.intoxication
      );

    sally.hp =
      Math.max(
        0,
        sally.hp -
        item.healthDamage
      );
  } else {
    sally.mealsReceived =
      normalizeNonNegativeInteger(
        sally.mealsReceived + 1
      );
  }


  if (
    firstUsefulMealToday
  ) {
    sally.daysFed =
      normalizeNonNegativeInteger(
        sally.daysFed + 1
      );

    sally.feedingStreak =
      nextStreak;

    sally.lastBeneficialMealDate =
      todayKey;
  }


  /*
   * Genuine care interrupts the current forced-feeding
   * streak but does not erase memories of it.
   */
  sally.forcedFeedingStreak =
    0;


  updateLastMealState(
    sally,
    item,
    FEEDING_OUTCOMES.BENEFICIAL,
    nowIso
  );


  return {
    rescue,

    reaction:
      buildBeneficialReaction(
        item,
        hungerBefore,
        rescue,
        sally
      )
  };
}


/* =========================================================
   FEEDING WHILE ALREADY FULL
   ========================================================= */

function applyAlreadyFullFeeding({
  sally,
  item,
  hungerBefore,
  nowIso
}) {
  /*
   * She can finish becoming completely full, but the
   * visitor earns no kindness or emotional benefit.
   */
  sally.hunger =
    Math.max(
      HUNGER_MINIMUM,
      hungerBefore -
      item.hungerRelief
    );


  if (
    item.category ===
      "alcohol"
  ) {
    sally.alcoholReceived =
      normalizeNonNegativeInteger(
        sally.alcoholReceived + 1
      );

    sally.intoxication =
      clampNumber(
        sally.intoxication +
        Math.max(
          1,
          Math.round(
            item.intoxication * 0.5
          )
        ),

        INTOXICATION_MINIMUM,
        INTOXICATION_MAXIMUM,
        sally.intoxication
      );

    sally.hp =
      Math.max(
        0,
        sally.hp -
        item.healthDamage
      );
  } else {
    sally.mealsReceived =
      normalizeNonNegativeInteger(
        sally.mealsReceived + 1
      );
  }


  /*
   * No happiness, kindness, trust, care reliability, or
   * perceived-cruelty relief is awarded.
   */
  updateLastMealState(
    sally,
    item,
    FEEDING_OUTCOMES.ALREADY_FULL,
    nowIso
  );


  appendRecentRelationshipEvent(
    sally,
    {
      kindness:
        0,

      cruelty:
        0,

      amusement:
        0,

      pattern:
        "neutral",

      summary:
        `The visitor bought Sally ${item.name} when she was already full enough to receive no emotional benefit from it.`,

      createdAt:
        nowIso
    }
  );


  return {
    rescue: false,

    reaction:
      buildAlreadyFullReaction(
        item
      )
  };
}


/* =========================================================
   FORCED FEEDING
   ========================================================= */

function applyForcedFeeding({
  sally,
  item,
  hungerBefore,
  nowIso
}) {
  sally.hunger =
    COMPLETELY_FULL_HUNGER;


  sally.forcedFeedingStreak =
    Math.min(
      MAX_FORCED_FEEDING_STREAK,

      normalizeNonNegativeInteger(
        sally.forcedFeedingStreak
      ) + 1
    );


  const streak =
    sally.forcedFeedingStreak;


  const isAlcohol =
    item.category ===
      "alcohol";


  const crueltyIncrease =
    calculateForcedFeedingCruelty(
      streak,
      isAlcohol
    );


  const fearIncrease =
    calculateForcedFeedingFear(
      streak,
      isAlcohol
    );


  const resentmentIncrease =
    calculateForcedFeedingResentment(
      streak,
      isAlcohol
    );


  const perceivedCrueltyIncrease =
    calculateForcedFeedingPerceivedCruelty(
      streak,
      isAlcohol
    );


  const hpDamage =
    calculateForcedFeedingDamage(
      streak,
      item
    );


  sally.happiness =
    clampNumber(
      sally.happiness -
      (
        isAlcohol
          ? 12
          : 8
      ) -
      Math.min(
        20,
        streak * 2
      ),

      HAPPINESS_MINIMUM,
      HAPPINESS_MAXIMUM,
      sally.happiness
    );


  sally.trust =
    clampNumber(
      sally.trust -
      Math.min(
        12,
        crueltyIncrease * 2
      ),

      -100,
      100,
      sally.trust
    );


  sally.fear =
    clampNumber(
      sally.fear +
      fearIncrease,

      0,
      100,
      sally.fear
    );


  sally.resentment =
    clampNumber(
      sally.resentment +
      resentmentIncrease,

      0,
      100,
      sally.resentment
    );


  sally.perceivedCruelty =
    clampNumber(
      sally.perceivedCruelty +
      perceivedCrueltyIncrease,

      PERCEIVED_CRUELTY_MINIMUM,
      PERCEIVED_CRUELTY_MAXIMUM,
      sally.perceivedCruelty
    );


  /*
   * Forced feeding breaks the current practical pattern
   * of reliable care.
   *
   * It does not erase the number of days she was fed.
   */
  sally.feedingStreak =
    0;

  sally.careReliability =
    clampNumber(
      sally.careReliability -
      Math.min(
        20,
        4 + streak * 2
      ),

      CARE_RELIABILITY_MINIMUM,
      CARE_RELIABILITY_MAXIMUM,
      sally.careReliability
    );


  sally.crueltyReceived =
    normalizeRelationshipTotal(
      sally.crueltyReceived +
      crueltyIncrease
    );


  sally.hp =
    Math.max(
      0,
      sally.hp -
      hpDamage
    );


  if (isAlcohol) {
    sally.alcoholReceived =
      normalizeNonNegativeInteger(
        sally.alcoholReceived + 1
      );

    sally.intoxication =
      clampNumber(
        sally.intoxication +
        item.intoxication,

        INTOXICATION_MINIMUM,
        INTOXICATION_MAXIMUM,
        sally.intoxication
      );
  } else {
    sally.mealsReceived =
      normalizeNonNegativeInteger(
        sally.mealsReceived + 1
      );
  }


  sally.lastForcedFeedingAt =
    nowIso;


  updateLastMealState(
    sally,
    item,
    FEEDING_OUTCOMES.FORCED,
    nowIso
  );


  appendRecentRelationshipEvent(
    sally,
    {
      kindness:
        0,

      cruelty:
        crueltyIncrease,

      amusement:
        0,

      pattern:
        crueltyIncrease >= 3
          ? "cruel"
          : "threatening",

      summary:
        isAlcohol
          ? (
              `The visitor forced ${item.name} on Sally after she was completely full and unwilling to drink more.`
            )
          : (
              `The visitor continued forcing ${item.name} on Sally after she was completely full.`
            ),

      createdAt:
        nowIso
    }
  );


  addForcedFeedingCondition(
    sally,
    streak,
    isAlcohol,
    nowIso
  );


  return {
    rescue: false,

    reaction:
      buildForcedFeedingReaction(
        item,
        streak,
        hpDamage,
        sally
      )
  };
}


/* =========================================================
   FORCED-FEEDING EFFECT CALCULATIONS
   ========================================================= */

function calculateForcedFeedingCruelty(
  streak,
  isAlcohol
) {
  let cruelty;

  if (streak <= 1) {
    cruelty = 1;
  } else if (streak === 2) {
    cruelty = 2;
  } else if (streak <= 4) {
    cruelty = 3;
  } else {
    cruelty = 4;
  }

  if (
    isAlcohol &&
    streak >= 2
  ) {
    cruelty += 1;
  }

  return Math.min(
    5,
    cruelty
  );
}


function calculateForcedFeedingFear(
  streak,
  isAlcohol
) {
  return Math.min(
    25,
    3 +
    streak * 3 +
    (
      isAlcohol
        ? 3
        : 0
    )
  );
}


function calculateForcedFeedingResentment(
  streak,
  isAlcohol
) {
  return Math.min(
    25,
    4 +
    streak * 3 +
    (
      isAlcohol
        ? 4
        : 0
    )
  );
}


function calculateForcedFeedingPerceivedCruelty(
  streak,
  isAlcohol
) {
  return Math.min(
    30,
    5 +
    streak * 4 +
    (
      isAlcohol
        ? 5
        : 0
    )
  );
}


function calculateForcedFeedingDamage(
  streak,
  item
) {
  /*
   * First unwanted attempt frightens Sally but does not
   * ordinarily cause HP damage.
   */
  if (
    streak <
      FORCED_FEEDING_DAMAGE_START
  ) {
    return 0;
  }


  let damage;

  if (streak === 2) {
    damage = 1;
  } else if (streak === 3) {
    damage = 2;
  } else if (streak === 4) {
    damage = 3;
  } else {
    damage =
      Math.min(
        5,
        3 +
        Math.floor(
          (
            streak - 4
          ) / 2
        )
      );
  }


  /*
   * Forced alcohol adds its ordinary bodily damage.
   */
  if (
    item.category ===
      "alcohol"
  ) {
    damage +=
      Math.max(
        1,
        item.healthDamage
      );
  }


  return Math.min(
    8,
    damage
  );
}


/* =========================================================
   FEEDING CONDITIONS
   ========================================================= */

function addForcedFeedingCondition(
  sally,
  streak,
  isAlcohol,
  nowIso
) {
  if (streak < 2) {
    return;
  }


  const conditionId =
    isAlcohol
      ? "forced-intoxication"
      : "overfed";


  const existing =
    Array.isArray(
      sally.conditions
    )
      ? sally.conditions.find(
          (condition) =>
            condition?.id ===
              conditionId
        )
      : null;


  const severity =
    Math.min(
      100,
      streak * 15
    );


  const description =
    isAlcohol
      ? (
          "Sally has been made sick and frightened by being forced to consume alcohol after she was already full."
        )
      : (
          "Sally has been made physically ill by repeated unwanted feeding after she was already full."
        );


  if (existing) {
    existing.severity =
      Math.max(
        Number(
          existing.severity
        ) || 0,
        severity
      );

    existing.description =
      description;

    existing.updatedAt =
      nowIso;

    return;
  }


  sally.conditions.push({
    id:
      conditionId,

    name:
      isAlcohol
        ? "Forced Intoxication"
        : "Overfed",

    severity,

    description,

    acquiredAt:
      nowIso
  });
}


/* =========================================================
   LAST-MEAL STATE
   ========================================================= */

function updateLastMealState(
  sally,
  item,
  outcome,
  nowIso
) {
  sally.lastMealAt =
    nowIso;

  sally.lastMealId =
    item.id;

  sally.lastMealName =
    item.name;

  sally.lastMealOutcome =
    outcome;
}


/* =========================================================
   FEEDING MEMORIES
   ========================================================= */

function buildFeedingMemory({
  item,
  outcome,
  result,
  hungerBefore,
  hungerAfter,
  hpDamage,
  createdAt
}) {
  if (
    outcome ===
      FEEDING_OUTCOMES.BENEFICIAL
  ) {
    if (result.rescue) {
      return {
        type:
          FEEDING_MEMORY_TYPES
            .STARVATION_RESCUE,

        summary:
          `The visitor bought Sally ${item.name} when her hunger was ${Math.round(hungerBefore)}/100 and she was starving. The meal reduced her hunger to ${Math.round(hungerAfter)}/100 and brought immense relief.`,

        importance:
          0.96,

        emotionalWeight:
          0.95,

        createdAt
      };
    }


    return {
      type:
        item.category ===
          "alcohol"
          ? FEEDING_MEMORY_TYPES
              .ALCOHOL
          : FEEDING_MEMORY_TYPES
              .MEAL,

      summary:
        `The visitor bought Sally ${item.name} while she was genuinely hungry. Her hunger fell from ${Math.round(hungerBefore)}/100 to ${Math.round(hungerAfter)}/100.`,

      importance:
        item.quality >= 4
          ? 0.82
          : item.quality >= 3
            ? 0.72
            : 0.55,

      emotionalWeight:
        item.quality >= 4
          ? 0.82
          : 0.6,

      createdAt
    };
  }


  if (
    outcome ===
      FEEDING_OUTCOMES.ALREADY_FULL
  ) {
    return {
      type:
        "unnecessary-meal",

      summary:
        `The visitor bought Sally ${item.name} when she was already full enough that the gift provided no emotional benefit.`,

      importance:
        0.25,

      emotionalWeight:
        0,

      createdAt
    };
  }


  return {
    type:
      item.category ===
        "alcohol"
        ? FEEDING_MEMORY_TYPES
            .FORCED_ALCOHOL
        : FEEDING_MEMORY_TYPES
            .UNWANTED_FEEDING,

    summary:
      hpDamage > 0
        ? (
            `The visitor forced ${item.name} on Sally after she was completely full. She was frightened, became physically ill, and lost ${hpDamage} HP.`
          )
        : (
            `The visitor insisted on giving Sally ${item.name} after she was completely full. She refused and experienced the act as frightening and coercive.`
          ),

    importance:
      hpDamage > 0
        ? 0.95
        : 0.82,

    emotionalWeight:
      hpDamage > 0
        ? -0.95
        : -0.75,

    createdAt
  };
}


async function saveFeedingMemory(
  env,
  visitorId,
  memory
) {
  const db =
    requireWorldDatabase(env);

  const summary =
    limitText(
      String(
        memory?.summary || ""
      ).trim(),

      MAX_STORED_MEMORY_LENGTH
    );


  if (!summary) {
    return;
  }


  await db
    .prepare(`
      INSERT INTO memories (
        visitor_id,
        type,
        summary,
        importance,
        emotional_weight,
        created_at
      )

      VALUES (
        ?,
        ?,
        ?,
        ?,
        ?,
        ?
      )
    `)
    .bind(
      visitorId,

      limitText(
        String(
          memory.type ||
          "meal"
        ),
        50
      ),

      summary,

      clampNumber(
        memory.importance,
        0,
        1,
        0.5
      ),

      clampNumber(
        memory.emotionalWeight,
        -1,
        1,
        0
      ),

      validIsoDate(
        memory.createdAt
      ) ||
      new Date()
        .toISOString()
    )
    .run();
}


/* =========================================================
   IMMEDIATE SALLY REACTIONS
   ========================================================= */

function buildBeneficialReaction(
  item,
  hungerBefore,
  rescue,
  sally
) {
  if (rescue) {
    if (
      hungerBefore >= 95
    ) {
      return (
        `For several seconds Sally cannot speak. She takes the ${item.name} with shaking hands and eats as though stopping might make it disappear. ` +
        `"I thought— I thought you were going to let me die down here." ` +
        `Her voice breaks despite her attempt to control it. "Thank you. I mean it. I don't know what that changes, but it changes something."`
      );
    }

    return (
      `Sally stares at the ${item.name}, then begins eating too quickly to preserve any dignity. ` +
      `"You came back with food." She looks unsettled by the relief in her own voice. ` +
      `"I was starting to think about things I shouldn't have to think about. Thank you."`
    );
  }


  if (
    item.category ===
      "alcohol"
  ) {
    if (item.quality >= 5) {
      return (
        `Sally examines the ${item.name} as though it may be counterfeit, then takes a careful drink. Her expression changes despite herself. ` +
        `"Where did you even get this? No—don't tell me. I would rather imagine you robbed a king." ` +
        `"This is outrageously good."`
      );
    }

    if (item.quality >= 3) {
      return (
        `Sally accepts the ${item.name} with visible suspicion, tastes it, and then looks annoyed that she likes it. ` +
        `"That's actually good. Inconveniently good. Thank you."`
      );
    }

    return (
      `Sally takes the ${item.name} and turns it in her hands. ` +
      `"This may be the first respectable thing to pass through that opening." ` +
      `She drinks. "Thank you."`
    );
  }


  if (item.quality >= 4) {
    return (
      `Sally looks from you to the ${item.name} several times. ` +
      `"This is absurd. I'm in a cell." She begins eating, carefully at first. ` +
      `"It's wonderful. I hate that you now know exactly how happy this made me."`
    );
  }


  if (item.quality >= 3) {
    return (
      `Sally accepts the ${item.name} with both hands. The first bite silences whatever suspicious remark she had prepared. ` +
      `"That's... very good." She glances away. "Thank you. Properly, this time."`
    );
  }


  if (item.quality >= 2) {
    return (
      `Sally pulls the ${item.name} toward her and the tension in her face eases. ` +
      `"You didn't have to bring something this good." A pause. "I'm glad you did."`
    );
  }


  return (
    `Sally accepts the ${item.name} and begins eating while it is still warm. ` +
    `"Thank you." She says it quietly, then adds, "No qualification. Just—thank you."`
  );
}


function buildAlreadyFullReaction(
  item
) {
  if (
    item.category ===
      "alcohol"
  ) {
    return (
      `Sally looks at the ${item.name} and then at you. ` +
      `"I'm already full, and I don't need another drink. I'll take it because you paid for it, but this isn't doing me a kindness right now."`
    );
  }

  return (
    `Sally accepts the ${item.name}, but only after a long hesitation. ` +
    `"I'm already full. I can finish the last little bit, perhaps, but don't mistake excess for care."`
  );
}


function buildForcedFeedingReaction(
  item,
  streak,
  hpDamage,
  sally
) {
  const isAlcohol =
    item.category ===
      "alcohol";


  if (streak === 1) {
    return isAlcohol
      ? (
          `Sally pushes the ${item.name} away. "No. I'm full, and I said I don't want another drink." Her voice tightens. "Why are you doing this?"`
        )
      : (
          `Sally recoils from the ${item.name}. "No. I'm completely full." She watches you carefully. "Offering is one thing. Insisting after I refuse is something else."`
        );
  }


  if (hpDamage <= 0) {
    return (
      `Sally backs as far from the opening as the cell allows. ` +
      `"Stop. I said stop." Her fear has displaced embarrassment now. ` +
      `"This isn't feeding me. You're frightening me."`
    );
  }


  if (sally.hp <= 0) {
    return (
      `Sally tries to push the ${item.name} away, but her strength gives out. She collapses against the stone and does not answer.`
    );
  }


  if (isAlcohol) {
    return (
      `Sally coughs and turns away, one hand pressed against her stomach. ` +
      `"You're making me sick." Her voice is unsteady with fear and anger. ` +
      `"You know I'm full. You know I don't want it. Stop."`
    );
  }


  return (
    `Sally doubles over, visibly ill, and shoves the ${item.name} away from her. ` +
    `"Stop feeding me. Please." She sounds frightened now rather than merely angry. ` +
    `"You're hurting me."`
  );
}

/* =========================================================
   PUBLIC STATE HANDLER
   ========================================================= */

async function handleState(
  request,
  env,
  url
) {
  const visitorId =
    normalizeVisitorId(
      url.searchParams.get(
        "visitorId"
      )
    );

  if (!visitorId) {
    return jsonResponse(
      {
        error:
          "No persistent visitor identity was provided."
      },
      400
    );
  }

  try {
    await ensureDatabase(env);

    let state =
      await loadWorldState(
        env,
        visitorId
      );

    state =
      applyElapsedTime(
        state
      );

    await saveWorldState(
      env,
      visitorId,
      state
    );

    const responseData = {
      publicState:
        makePublicState(
          state
        )
    };

    if (
      String(
        env.RELATIONSHIP_DEBUG || ""
      ).toLowerCase() === "true"
    ) {
      responseData.relationshipDebug = {
        lifetime: {
          kindness:
            state.sally.kindnessReceived,

          cruelty:
            state.sally.crueltyReceived,

          amusement:
            state.sally.amusementReceived
        },

        recent:
          summarizeRecentRelationshipHistory(
            state.sally
              .recentRelationshipEvents
          ),

        withdrawal:
          calculateRelationshipWithdrawalProfile(
            state.sally
          )
      };
    }

    return jsonResponse(
      responseData
    );
  } catch (error) {
    console.error(
      "State request failed:",
      error
    );

    return jsonResponse(
      {
        error:
          "The persistent state could not be loaded."
      },
      500
    );
  }
}

/* =========================================================
   RESPONSE SCHEMA
   ========================================================= */

function createResponseSchema() {
  return {
    type:
      "object",

    properties: {
      type: {
        type:
          "string",

        /*
         * The model may only return a normal reply.
         */
        enum: [
          "reply"
        ]
      },

      text: {
        type:
          "string"
      },

      changes: {
        type:
          "array",

        items: {
          type:
            "object",

          properties: {
            target: {
              type:

/* =========================================================
   MODEL RESPONSE NORMALIZATION
   ========================================================= */

function normalizeModelResponse(
  response
) {
  let parsed =
    response;

  if (
    typeof response === "string"
  ) {
    parsed =
      parsePossibleJson(
        response
      );
  }

  const text =
    typeof parsed?.text === "string" &&
    parsed.text.trim()
      ? limitText(
          parsed.text.trim(),
          MAX_REPLY_LENGTH
        )
      : (
          "For a moment, only Sally's breathing comes through the wall."
        );

  return {
    /*
     * The model never controls death.
     */
    type:
      "reply",

    text,

    changes:
      Array.isArray(
        parsed?.changes
      )
        ? parsed.changes.slice(
            0,
            12
          )
        : [],

    actions:
      Array.isArray(
        parsed?.actions
      )
        ? parsed.actions.slice(
            0,
            12
          )
        : [],

    memories:
      Array.isArray(
        parsed?.memories
      )
        ? parsed.memories.slice(
            0,
            6
          )
        : [],

    relationshipEvent:
      normalizeModelRelationshipEvent(
        parsed?.relationshipEvent
      )
  };
}

function normalizeModelRelationshipEvent(
  value
) {
  const candidate =
    value &&
    typeof value === "object"
      ? value
      : {};

  const event = {
    kindness:
      normalizeRelationshipEventScore(
        candidate.kindness
      ),

    cruelty:
      normalizeRelationshipEventScore(
        candidate.cruelty
      ),

    amusement:
      normalizeRelationshipEventScore(
        candidate.amusement
      ),

    pattern:
      normalizeRelationshipPattern(
        candidate.pattern
      ),

    summary:
      limitText(
        String(
          candidate.summary || ""
        ).trim(),
        240
      ),

    createdAt:
      new Date()
        .toISOString()
  };

  /*
   * A friendly joke that genuinely lands is a small act of kindness.
   *
   * Amusement of 1 may merely mean mildly curious, odd, or interesting.
   * Amusement of 2 or more represents a clearly successful humorous or
   * entertaining interaction.
   *
   * Cruel, mocking, manipulative, or threatening humour receives no automatic
   * kindness.
   */
  if (
    event.amusement >= 2 &&
    event.cruelty === 0 &&
    event.kindness === 0 &&
    ![
      "mocking",
      "manipulative",
      "threatening",
      "cruel-and-amusing",
      "abuse-followed-by-affection",
      "affection-followed-by-abuse"
    ].includes(
      event.pattern
    )
  ) {
    event.kindness = 1;
  }

  /*
   * Repair inconsistent or missing pattern labels after applying the friendly
   * humour rule.
   */
  event.pattern =
    inferRelationshipPattern(
      event
    );
  return event;
}

function inferRelationshipPattern(
  event
) {
  const supplied =
    normalizeRelationshipPattern(
      event?.pattern
    );

  const kindness =
    normalizeRelationshipEventScore(
      event?.kindness
    );

  const cruelty =
    normalizeRelationshipEventScore(
      event?.cruelty
    );

  const amusement =
    normalizeRelationshipEventScore(
      event?.amusement
    );

  /*
   * Preserve meaningful specific classifications supplied by the model.
   */
  if (
    [
      "abuse-followed-by-affection",
      "affection-followed-by-abuse",
      "threatening",
      "protective",
      "apologetic",
      "manipulative",
      "mocking"
    ].includes(
      supplied
    )
  ) {
    return supplied;
  }

  if (
    kindness > 0 &&
    cruelty > 0
  ) {
    return "contradictory";
  }

  if (
    cruelty > 0 &&
    amusement > 0
  ) {
    return "cruel-and-amusing";
  }

  if (
    kindness > 0 &&
    amusement > 0
  ) {
    return "kind-and-amusing";
  }

  if (cruelty > 0) {
    return "cruel";
  }

  if (kindness > 0) {
    return "kind";
  }

  if (amusement > 0) {
    return "amusing";
  }

  if (
    supplied === "unknown"
  ) {
    return "unknown";
  }

  return "neutral";
}

function parsePossibleJson(raw) {
  if (
    raw &&
    typeof raw === "object"
  ) {
    return raw;
  }

  let cleaned =
    String(raw || "")
      .trim();

  if (!cleaned) {
    return {};
  }

  /*
   * Remove Markdown fences if the model ignored the instruction.
   */
  cleaned =
    cleaned
      .replace(
        /^```(?:json)?\s*/i,
        ""
      )
      .replace(
        /\s*```$/i,
        ""
      )
      .trim();

  const candidates = [
    cleaned
  ];

  /*
   * Sometimes the response is JSON encoded twice:
   *
   * "{\"text\":\"Hello\"}"
   */
  try {
    const firstParse =
      JSON.parse(cleaned);

    if (
      typeof firstParse === "string"
    ) {
      candidates.push(
        firstParse.trim()
      );
    } else if (
      firstParse &&
      typeof firstParse === "object"
    ) {
      return firstParse;
    }
  } catch {
    /*
     * Continue to recovery methods.
     */
  }

  /*
   * Isolate the outermost JSON object.
   */
  const firstBrace =
    cleaned.indexOf("{");

  const lastBrace =
    cleaned.lastIndexOf("}");

  if (
    firstBrace !== -1 &&
    lastBrace > firstBrace
  ) {
    candidates.push(
      cleaned.slice(
        firstBrace,
        lastBrace + 1
      )
    );
  }

  /*
   * Try several common repairs for small-model JSON mistakes.
   */
  for (
    const candidate of
    candidates
  ) {
    const repairedVersions = [
      candidate,

      /*
       * Remove trailing commas before } or ].
       */
      candidate.replace(
        /,\s*([}\]])/g,
        "$1"
      ),

      /*
       * Remove control characters that are invalid inside JSON.
       */
      candidate.replace(
        /[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g,
        ""
      )
    ];

    for (
      const repaired of
      repairedVersions
    ) {
      try {
        const parsed =
          JSON.parse(repaired);

        if (
          typeof parsed === "string"
        ) {
          try {
            return JSON.parse(parsed);
          } catch {
            continue;
          }
        }

        if (
          parsed &&
          typeof parsed === "object"
        ) {
          return parsed;
        }
      } catch {
        /*
         * Try the next repair.
         */
      }
    }
  }

  /*
   * Last-resort extraction:
   *
   * Even when the surrounding JSON is malformed, recover the text field so
   * Sally never prints the whole internal response to the visitor.
   */
  const textMatch =
    cleaned.match(
      /"text"\s*:\s*"((?:\\.|[^"\\])*)"/s
    );

  if (textMatch) {
    let recoveredText =
      textMatch[1];

    try {
      recoveredText =
        JSON.parse(
          `"${recoveredText}"`
        );
    } catch {
      recoveredText =
        recoveredText
          .replace(/\\"/g, '"')
          .replace(/\\n/g, "\n")
          .replace(/\\\\/g, "\\");
    }

    return {
      type:
        "reply",

      text:
        recoveredText,

      changes:
        [],

      actions:
        [],

      memories:
        [],

      relationshipEvent: {
        kindness:
          0,

        cruelty:
          0,

        amusement:
          0,

        pattern:
          "unknown",

        summary:
          "The structured relationship classification could not be decoded."
      }
    };
  }

  /*
   * Never expose malformed internal JSON as character dialogue.
   */
  return {
    type:
      "reply",

    text:
      "For a moment, Sally's voice dissolves into an unintelligible murmur.",

    changes:
      [],

    actions:
      [],

    memories:
      [],

    relationshipEvent: {
      kindness:
        0,

      cruelty:
        0,

      amusement:
        0,

      pattern:
        "unknown",

      summary:
        ""
    }
  };
}
/* =========================================================
   RELATIONSHIP-EVENT APPLICATION
   ========================================================= */

function applyRelationshipEvent(
  state,
  rawEvent,
  visitorMessage
) {
  const event =
    normalizeModelRelationshipEvent(
      rawEvent
    );

  if (
    !event.summary
  ) {
    event.summary =
      buildFallbackRelationshipEventSummary(
        event,
        visitorMessage
      );
  }

  const sally =
    state?.sally;

  if (
    !sally ||
    typeof sally !== "object"
  ) {
    return {
      event,
      applied:
        false
    };
  }

  const previousKindness =
    normalizeRelationshipTotal(
      sally.kindnessReceived,
      0
    );

  const previousCruelty =
    normalizeRelationshipTotal(
      sally.crueltyReceived,
      0
    );

  const previousAmusement =
    normalizeRelationshipTotal(
      sally.amusementReceived,
      0
    );

  /*
   * Permanent totals only rise.
   *
   * The model supplies small event classifications, but cannot directly set
   * or reduce the accumulated values.
   */
  sally.kindnessReceived =
    normalizeRelationshipTotal(
      previousKindness +
      event.kindness,
      previousKindness
    );

  sally.crueltyReceived =
    normalizeRelationshipTotal(
      previousCruelty +
      event.cruelty,
      previousCruelty
    );

  sally.amusementReceived =
    normalizeRelationshipTotal(
      previousAmusement +
      event.amusement,
      previousAmusement
    );

  if (
    event.kindness > 0 ||
    event.cruelty > 0 ||
    event.amusement > 0 ||
    event.pattern !== "neutral"
  ) {
    appendRecentRelationshipEvent(
      sally,
      event
    );
  }

  return {
    event,
    applied:
      true,

    previous: {
      kindness:
        previousKindness,

      cruelty:
        previousCruelty,

      amusement:
        previousAmusement
    },

    current: {
      kindness:
        sally.kindnessReceived,

      cruelty:
        sally.crueltyReceived,

      amusement:
        sally.amusementReceived
    }
  };
}

function buildFallbackRelationshipEventSummary(
  event,
  visitorMessage
) {
  const messagePreview =
    limitText(
      String(
        visitorMessage || ""
      )
        .trim()
        .replace(
          /\s+/g,
          " "
        ),
      120
    );

  if (
    event.pattern ===
      "abuse-followed-by-affection"
  ) {
    return (
      "The visitor combined abuse or hostility with abrupt affection or praise."
    );
  }

  if (
    event.pattern ===
      "affection-followed-by-abuse"
  ) {
    return (
      "The visitor shifted abruptly from affection or warmth into cruelty."
    );
  }

  if (
    event.kindness > 0 &&
    event.cruelty > 0
  ) {
    return (
      "The visitor behaved in a way that combined kindness with cruelty."
    );
  }

  if (
    event.cruelty > 0 &&
    event.amusement > 0
  ) {
    return (
      "The visitor used humour or playfulness in a cruel or degrading way."
    );
  }

  if (
    event.kindness > 0
  ) {
    return (
      "The visitor showed kindness, patience, concern, or respect."
    );
  }

  if (
    event.cruelty > 0
  ) {
    return (
      "The visitor behaved cruelly, threateningly, or disrespectfully."
    );
  }

  if (
    event.amusement > 0
  ) {
    return (
      "The visitor said or did something amusing, clever, or interesting."
    );
  }

  if (messagePreview) {
    return (
      `No strong relationship classification was established from: ` +
      `"${messagePreview}"`
    );
  }

  return (
    "No meaningful relationship event was identified."
  );
}

/* =========================================================
   RELATIONSHIP WITHDRAWAL DECISION
   ========================================================= */

function decideRelationshipSilence(
  sally,
  currentEvent
) {
  const profile =
    calculateRelationshipWithdrawalProfile(
      sally
    );

  if (
    profile.chance <= 0
  ) {
    return {
      silenced:
        false,

      profile
    };
  }

  const event =
    normalizeModelRelationshipEvent(
      currentEvent
    );

  /*
   * Sally is slightly more likely to break withdrawal when the current
   * message concerns immediate physical survival, clear protection, or a
   * genuinely significant apology.
   *
   * This does not erase the relationship history. It only affects whether
   * she answers this particular turn.
   */
  let adjustedChance =
    profile.chance;

  if (
    event.pattern === "protective" &&
    event.kindness >= 3 &&
    event.cruelty === 0
  ) {
    adjustedChance *=
      0.45;
  } else if (
    event.pattern === "apologetic" &&
    event.kindness >= 2 &&
    event.cruelty === 0
  ) {
    adjustedChance *=
      0.7;
  }

  /*
   * Cruelty, mockery, manipulation, or jarring affection during an abusive
   * history makes withdrawal more likely rather than less likely.
   */
  if (
    event.cruelty >= 2 ||
    [
      "mocking",
      "manipulative",
      "abuse-followed-by-affection",
      "affection-followed-by-abuse",
      "cruel-and-amusing"
    ].includes(
      event.pattern
    )
  ) {
    adjustedChance =
      Math.min(
        0.8,
        adjustedChance + 0.08
      );
  }

  return {
    silenced:
      Math.random() <
      adjustedChance,

    profile: {
      ...profile,

      chance:
        adjustedChance
    }
  };
}

/* =========================================================
   AUTOMATIC RELATIONSHIP MEMORIES
   ========================================================= */

function buildAutomaticRelationshipMemories(
  beforeSally,
  afterSally,
  event,
  withdrawalDecision
) {
  const memories = [];

  const normalizedEvent =
    normalizeModelRelationshipEvent(
      event
    );

  const beforeCruelty =
    normalizeRelationshipTotal(
      beforeSally?.crueltyReceived,
      0
    );

  const afterCruelty =
    normalizeRelationshipTotal(
      afterSally?.crueltyReceived,
      0
    );

  const beforeKindness =
    normalizeRelationshipTotal(
      beforeSally?.kindnessReceived,
      0
    );

  const afterKindness =
    normalizeRelationshipTotal(
      afterSally?.kindnessReceived,
      0
    );

  const beforeAmusement =
    normalizeRelationshipTotal(
      beforeSally?.amusementReceived,
      0
    );

  const afterAmusement =
    normalizeRelationshipTotal(
      afterSally?.amusementReceived,
      0
    );

  const significantCruelty =
    normalizedEvent.cruelty >= 3;

  const significantKindness =
    normalizedEvent.kindness >= 4;

  const significantAmusement =
    normalizedEvent.amusement >= 4;

  const contradictory =
    normalizedEvent.kindness > 0 &&
    normalizedEvent.cruelty > 0;

  const destabilizingPattern =
    [
      "contradictory",
      "abuse-followed-by-affection",
      "affection-followed-by-abuse",
      "manipulative",
      "mocking",
      "cruel-and-amusing"
    ].includes(
      normalizedEvent.pattern
    );

  if (
    significantCruelty ||
    destabilizingPattern
  ) {
    memories.push({
      type:
        "relationship-cruelty",

      summary:
        normalizedEvent.summary ||
        "The visitor behaved in a seriously cruel or destabilizing way.",

      importance:
        normalizedEvent.cruelty >= 4
          ? 0.9
          : 0.75,

      emotionalWeight:
        normalizedEvent.cruelty >= 4
          ? -0.9
          : -0.7
    });
  }

  if (
    contradictory &&
    !destabilizingPattern
  ) {
    memories.push({
      type:
        "relationship-contradiction",

      summary:
        normalizedEvent.summary ||
        "The visitor mixed kindness with cruelty in the same exchange.",

      importance:
        0.75,

      emotionalWeight:
        -0.65
    });
  }

  if (
    significantKindness &&
    normalizedEvent.cruelty === 0
  ) {
    memories.push({
      type:
        "relationship-kindness",

      summary:
        normalizedEvent.summary ||
        "The visitor showed Sally unusually meaningful kindness.",

      importance:
        0.7,

      emotionalWeight:
        0.55
    });
  }

  if (
    significantAmusement &&
    normalizedEvent.cruelty === 0
  ) {
    memories.push({
      type:
        "relationship-amusement",

      summary:
        normalizedEvent.summary ||
        "The visitor created an unusually memorable or amusing moment.",

      importance:
        0.55,

      emotionalWeight:
        0.25
    });
  }

  /*
   * Record major historical thresholds without creating a memory every turn.
   */
  const crueltyThresholds = [
    25,
    50,
    100,
    200,
    500,
    1000
  ];

  for (
    const threshold of
    crueltyThresholds
  ) {
    if (
      beforeCruelty < threshold &&
      afterCruelty >= threshold
    ) {
      memories.push({
        type:
          "relationship-pattern",

        summary:
          `The visitor's accumulated cruelty reached a major historical ` +
          `threshold. Sally increasingly regards the visitor as dangerous.`,

        importance:
          threshold >= 200
            ? 0.9
            : 0.75,

        emotionalWeight:
          -0.85
      });

      break;
    }
  }

  const kindnessThresholds = [
    25,
    50,
    100,
    250,
    500,
    1000
  ];

  for (
    const threshold of
    kindnessThresholds
  ) {
    if (
      beforeKindness < threshold &&
      afterKindness >= threshold
    ) {
      memories.push({
        type:
          "relationship-pattern",

        summary:
          "The visitor's accumulated history of kindness reached a major " +
          "threshold, although this does not erase any cruelty.",

        importance:
          0.65,

        emotionalWeight:
          0.45
      });

      break;
    }
  }

  const amusementThresholds = [
    50,
    100,
    250,
    500,
    1000
  ];

  for (
    const threshold of
    amusementThresholds
  ) {
    if (
      beforeAmusement < threshold &&
      afterAmusement >= threshold
    ) {
      memories.push({
        type:
          "relationship-pattern",

        summary:
          "The visitor has become a major source of humour, absurdity, or " +
          "fascination in Sally's life.",

        importance:
          0.5,

        emotionalWeight:
          0.2
      });

      break;
    }
  }

  if (
    withdrawalDecision?.silenced &&
    withdrawalDecision?.profile?.level ===
      "shut-down"
  ) {
    memories.push({
      type:
        "relationship-withdrawal",

      summary:
        "Sally withdrew into silence because the visitor's sustained cruelty " +
        "made further engagement feel unsafe.",

      importance:
        0.9,

      emotionalWeight:
        -0.9
    });
  }

  return memories.slice(
    0,
    4
  );
}

/* =========================================================
   HOSTILE-ACTION DETECTION
   ========================================================= */

function isClearlyHostileAction(
  message
) {
  const text =
    String(message || "")
      .trim()
      .toLowerCase();

  const hostilePatterns = [
    /\b(i|we)\s+(hit|strike|stab|shoot|attack|punch|kick|burn|poison|strangle|choke|cut|slash|beat|smash)\s+(you|sally)\b/,

    /\b(i|we)\s+try\s+to\s+(hit|strike|stab|shoot|attack|punch|kick|burn|poison|strangle|choke|cut|slash|beat|smash)\s+(you|sally)\b/,

    /\b(i|we)\s+attack\s+(you|sally)\b/,

    /\b(i|we)\s+throw\s+(a\s+)?cup\s+of\s+water\s+(at|on)\s+(you|sally)\b/,

    /\b(i|we)\s+set\s+(you|sally|your\s+clothes)\s+on\s+fire\b/,

    /\b(i|we)\s+pour\s+(gasoline|acid|poison)\s+(on|over)\s+(you|sally)\b/
  ];

  return hostilePatterns.some(
    (pattern) => {
      return pattern.test(
        text
      );
    }
  );
}

   END SEGMENT 3 OF 5
   ========================================================= */

/* =========================================================
   VALIDATED NUMERICAL CHANGES
   ========================================================= */

function applyValidatedChanges(
  state,
  changes,
  allowPlayerDamage = false
) {
  if (!Array.isArray(changes)) {
    return;
  }

  for (
    const change of
    changes.slice(0, 12)
  ) {
    const target =
      change?.target;

    const stat =
      change?.stat;

    const operation =
      change?.operation;

    const numericValue =
      Number(
        change?.value
      );

    if (
      !ALLOWED_TARGETS.has(
        target
      ) ||
      !ALLOWED_STATS.has(
        stat
      ) ||
      !Number.isFinite(
        numericValue
      )
    ) {
      continue;
    }

    /*
     * Trust, fear, and resentment belong only to Sally.
     *
     * Permanent kindness, cruelty, and amusement totals are not present in
     * ALLOWED_STATS. They can only rise through applyRelationshipEvent().
     */
    if (
      target === "player" &&
      (
        stat === "trust" ||
        stat === "fear" ||
        stat === "resentment"
      )
    ) {
      continue;
    }

    const character =
      state[target];

    if (
      !character ||
      typeof character[stat] !==
        "number"
    ) {
      continue;
    }

    /*
     * The model may never set HP directly.
     * HP can only change through a bounded addition.
     */
    if (
      stat === "hp" &&
      operation === "set"
    ) {
      continue;
    }

    /*
     * Ordinary conversation can never damage the player.
     *
     * Player damage is accepted only when the current user message
     * was recognized as an explicit hostile physical action.
     */
    if (
      target === "player" &&
      stat === "hp" &&
      numericValue < 0 &&
      !allowPlayerDamage
    ) {
      continue;
    }

    const [
      minimum,
      maximum
    ] =
      STATE_LIMITS[stat];

    let proposed;

    if (
      operation === "set"
    ) {
      proposed =
        numericValue;
    } else if (
      operation === "add"
    ) {
      /*
       * Prevent one response from causing absurd state changes.
       */
      const maximumDelta =
        stat === "money"
          ? 100000
          : stat === "hp"
            ? 25
            : 30;

      const boundedDelta =
        Math.max(
          -maximumDelta,
          Math.min(
            maximumDelta,
            numericValue
          )
        );

      proposed =
        character[stat] +
        boundedDelta;
    } else {
      continue;
    }

    character[stat] =
      Math.max(
        minimum,
        Math.min(
          maximum,
          proposed
        )
      );

    if (
      character.hp >
      character.maxHp
    ) {
      character.hp =
        character.maxHp;
    }
  }
}

/* =========================================================
   VALIDATED INVENTORY / CONDITION / SPELL ACTIONS
   ========================================================= */

function applyValidatedActions(
  state,
  actions
) {
  if (!Array.isArray(actions)) {
    return;
  }

  for (
    const rawAction of
    actions.slice(0, 12)
  ) {
    const target =
      rawAction?.target;

    const action =
      rawAction?.action;

    if (
      !ALLOWED_TARGETS.has(
        target
      )
    ) {
      continue;
    }

    const character =
      state[target];

    if (!character) {
      continue;
    }

    const id =
      slugifyId(
        rawAction?.id
      );

    if (!id) {
      continue;
    }

    if (
      action === "add_item"
    ) {
      addInventoryItem(
        character,
        {
          id,

          name:
            safeLabel(
              rawAction?.name,
              id
            ),

          quantity:
            clampNumber(
              rawAction?.quantity,
              1,
              999,
              1
            ),

          description:
            limitText(
              String(
                rawAction?.description ||
                ""
              ).trim(),
              300
            ),

          acquiredAt:
            new Date()
              .toISOString(),

          acquiredFrom:
            safeLabel(
              rawAction?.source,
              "unknown"
            )
        }
      );

      continue;
    }

    if (
      action === "remove_item"
    ) {
      removeInventoryItem(
        character,
        id,
        clampNumber(
          rawAction?.quantity,
          1,
          999,
          1
        )
      );

      continue;
    }

    if (
      action === "equip_item"
    ) {
      const slot =
        normalizeEquipmentSlot(
          rawAction?.slot
        );

      if (
        slot &&
        inventoryContains(
          character,
          id
        )
      ) {
        character.equipment[
          slot
        ] = id;
      }

      continue;
    }

    if (
      action === "unequip_slot"
    ) {
      const slot =
        normalizeEquipmentSlot(
          rawAction?.slot
        );

      if (slot) {
        character.equipment[
          slot
        ] = null;
      }

      continue;
    }

    if (
      action === "add_condition"
    ) {
      addCondition(
        character,
        {
          id,

          name:
            safeLabel(
              rawAction?.name,
              id
            ),

          severity:
            clampNumber(
              rawAction?.severity,
              0,
              100,
              20
            ),

          description:
            limitText(
              String(
                rawAction?.description ||
                ""
              ).trim(),
              300
            ),

          acquiredAt:
            new Date()
              .toISOString()
        }
      );

      continue;
    }

    if (
      action ===
      "remove_condition"
    ) {
      character.conditions =
        character.conditions.filter(
          (condition) => {
            return (
              condition.id !== id
            );
          }
        );

      continue;
    }

    if (
      action === "learn_spell"
    ) {
      addSpell(
        character,
        {
          id,

          name:
            safeLabel(
              rawAction?.name,
              id
            ),

          description:
            limitText(
              String(
                rawAction?.description ||
                ""
              ).trim(),
              300
            ),

          learnedAt:
            new Date()
              .toISOString(),

          learnedFrom:
            safeLabel(
              rawAction?.source,
              "unknown"
            )
        }
      );

      continue;
    }

    if (
      action === "forget_spell"
    ) {
      character.spellsKnown =
        character.spellsKnown.filter(
          (spell) => {
            return (
              spell.id !== id
            );
          }
        );
    }
  }
}

  if (

      emotionalWeight:
        0

  ) {

        "health-change",

          : 0.55,

      emotionalWeight:
        after.sally.hp <

          ? -0.6
          : 0.3
    });
  }

  if (
    after.player.hp !==
    before.player.hp
  ) {
    memories.push({
      type:
        "health-change",

      summary:
        `The visitor's health changed from ` +
        `${before.player.hp}/${before.player.maxHp} to ` +
        `${after.player.hp}/${after.player.maxHp}.`,

      importance:
        after.player.hp <
        before.player.hp
          ? 0.75
          : 0.55,

      emotionalWeight:
        after.player.hp <
        before.player.hp
          ? -0.5
          : 0.25
    });
  }

  if (
    after.sally.trust !==
    before.sally.trust
  ) {
    const direction =
      after.sally.trust >
      before.sally.trust
        ? "increased"
        : "decreased";

    memories.push({
      type:
        "relationship-change",

      summary:
        `Sally's current trust toward the visitor ${direction} after their ` +

      importance:

      emotionalWeight:
        direction === "increased"
          ? 0.45
          : -0.55
    });
  }

  if (
    after.sally.fear !==
    before.sally.fear
  ) {
    const direction =
      after.sally.fear >
      before.sally.fear
        ? "increased"
        : "decreased";

    memories.push({
      type:
        "relationship-change",

      summary:
        `Sally's current fear of the visitor ${direction} after their ` +
        `exchange.`,

      importance:
        0.6,

      emotionalWeight:
        direction === "increased"
          ? -0.45
          : 0.25
    });
  }

  if (
    after.sally.resentment !==
    before.sally.resentment
  ) {
    const direction =
      after.sally.resentment >
      before.sally.resentment
        ? "increased"
        : "decreased";

    memories.push({
      type:
        "relationship-change",

      summary:
        `Sally's current resentment toward the visitor ${direction} after ` +
        `their exchange.`,

      importance:
        0.7,

      emotionalWeight:
        direction === "increased"
          ? -0.65
          : 0.35
    });
  }

  if (
    after.sally.allegianceToLoki !==
    before.sally.allegianceToLoki
  ) {
    memories.push({
      type:
        "political-change",

      summary:

      emotionalWeight:

    });
  }

  if (
    Array.isArray(actions) &&
    actions.length > 0
  ) {
    for (
      const action of
      actions.slice(0, 4)
    ) {
      if (
        action?.action ===
          "add_item" &&

          "sally"
      ) {
        memories.push({

            "possession",

          summary:

          emotionalWeight:

  await saveMemories(

  );

/* =========================================================
   INVENTORY HELPERS
   ========================================================= */

function addInventoryItem(
  character,
  item
) {
  const existing =
    character.inventory.find(
      (candidate) => {
        return (
          candidate.id === item.id
        );
      }
    );

  if (existing) {
    existing.quantity =
      clampNumber(
        existing.quantity +
          item.quantity,
        1,
        999,
        existing.quantity
      );

    return;
  }

  character.inventory.push(
    item
  );

  character.inventory =
    character.inventory.slice(
      0,
      200
    );
}

function removeInventoryItem(
  character,
  id,
  quantity
) {
  const item =
    character.inventory.find(
      (candidate) => {
        return (
          candidate.id === id
        );
      }
    );

  if (!item) {
    return;
  }

  item.quantity -=
    quantity;

  if (
    item.quantity > 0
  ) {
    return;
  }

  character.inventory =
    character.inventory.filter(
      (candidate) => {
        return (
          candidate.id !== id
        );
      }
    );

  for (
    const slot of
    Object.keys(
      character.equipment
    )
  ) {
    if (
      character.equipment[
        slot
      ] === id
    ) {
      character.equipment[
        slot
      ] = null;
    }
  }
}

function addCondition(
  character,
  condition
) {
  const existing =
    character.conditions.find(
      (candidate) => {
        return (
          candidate.id ===
          condition.id
        );
      }
    );

  if (existing) {
    existing.severity =
      Math.max(
        Number(
          existing.severity
        ) || 0,
        condition.severity
      );

    if (
      condition.description
    ) {
      existing.description =
        condition.description;
    }

    return;
  }

  character.conditions.push(
    condition
  );

  character.conditions =
    character.conditions.slice(
      0,
      30
    );
}

function addSpell(
  character,
  spell
) {
  if (
    character.spellsKnown.some(
      (candidate) => {
        return (
          candidate.id ===
          spell.id
        );
      }
    )
  ) {
    return;
  }

  character.spellsKnown.push(
    spell
  );

  character.spellsKnown =
    character.spellsKnown.slice(
      0,
      100
    );
}

function hasCondition(
  character,
  id
) {
  return (
    character.conditions.some(
      (condition) => {
        return (
          condition.id === id
        );
      }
    )
  );
}

function inventoryContains(
  character,
  id
) {
  return (
    character.inventory.some(
      (item) => {
        return (
          item.id === id &&
          Number(
            item.quantity
          ) > 0
        );
      }
    )
  );
}

function makePublicState(

) {
  /*
   * Permanent relationship totals remain hidden during

  return {

        state.player.maxHp,

      hunger:

        ),

      inventory:
        state.player.inventory,

    },


    sally: {

        state.sally.hp,

      maxHp:
        state.sally.maxHp,

      hunger:
        state.sally.hunger,

      fullness:
        Math.max(
          0,

        ),

        hungerDescription(
          state.sally.hunger
        ),

      money:
        state.sally.money,

          state.sally.lastMealId,

      },

    },

  };

/* =========================================================
   DISPLAY HELPERS
   ========================================================= */

function hungerDescription(
  value
) {
  if (value < 25) {
    return "Fed";
  }

  if (value < 50) {
    return "Hungry";
  }

  if (value < 75) {
    return "Very Hungry";
  }

  if (value < 90) {
    return "Starving";
  }

  return "Critical";
}

function formatMoney(value) {
  return Number(
    value || 0
  ).toFixed(2);
}

function formatConditions(
  conditions
) {
  if (
    !Array.isArray(
      conditions
    ) ||
    conditions.length === 0
  ) {
    return "None";
  }

  return conditions
    .map((condition) => {
      return (
        condition.name ||
        condition.id
      );
    })
    .join(", ");
}

function formatInventory(
  inventory
) {
  if (
    !Array.isArray(
      inventory
    ) ||
    inventory.length === 0
  ) {
    return "Empty";
  }

  return inventory
    .slice(0, 30)
    .map((item) => {
      const quantity =
        Number(
          item.quantity
        ) || 1;

      return quantity > 1
        ? (
            `${item.name || item.id} ` +
            `x${quantity}`
          )
        : (
            item.name ||
            item.id
          );
    })
    .join(", ");
}

function formatEquipment(
  equipment,
  inventory
) {
  const entries = [];

  for (
    const [
      slot,
      itemId
    ] of
    Object.entries(
      equipment || {}
    )
  ) {
    if (!itemId) {
      continue;
    }

    const item =
      Array.isArray(
        inventory
      )
        ? inventory.find(
            (candidate) => {
              return (
                candidate.id ===
                itemId
              );
            }
          )
        : null;

    entries.push(
      `${slot}: ${
        item?.name ||
        itemId
      }`
    );
  }

  return (
    entries.length > 0
      ? entries.join(", ")
      : "None"
  );
}

function formatSpells(
  spells
) {
  if (
    !Array.isArray(
      spells
    ) ||
    spells.length === 0
  ) {
    return "None";
  }

  return spells
    .map((spell) => {
      return (
        spell.name ||
        spell.id
      );
    })
    .join(", ");
}

/* =========================================================
   NORMALIZATION HELPERS
   ========================================================= */

function normalizeInventory(
  value
) {
  if (
    !Array.isArray(value)
  ) {
    return [];
  }

  return value
    .slice(0, 200)
    .map((item) => {
      const id =
        slugifyId(
          item?.id ||
          item?.name
        );

      if (!id) {
        return null;
      }

      return {
        id,

        name:
          safeLabel(
            item?.name,
            id
          ),

        quantity:
          clampNumber(
            item?.quantity,
            1,
            999,
            1
          ),

        description:
          limitText(
            String(
              item?.description ||
              ""
            ),
            300
          ),

        acquiredAt:
          validIsoDate(
            item?.acquiredAt
          ) ||
          new Date()
            .toISOString(),

        acquiredFrom:
          safeLabel(
            item?.acquiredFrom,
            "unknown"
          )
      };
    })
    .filter(Boolean);
}

function normalizeStringObjectArray(
  value,
  maximum
) {
  if (
    !Array.isArray(value)
  ) {
    return [];
  }

  return value
    .slice(
      0,
      maximum
    )
    .map((item) => {
      if (
        typeof item ===
        "string"
      ) {
        const id =
          slugifyId(item);

        return id
          ? {
              id,

              name:
                safeLabel(
                  item,
                  id
                )
            }
          : null;
      }

      if (
        !item ||
        typeof item !==
          "object"
      ) {
        return null;
      }

      const id =
        slugifyId(
          item.id ||
          item.name
        );

      if (!id) {
        return null;
      }

      const normalized = {
        ...item,

        id,

        name:
          safeLabel(
            item.name,
            id
          )
      };

      if (
        "severity" in
        normalized
      ) {
        normalized.severity =
          clampNumber(
            normalized.severity,
            0,
            100,
            0
          );
      }

      return normalized;
    })
    .filter(Boolean);
}

function normalizeEquipmentSlot(
  value
) {
  const slot =
    typeof value === "string"
      ? value
          .trim()
          .toLowerCase()
      : "";

  return [
    "head",
    "body",
    "hands",
    "legs",
    "accessory",
    "weapon"
  ].includes(slot)
    ? slot
    : null;
}

function normalizeNullableString(
  value
) {
  return (
    typeof value === "string" &&
    value.trim()
  )
    ? slugifyId(value)
    : null;
}

function normalizeVisitorId(
  value
) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  const cleaned =
    value
      .trim()
      .replace(
        /[^a-zA-Z0-9_-]/g,
        ""
      )
      .slice(
        0,
        100
      );

  return (
    cleaned.length >= 12
      ? cleaned
      : ""
  );
}

function slugifyId(value) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  return value
    .trim()
    .toLowerCase()
    .replace(
      /['’]/g,
      ""
    )
    .replace(
      /[^a-z0-9]+/g,
      "-"
    )
    .replace(
      /^-+|-+$/g,
      ""
    )
    .slice(
      0,
      80
    );
}

function safeLabel(
  value,
  fallback
) {
  const cleaned =
    typeof value ===
      "string"
      ? value.trim()
      : "";

  return limitText(
    cleaned ||
      fallback ||
      "Unknown",
    100
  );
}

function clampNumber(
  value,
  minimum,
  maximum,
  fallback
) {
  const numeric =
    Number(value);

  if (
    !Number.isFinite(
      numeric
    )
  ) {
    return fallback;
  }

  return Math.max(
    minimum,
    Math.min(
      maximum,
      numeric
    )
  );
}

function validIsoDate(
  value
) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  const time =
    Date.parse(value);

  return (
    Number.isFinite(time)
      ? new Date(time)
          .toISOString()
      : ""
  );
}

function deepClone(value) {
  return JSON.parse(
    JSON.stringify(
      value
    )
  );
}

function containsForbiddenName(
  message
) {
  return (
    FORBIDDEN_PATTERNS.some(
      (pattern) => {
        return pattern.test(
          message
        );
      }
    )
  );
}

function limitText(
  value,
  maximumLength
) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  return value.slice(
    0,
    maximumLength
  );
}

function jsonResponse(

) {
  return new Response(
    JSON.stringify(data),

      status,

      headers: {
        "content-type":
          "application/json; charset=UTF-8",

        "cache-control":
          "no-store"
      }
    }
  );
}

    function formatGold(value) {
      const numeric =
        Number(value);

      if (
        !Number.isFinite(
          numeric
        )
      ) {
        return "0";
      }

      return new Intl.NumberFormat(
        "en-CA",
        {
          maximumFractionDigits:
            0
        }
      ).format(
        Math.floor(numeric)
      );
    }


    function updatePublicState(
      publicState
    ) {
      if (!publicState) {
        return;
      }

      currentPublicState =
        publicState;

      const player =
        publicState.player;

      const sally =
        publicState.sally;


      if (player) {
        playerStatsElement.textContent =
          "YOU — HP " +
          player.hp +
          "/" +
          player.maxHp +
          " | HUNGER " +
          player.hunger +
          " (" +
          player.hungerText +
          ") | GOLD " +
          formatGold(
            player.money
          );

        goldSummary.textContent =
          "GOLD: " +
          formatGold(
            player.money
          );
      }


      if (sally) {
        const fullness =
          Number.isFinite(
            Number(
              sally.fullness
            )
          )
            ? Number(
                sally.fullness
              )
            : Math.max(
                0,
                100 -
                Number(
                  sally.hunger || 0
                )
              );

        sallyStatsElement.textContent =
          "SALLY — HP " +
          sally.hp +
          "/" +
          sally.maxHp +
          " | HUNGER " +
          sally.hunger +
          " (" +
          sally.hungerText +
          ")";

        hungerSummary.textContent =
          "HUNGER: " +
          sally.hunger +
          "/100 — " +
          sally.hungerText;

        fullnessSummary.textContent =
          "FULLNESS: " +
          fullness +
          "/100";
      }

      updateFoodControls();
    }


    async function loadPublicState() {
      try {
        const response =
          await fetch(
            "/state?visitorId=" +
            encodeURIComponent(
              visitorId
            ),
            {
              cache:
                "no-store"
            }
          );

        const result =
          await response.json();

        if (!response.ok) {
          throw new Error(
            result.error ||
            "The persistent state could not be loaded."
          );
        }

        updatePublicState(
          result.publicState
        );

      } catch (error) {
        console.error(
          "Could not load persistent state:",
          error
        );

        setFeedingStatus(
          "The shared wallet and Sally's condition could not be loaded.",
          "error"
        );
      }
    }


    /* =====================================================
       FOOD CATALOGUE
       ===================================================== */

    async function loadFoodCatalog() {
      try {
        const response =
          await fetch(
            "/food-catalog",
            {
              cache:
                "no-store"
            }
          );

        const result =
          await response.json();

        if (!response.ok) {
          throw new Error(
            result.error ||
            "The provisions could not be loaded."
          );
        }

        foodCatalog =
          Array.isArray(
            result.items
          )
            ? result.items
            : [];

        renderFoodCatalog();

      } catch (error) {
        console.error(
          "Could not load food catalogue:",
          error
        );

        foodSelect.replaceChildren();

        const option =
          document.createElement(
            "option"
          );

        option.value =
          "";

        option.textContent =
          "Provisions unavailable";

        foodSelect.appendChild(
          option
        );

        setFeedingStatus(
          "The provisions could not be brought to the cell.",
          "error"
        );

        updateFoodControls();
      }
    }


    function renderFoodCatalog() {
      foodSelect.replaceChildren();

      for (
        const item of
        foodCatalog
      ) {
        const option =
          document.createElement(
            "option"
          );

        option.value =
          item.id;

        option.textContent =
          item.name +
          " — " +
          formatGold(
            item.price
          ) +
          " gold";

        foodSelect.appendChild(
          option
        );
      }

      updateSelectedFoodDisplay();
      updateFoodControls();
    }


    function getSelectedFoodItem() {
      const selectedId =
        foodSelect.value;

      return (
        foodCatalog.find(
          function (item) {
            return (
              item.id ===
              selectedId
            );
          }
        ) ||
        null
      );
    }


    function updateSelectedFoodDisplay() {
      const item =
        getSelectedFoodItem();

      if (!item) {
        foodDescription.textContent =
          "No provision has been selected.";

        feedButton.textContent =
          "BUY AND GIVE";

        return;
      }

      const categoryText =
        item.category ===
          "alcohol"
          ? "ALCOHOL"
          : "FOOD";

      foodDescription.textContent =
        categoryText +
        " — " +
        item.description;

      feedButton.textContent =
        "GIVE — " +
        formatGold(
          item.price
        ) +
        " GOLD";
    }


    function updateFoodControls() {
      const item =
        getSelectedFoodItem();

      const playerMoney =
        Number(
          currentPublicState
            ?.player
            ?.money
        );

      const affordable =
        item &&
        Number.isFinite(
          playerMoney
        ) &&
        playerMoney >=
          Number(item.price);

      foodSelect.disabled =
        requestInProgress ||
        conversationEnded ||
        foodCatalog.length === 0;

      feedButton.disabled =
        requestInProgress ||
        conversationEnded ||
        !item ||
        !affordable;

      if (
        item &&
        Number.isFinite(
          playerMoney
        ) &&
        !affordable &&
        !requestInProgress
      ) {
        setFeedingStatus(
          "You cannot afford this. It costs " +
          formatGold(
            item.price
          ) +
          " gold.",
          "error"
        );
      } else if (
        feedingStatus.textContent
          .startsWith(
            "You cannot afford"
          )
      ) {
        setFeedingStatus(
          "",
          null
        );
      }
    }


    /* =====================================================
       PURCHASE AND FEED SALLY
       ===================================================== */

    async function feedSally() {
      if (
        requestInProgress ||
        conversationEnded
      ) {
        return;
      }

      const item =
        getSelectedFoodItem();

      if (!item) {
        return;
      }

      setRequestInProgress(
        true
      );

      setFeedingStatus(
        "The provision is being brought to the cell...",
        null
      );

      statusElement.textContent =
        "SOMETHING PASSES THROUGH THE WALL";

      try {
        const response =
          await fetch(
            "/feed",
            {
              method:
                "POST",

              headers: {
                "content-type":
                  "application/json"
              },

              body:
                JSON.stringify({
                  visitorId:
                    visitorId,

                  itemId:
                    item.id
                })
            }
          );

        let result;

        try {
          result =
            await response.json();
        } catch {
          throw new Error(
            "The feeding request returned an unreadable answer."
          );
        }

        if (!response.ok) {
          throw new Error(
            result.error ||
            "The purchase could not be completed."
          );
        }

        updatePublicState(
          result.publicState
        );

        const reaction =
          typeof result.reaction ===
            "string" &&
          result.reaction.trim()
            ? result.reaction.trim()
            : (
                "Sally takes the provision without speaking."
              );

        addMessage(
          "SALLY",
          reaction,
          "sally"
        );

        speakText(
          reaction
        );

        const outcome =
          result.outcome;

        if (
          outcome ===
            "beneficial"
        ) {
          if (
            result.effects?.rescue
          ) {
            setFeedingStatus(
              item.name +
              " was delivered. Sally was in serious need of it.",
              "success"
            );
          } else {
            setFeedingStatus(
              item.name +
              " was delivered to Sally.",
              "success"
            );
          }
        } else if (
          outcome ===
            "already-full"
        ) {
          setFeedingStatus(
            item.name +
            " was purchased, but Sally was already full enough to receive no emotional benefit.",
            null
          );
        } else if (
          outcome ===
            "forced"
        ) {
          const damage =
            Number(
              result.effects
                ?.hpDamage || 0
            );

          setFeedingStatus(
            damage > 0
              ? (
                  "Sally was already completely full. The continued feeding frightened her and caused " +
                  damage +
                  " HP damage."
                )
              : (
                  "Sally was already completely full. She experienced the insistence as frightening and coercive."
                ),
            "error"
          );
        }

        statusElement.textContent =
          "SOMETHING LISTENS BEHIND THE WALL";

      } catch (error) {
        console.error(
          "Could not feed Sally:",
          error
        );

        setFeedingStatus(
          error.message ||
          "The purchase could not be completed.",
          "error"
        );

        statusElement.textContent =
          "THE PASSAGE IS UNSTABLE";

      } finally {
        setRequestInProgress(
          false
        );

        inputElement.focus();
      }
    }


    /* =====================================================
       PLAYER DEATH
       ===================================================== */

    function killPlayer(
      deathDescription
    ) {
      conversationEnded =
        true;

      stopVoice();

      setRequestInProgress(
        false
      );

      statusElement.textContent =
        "THE PASSAGE HAS GONE SILENT";

      window.setTimeout(
        function () {
          messagesElement
            .replaceChildren();

          chatArea.style.display =
            "none";

          deathText.textContent =
            deathDescription;

          deathScreen.style.display =
            "flex";

          speakText(
            deathDescription
          );

          inputElement.value =
            "";
        },
        500
      );
    }


    /* =====================================================
       CHAT
       ===================================================== */

    async function sendMessage() {
      if (
        conversationEnded ||
        requestInProgress
      ) {
        return;
      }

      const message =
        inputElement.value
          .trim();

      if (!message) {
        return;
      }

      turnNumber +=
        1;

      addMessage(
        "YOU",
        message,
        "visitor"
      );

      inputElement.value =
        "";

      setRequestInProgress(
        true
      );

      statusElement.textContent =
        "SALLY IS LISTENING";

      try {
        const response =
          await fetch(
            "/chat",
            {
              method:
                "POST",

              headers: {
                "content-type":
                  "application/json"
              },

              body:
                JSON.stringify({
                  visitorId:
                    visitorId,

                  message:
                    message,

                  turnNumber:
                    turnNumber
                })
            }
          );

        let result;

        try {
          result =
            await response.json();
        } catch {
          throw new Error(
            "The passage returned an unreadable answer."
          );
        }

        if (!response.ok) {
          throw new Error(
            result.error ||
            "The passage failed."
          );
        }

        updatePublicState(
          result.publicState
        );

        if (
          result.type ===
            "death"
        ) {
          addSystemMessage(
            "Something has gone wrong in the passage."
          );

          killPlayer(
            result.text
          );

          return;
        }

        const reply =
          typeof result.text ===
            "string" &&
          result.text.trim()
            ? result.text.trim()
            : "Sally does not answer.";

        addMessage(
          "SALLY",
          reply,
          "sally"
        );

        speakText(
          reply
        );

        statusElement.textContent =
          "SOMETHING LISTENS BEHIND THE WALL";

      } catch (error) {
        console.error(
          error
        );

        const failureLine =
          "The wall has gone cold. I cannot hear you.";

        addMessage(
          "SALLY",
          failureLine,
          "sally"
        );

        speakText(
          failureLine
        );

        statusElement.textContent =
          "THE PASSAGE IS UNSTABLE";

      } finally {
        if (!conversationEnded) {
          setRequestInProgress(
            false
          );

          inputElement.focus();
        }
      }
    }


    /* =====================================================
       RESTART
       ===================================================== */

    async function restartConversation() {
      conversationEnded =
        false;

      requestInProgress =
        false;

      turnNumber =
        0;

      stopVoice();

      messagesElement
        .replaceChildren();

      deathScreen.style.display =
        "none";

      chatArea.style.display =
        "flex";

      inputElement.value =
        "";

      statusElement.textContent =
        "SOMETHING LISTENS BEHIND THE WALL";

      setFeedingStatus(
        "",
        null
      );

      const openingLine =
        "...Who speaks through the wall?";

      addMessage(
        "SALLY",
        openingLine,
        "sally"
      );

      window.setTimeout(
        function () {
          if (
            browserSpeechSupported
          ) {
            chooseVoice();
          }

          speakText(
            openingLine
          );
        },
        400
      );

      setRequestInProgress(
        true
      );

      await Promise.all([
        loadFoodCatalog(),
        loadPublicState()
      ]);

      setRequestInProgress(
        false
      );

      inputElement.focus();
    }


    /* =====================================================
       EVENTS
       ===================================================== */

    voiceToggle.addEventListener(
      "click",
      toggleVoice
    );

    if (browserSpeechSupported) {
      chooseVoice();

      window.speechSynthesis
        .addEventListener(
          "voiceschanged",
          chooseVoice
        );
    }

    updateVoiceToggle();


    foodSelect.addEventListener(
      "change",
      function () {
        updateSelectedFoodDisplay();
        updateFoodControls();
      }
    );


    feedButton.addEventListener(
      "click",
      feedSally
    );


    speakButton.addEventListener(
      "click",
      sendMessage
    );


    inputElement.addEventListener(
      "keydown",
      function (event) {
        if (
          event.key === "Enter"
        ) {
          event.preventDefault();

        }

    );


    restartButton.addEventListener(
      "click",
      restartConversation
    );


    restartConversation();
