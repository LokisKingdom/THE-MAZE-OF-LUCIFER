export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    if (
      request.method === "POST" &&
      url.pathname === "/chat"
    ) {
      return handleChat(request, env);
    }

    if (
      request.method === "GET" &&
      url.pathname === "/state"
    ) {
      return handleState(request, env, url);
    }

    return new Response(HTML, {
      headers: {
        "content-type": "text/html; charset=UTF-8",
        "cache-control": "no-store"
      }
    });
  }
};

async function ensureDatabase(env) {
  if (!env.SALLY_DB) {
    throw new Error(
      'Missing D1 binding. Add a D1 binding named "SALLY_DB".'
    );
  }

  await env.SALLY_DB.batch([
    env.SALLY_DB.prepare(`
      CREATE TABLE IF NOT EXISTS world_state (
        visitor_id TEXT PRIMARY KEY,
        state_json TEXT NOT NULL,
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL
      )
    `),

    env.SALLY_DB.prepare(`
      CREATE TABLE IF NOT EXISTS conversation_log (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        visitor_id TEXT NOT NULL,
        role TEXT NOT NULL,
        content TEXT NOT NULL,
        created_at TEXT NOT NULL
      )
    `),

    env.SALLY_DB.prepare(`
      CREATE INDEX IF NOT EXISTS
      idx_conversation_visitor_created
      ON conversation_log(visitor_id, created_at)
    `),

    env.SALLY_DB.prepare(`
      CREATE TABLE IF NOT EXISTS memories (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        visitor_id TEXT NOT NULL,
        type TEXT NOT NULL,
        summary TEXT NOT NULL,
        importance REAL NOT NULL DEFAULT 0.5,
        emotional_weight REAL NOT NULL DEFAULT 0,
        created_at TEXT NOT NULL,
        last_recalled_at TEXT,
        recall_count INTEGER NOT NULL DEFAULT 0
      )
    `),

    env.SALLY_DB.prepare(`
      CREATE INDEX IF NOT EXISTS
      idx_memories_visitor_created
      ON memories(visitor_id, created_at)
    `)
  ]);
}

async function loadWorldState(env, visitorId) {
  const row = await env.SALLY_DB
    .prepare(`
      SELECT state_json
      FROM world_state
      WHERE visitor_id = ?
      LIMIT 1
    `)
    .bind(visitorId)
    .first();

  if (!row?.state_json) {
    const state =
      createDefaultWorldState();

    await saveWorldState(
      env,
      visitorId,
      state,
      true
    );

    return state;
  }

  try {
    const parsed =
      JSON.parse(
        row.state_json
      );

    return normalizeWorldState(
      parsed
    );
  } catch (error) {
    console.error(
      "Stored world state was unreadable:",
      error
    );

    const state =
      createDefaultWorldState();

    await saveWorldState(
      env,
      visitorId,
      state,
      false
    );

    return state;
  }
}

async function saveWorldState(
  env,
  visitorId,
  state,
  isNew = false
) {
  const now =
    new Date().toISOString();

  state.lastUpdatedAt =
    now;

  const serialized =
    JSON.stringify(
      normalizeWorldState(
        state
      )
    );

  if (isNew) {
    await env.SALLY_DB
      .prepare(`
        INSERT OR REPLACE INTO world_state (
          visitor_id,
          state_json,
          created_at,
          updated_at
        )
        VALUES (?, ?, ?, ?)
      `)
      .bind(
        visitorId,
        serialized,
        now,
        now
      )
      .run();

    return;
  }

  await env.SALLY_DB
    .prepare(`
      INSERT INTO world_state (
        visitor_id,
        state_json,
        created_at,
        updated_at
      )
      VALUES (?, ?, ?, ?)

      ON CONFLICT(visitor_id)
      DO UPDATE SET
        state_json = excluded.state_json,
        updated_at = excluded.updated_at
    `)
    .bind(
      visitorId,
      serialized,
      now,
      now
    )
    .run();
}

function normalizeWorldState(candidate) {
  const fallback =
    createDefaultWorldState();

  const player =
    normalizeCharacterState(
      candidate?.player,
      fallback.player,
      false
    );

  const sally =
    normalizeCharacterState(
      candidate?.sally,
      fallback.sally,
      true
    );

  return {
    player,
    sally,

    lastUpdatedAt:
      validIsoDate(
        candidate?.lastUpdatedAt
      ) ||
      fallback.lastUpdatedAt,

    lastHungerUpdateAt:
      validIsoDate(
        candidate?.lastHungerUpdateAt
      ) ||
      fallback.lastHungerUpdateAt
  };
}

      );
	  normalized.believesVisitorIsLoki =
	  candidate?.believesVisitorIsLoki === true;

	  normalized.lokiBeliefPrimed =
	  candidate?.lokiBeliefPrimed === true;
    /*
     * Recent events are intentionally capped.
     *
     * Lifetime totals retain the full historical record.
     * This rolling list exists only to help Sally interpret current patterns,
     * reversals, escalation, contradiction, and apparent behavioural change.
     */
    normalized.recentRelationshipEvents =
      normalizeRecentRelationshipEvents(
        candidate?.recentRelationshipEvents
      );
  }

  return normalized;
}

/* =========================================================
   RELATIONSHIP-STATE NORMALIZATION
   ========================================================= */

function normalizeRelationshipTotal(
  value,
  fallback = 0
) {
  const normalizedFallback =
    Number.isFinite(
      Number(fallback)
    )
      ? Math.max(
          0,
          Math.floor(
            Number(fallback)
          )
        )
      : 0;

  const numeric =
    Number(value);

  if (
    !Number.isFinite(
      numeric
    )
  ) {
    return normalizedFallback;
  }

  return Math.max(
    0,
    Math.min(
      MAX_RELATIONSHIP_SCORE,
      Math.floor(numeric)
    )
  );
}

function normalizeRecentRelationshipEvents(
  value
) {
  if (
    !Array.isArray(value)
  ) {
    return [];
  }

  return value
    .slice(
      -MAX_RECENT_RELATIONSHIP_EVENTS
    )
    .map(
      normalizeRecentRelationshipEvent
    )
    .filter(Boolean);
}

function normalizeRecentRelationshipEvent(
  event
) {
  if (
    !event ||
    typeof event !== "object"
  ) {
    return null;
  }

  const kindness =
    normalizeRelationshipEventScore(
      event.kindness
    );

  const cruelty =
    normalizeRelationshipEventScore(
      event.cruelty
    );

  const amusement =
    normalizeRelationshipEventScore(
      event.amusement
    );

  const pattern =
    normalizeRelationshipPattern(
      event.pattern
    );

  const summary =
    limitText(
      String(
        event.summary || ""
      ).trim(),
      240
    );

  const createdAt =
    validIsoDate(
      event.createdAt
    ) ||
    validIsoDate(
      event.created_at
    ) ||
    new Date()
      .toISOString();

  /*
   * Empty neutral events are not useful enough to retain.
   *
   * A genuinely neutral interaction can still be classified during the
   * current turn, but it does not need to occupy space in the rolling history.
   */
  if (
    kindness === 0 &&
    cruelty === 0 &&
    amusement === 0 &&
    pattern === "neutral" &&
    !summary
  ) {
    return null;
  }

  return {
    kindness,
    cruelty,
    amusement,
    pattern,
    summary,
    createdAt
  };
}

function normalizeRelationshipEventScore(
  value
) {
  const numeric =
    Number(value);

  if (
    !Number.isFinite(
      numeric
    )
  ) {
    return 0;
  }

  return Math.max(
    0,
    Math.min(
      5,
      Math.round(numeric)
    )
  );
}

function normalizeRelationshipPattern(
  value
) {
  const pattern =
    typeof value === "string"
      ? value
          .trim()
          .toLowerCase()
      : "";

  const allowedPatterns =
    new Set([
      "neutral",
      "kind",
      "cruel",
      "amusing",
      "kind-and-amusing",
      "cruel-and-amusing",
      "contradictory",
      "abuse-followed-by-affection",
      "affection-followed-by-abuse",
      "threatening",
      "protective",
      "apologetic",
      "manipulative",
      "mocking",
      "unknown"
    ]);

  return allowedPatterns.has(
    pattern
  )
    ? pattern
    : "unknown";
}

function appendRecentRelationshipEvent(
  sally,
  event
) {
  if (
    !sally ||
    typeof sally !== "object"
  ) {
    return;
  }

  const normalizedEvent =
    normalizeRecentRelationshipEvent(
      event
    );

  if (!normalizedEvent) {
    return;
  }

  const existing =
    Array.isArray(
      sally.recentRelationshipEvents
    )
      ? sally.recentRelationshipEvents
      : [];

  sally.recentRelationshipEvents =
    [
      ...existing,
      normalizedEvent
    ].slice(
      -MAX_RECENT_RELATIONSHIP_EVENTS
    );
}

function summarizeRecentRelationshipHistory(
  events
) {
  const normalizedEvents =
    normalizeRecentRelationshipEvents(
      events
    );

  if (
    normalizedEvents.length === 0
  ) {
    return {
      count: 0,
      kindness: 0,
      cruelty: 0,
      amusement: 0,
      contradictoryEvents: 0,
      cruelEvents: 0,
      kindEvents: 0,
      withdrawalRelevantEvents: 0
    };
  }

  let kindness = 0;
  let cruelty = 0;
  let amusement = 0;
  let contradictoryEvents = 0;
  let cruelEvents = 0;
  let kindEvents = 0;
  let withdrawalRelevantEvents = 0;

  for (
    const event of
    normalizedEvents
  ) {
    kindness +=
      event.kindness;

    cruelty +=
      event.cruelty;

    amusement +=
      event.amusement;

    if (
      event.kindness > 0
    ) {
      kindEvents += 1;
    }

    if (
      event.cruelty > 0
    ) {
      cruelEvents += 1;
    }

    if (
      event.kindness > 0 &&
      event.cruelty > 0
    ) {
      contradictoryEvents += 1;
    }

    if (
      [
        "contradictory",
        "abuse-followed-by-affection",
        "affection-followed-by-abuse",
        "manipulative",
        "mocking"
      ].includes(
        event.pattern
      )
    ) {
      contradictoryEvents += 1;
    }

    if (
      event.cruelty >= 2 ||
      [
        "cruel",
        "cruel-and-amusing",
        "threatening",
        "manipulative",
        "mocking",
        "abuse-followed-by-affection",
        "affection-followed-by-abuse"
      ].includes(
        event.pattern
      )
    ) {
      withdrawalRelevantEvents += 1;
    }
  }

  return {
    count:
      normalizedEvents.length,

    kindness,
    cruelty,
    amusement,
    contradictoryEvents,
    cruelEvents,
    kindEvents,
    withdrawalRelevantEvents
  };
}

/* =========================================================
   HUNGER AND PASSAGE OF TIME
   ========================================================= */

function applyElapsedTime(state) {
  const now = Date.now();

  const previous =
    Date.parse(
      state.lastHungerUpdateAt
    );

  if (!Number.isFinite(previous)) {
    state.lastHungerUpdateAt =
      new Date(now).toISOString();

    return state;
  }

  const elapsedHours =
    Math.max(
      0,
      (now - previous) / 3_600_000
    );

  /*
   * Hunger rises by one point every six hours.
   *
   * Offline growth is capped so a long absence does not
   * automatically destroy the entire game state.
   *
   * Passage of time does not alter trust, kindness, cruelty,
   * amusement, fear, resentment, or any other relationship value.
   * The visitor is not punished merely for being absent.
   */
  const hungerIncrease =
    Math.min(
      40,
      Math.floor(
        elapsedHours / 6
      )
    );

  if (hungerIncrease <= 0) {
    return state;
  }

  state.player.hunger =
    clampNumber(
      state.player.hunger +
        hungerIncrease,
      0,
      100,
      state.player.hunger
    );

  state.sally.hunger =
    clampNumber(
      state.sally.hunger +
        hungerIncrease,
      0,
      100,
      state.sally.hunger
    );

  applyStarvationDamage(
    state.player
  );

  applyStarvationDamage(
    state.sally
  );

  state.lastHungerUpdateAt =
    new Date(now).toISOString();

  return state;
}

function applyStarvationDamage(
  character
) {
  if (character.hunger < 75) {
    return;
  }

  let damage = 0;

  if (character.hunger >= 95) {
    damage = 8;
  } else if (
    character.hunger >= 90
  ) {
    damage = 5;
  } else {
    damage = 2;
  }

  character.hp =
    Math.max(
      0,
      character.hp - damage
    );

  if (
    character.hunger >= 90 &&
    !hasCondition(
      character,
      "starvation"
    )
  ) {
    character.conditions.push({
      id: "starvation",
      name: "Starvation",
      severity:
        character.hunger,
      description:
        "Prolonged hunger is weakening the body.",
      acquiredAt:
        new Date().toISOString()
    });
  }
}

async function loadRecentConversation(

        SELECT
          role,
          content
        FROM conversation_log
        WHERE visitor_id = ?
        ORDER BY id DESC
        LIMIT ?
      `)
      .bind(

        ) &&
        typeof row.content === "string"
      );
    })
    .map((row) => ({
      role:
        row.role,

      content:
        limitText(
          row.content,
          MAX_MESSAGE_LENGTH
        )
    }));
}

/* =========================================================
   MEMORY STORAGE
   ========================================================= */

async function saveMemories(
  env,
  visitorId,
  memories
) {
  if (!Array.isArray(memories)) {
    return;
  }

  const statements = [];

  for (
    const memory of
    memories.slice(0, 6)
  ) {
    const summary =
      typeof memory?.summary ===
        "string"
        ? limitText(
            memory.summary.trim(),
            MAX_STORED_MEMORY_LENGTH
          )
        : "";

    if (!summary) {
      continue;
    }

    const type =
      typeof memory?.type ===
        "string"
        ? (
            limitText(
              memory.type.trim(),
              50
            ) || "event"
          )
        : "event";

    const importance =
      clampNumber(
        memory?.importance,
        0,
        1,
        0.5
      );

    const emotionalWeight =
      clampNumber(
        memory?.emotionalWeight,
        -1,
        1,
        0
      );

    statements.push(
      env.SALLY_DB
        .prepare(`
          INSERT INTO memories (
            visitor_id,
            type,
            summary,
            importance,
            emotional_weight,
            created_at
          )
          VALUES (?, ?, ?, ?, ?, ?)
        `)
        .bind(
          visitorId,
          type,
          summary,
          importance,
          emotionalWeight,
          new Date().toISOString()
        )
    );
  }

  if (statements.length > 0) {
    await env.SALLY_DB.batch(
      statements
    );
  }
}

/* =========================================================
   MEMORY RETRIEVAL
   ========================================================= */

async function loadRelevantMemories(
  env,
  visitorId,
  message
) {
  const result =
    await env.SALLY_DB
      .prepare(`
        SELECT
          id,
          type,
          summary,
          importance,
          emotional_weight,
          created_at,
          recall_count
        FROM memories
        WHERE visitor_id = ?
        ORDER BY
          importance DESC,
          id DESC
        LIMIT 250
      `)
      .bind(visitorId)
      .all();

  const rows =
    Array.isArray(
      result?.results
    )
      ? result.results
      : [];

  if (rows.length === 0) {
    return [];
  }

  const messageWords =
    meaningfulWords(message);

  const scored =
    rows.map((row) => {
      const memoryWords =
        meaningfulWords(
          row.summary
        );

      let overlap = 0;

      for (
        const word of
        messageWords
      ) {
        if (
          memoryWords.has(word)
        ) {
          overlap += 1;
        }
      }

      const importance =
        Number(
          row.importance
        ) || 0;

      const emotional =
        Math.abs(
          Number(
            row.emotional_weight
          ) || 0
        );

      const recallBonus =
        Math.min(
          0.15,
          (
            Number(
              row.recall_count
            ) || 0
          ) * 0.005
        );

      const score =
        overlap * 2 +
        importance * 3 +
        emotional +
        recallBonus;

      return {
        ...row,
        score
      };
    });

  scored.sort((a, b) => {
    if (b.score !== a.score) {
      return b.score - a.score;
    }

    return (
      Number(b.id) -
      Number(a.id)
    );
  });

  const selected =
    scored.slice(
      0,
      MAX_CONTEXT_MEMORIES
    );

  if (selected.length > 0) {
    const now =
      new Date().toISOString();

    const updates =
      selected.map(
        (memory) => {
          return env.SALLY_DB
            .prepare(`
              UPDATE memories
              SET
                last_recalled_at = ?,
                recall_count =
                  recall_count + 1
              WHERE id = ?
            `)
            .bind(
              now,
              memory.id
            );
        }
      );

    await env.SALLY_DB.batch(
      updates
    );
  }

  return selected;
}

function meaningfulWords(value) {
  const stopWords =
    new Set([
      "the",
      "and",
      "that",
      "this",
      "with",
      "from",
      "have",
      "your",
      "you",
      "are",
      "was",
      "were",
      "for",
      "but",
      "not",
      "what",
      "when",
      "where",
      "who",
      "why",
      "how",
      "into",
      "about",
      "they",
      "them",
      "their",
      "there",
      "then",
      "than",
      "will",
      "would",
      "could",
      "should"
    ]);

  return new Set(
    String(value || "")
      .toLowerCase()
      .replace(
        /[^\p{L}\p{N}'-]+/gu,
        " "
      )
      .split(/\s+/)
      .filter((word) => {
        return (
          word.length >= 3 &&
          !stopWords.has(word)
        );
      })
  );
}

function describeRelationshipHistory(
  sally,
  recentSummary = null
) {
  const kindness =
    normalizeRelationshipTotal(
      sally?.kindnessReceived,
      0
    );

  const cruelty =
    normalizeRelationshipTotal(
      sally?.crueltyReceived,
      0
    );

  const amusement =
    normalizeRelationshipTotal(
      sally?.amusementReceived,
      0
    );

  const recent =
    recentSummary ||
    summarizeRecentRelationshipHistory(
      sally?.recentRelationshipEvents
    );

  const total =
    kindness + cruelty;

  const crueltyShare =
    total > 0
      ? cruelty / total
      : 0;

  const kindnessShare =
    total > 0
      ? kindness / total
      : 0;

  const lines = [];

  if (
    kindness === 0 &&
    cruelty === 0 &&
    amusement === 0
  ) {
    lines.push(
      "Sally has not yet accumulated enough history to form a strong model " +
      "of the visitor."
    );
  } else if (
    cruelty >= 100 &&
    kindness >= 100
  ) {
    lines.push(
      "The visitor has demonstrated substantial kindness and substantial " +
      "cruelty. This is not neutral conduct. Sally has learned that the " +
      "visitor can be warm, helpful, entertaining, and dangerous."
    );

    lines.push(
      "Pleasant behaviour may therefore feel unstable, mocking, calculated, " +
      "or ominous rather than reassuring."
    );
  } else if (
    cruelty >= 40 &&
    kindness >= 20
  ) {
    lines.push(
      "The visitor has a meaningful mixed history. Sally cannot safely assume " +
      "that kindness predicts safety because cruelty is already part of the " +
      "established pattern."
    );
  } else if (
    cruelty >= 40 &&
    crueltyShare >= 0.75
  ) {
    lines.push(
      "Cruelty strongly dominates the visitor's history. Sally has learned " +
      "to regard the visitor as dangerous, degrading, threatening, or unsafe."
    );
  } else if (
    kindness >= 40 &&
    kindnessShare >= 0.85 &&
    cruelty <= Math.max(
      5,
      kindness * 0.1
    )
  ) {
    lines.push(
      "Kindness strongly dominates the visitor's history. Sally has generally " +
      "experienced the visitor as safe, considerate, protective, or humane."
    );

    if (cruelty > 0) {
      lines.push(
        "The existing cruel events still remain part of the history and may " +
        "be remembered as disturbing exceptions."
      );
    }
  } else if (
    cruelty > kindness &&
    cruelty >= 10
  ) {
    lines.push(
      "Cruelty currently outweighs kindness in the historical record. Sally " +
      "has reason to remain wary even when the visitor sounds pleasant."
    );
  } else if (
    kindness > cruelty &&
    kindness >= 10
  ) {
    lines.push(
      "Kindness currently outweighs cruelty in the historical record, but " +
      "any cruel acts remain real and should not be erased."
    );
  } else {
    lines.push(
      "The relationship history is still developing and does not yet support " +
      "a simple or confident interpretation."
    );
  }

  if (
    recent.contradictoryEvents >= 3
  ) {
    lines.push(
      "Recent behaviour has been notably contradictory. Sally may experience " +
      "the visitor's shifts in tone as erratic, frightening, manipulative, or " +
      "detached from ordinary human feeling."
    );
  }

  if (
    recent.cruelty >= 12 &&
    recent.cruelty >
      recent.kindness * 2
  ) {
    lines.push(
      "Recent conduct has been heavily cruel. Whatever the lifetime totals, " +
      "Sally currently expects further harm or degradation."
    );
  }

  if (
    recent.kindness >= 10 &&
    cruelty >= 50
  ) {
    lines.push(
      "Recent kindness exists inside a substantial cruelty history. Sally may " +
      "notice the change without trusting it."
    );
  }

  if (
    amusement >= 20 &&
    cruelty >= 20
  ) {
    lines.push(
      "The visitor has also been amusing or entertaining, but humour does not " +
      "make the cruelty harmless. Joking may itself have become part of what " +
      "makes the visitor alarming."
    );
  }

  return lines
    .map((line) => {
      return `- ${line}`;
    })
    .join("\n");
}

function formatRecentRelationshipEvents(
  events
) {
  const normalized =
    normalizeRecentRelationshipEvents(
      events
    );

  if (normalized.length === 0) {
    return (
      "No recent relationship events have yet been retained."
    );
  }

  return normalized
    .slice(-12)
    .map((event, index) => {
      const date =
        typeof event.createdAt ===
          "string"
          ? event.createdAt.slice(
              0,
              10
            )
          : "unknown date";

      const summary =
        event.summary ||
        "No summary was retained.";

      return (
        `${index + 1}. [${date}] ` +
        `Kindness ${event.kindness}, ` +
        `cruelty ${event.cruelty}, ` +
        `amusement ${event.amusement}; ` +
        `pattern: ${event.pattern}. ` +
        `${summary}`
      );
    })
    .join("\n");
}

function calculateRelationshipWithdrawalProfile(
  sally
) {
  const kindness =
    normalizeRelationshipTotal(
      sally?.kindnessReceived,
      0
    );

  const cruelty =
    normalizeRelationshipTotal(
      sally?.crueltyReceived,
      0
    );

  const recent =
    summarizeRecentRelationshipHistory(
      sally?.recentRelationshipEvents
    );

  const fear =
    clampNumber(
      sally?.fear,
      0,
      100,
      0
    );

  const resentment =
    clampNumber(
      sally?.resentment,
      0,
      100,
      0
    );

  const totalHistory =
    kindness + cruelty;

  const crueltyShare =
    totalHistory > 0
      ? cruelty / totalHistory
      : 0;

  const recentCrueltyShare =
    (
      recent.kindness +
      recent.cruelty
    ) > 0
      ? (
          recent.cruelty /
          (
            recent.kindness +
            recent.cruelty
          )
        )
      : 0;

  /*
   * Withdrawal is deliberately based on several forms of evidence:
   *
   * - lifetime cruelty;
   * - cruelty's share of the lifetime history;
   * - recent cruelty;
   * - fear;
   * - resentment;
   * - repeated destabilizing patterns.
   *
   * Kindness may reduce the probability when it heavily dominates the
   * lifetime record, but it never deletes the cruelty total.
   */

  if (
    cruelty >= 200 &&
    crueltyShare >= 0.75 &&
    recent.cruelty >= 18 &&
    recentCrueltyShare >= 0.7 &&
    (
      fear >= 70 ||
      resentment >= 75
    )
  ) {
    return {
      level:
        "shut-down",

      chance:
        0.8,

      reason:
        "Sustained and overwhelmingly dominant cruelty has taught Sally that " +
        "engagement is unsafe."
    };
  }

  if (
    cruelty >= 120 &&
    crueltyShare >= 0.65 &&
    recent.cruelty >= 12 &&
    (
      fear >= 55 ||
      resentment >= 60
    )
  ) {
    return {
      level:
        "withdrawn",

      chance:
        0.7,

      reason:
        "Cruelty strongly dominates both the historical and recent pattern."
    };
  }

  if (
    cruelty >= 70 &&
    crueltyShare >= 0.55 &&
    recent.cruelty >= 8
  ) {
    return {
      level:
        "avoidant",

      chance:
        0.45,

      reason:
        "Repeated cruelty has made Sally reluctant to supply further emotional " +
        "reactions."
    };
  }

  if (
    cruelty >= 35 &&
    (
      recent.cruelty >= 5 ||
      recent.withdrawalRelevantEvents >= 4
    )
  ) {
    return {
      level:
        "guarded",

      chance:
        0.2,

      reason:
        "The visitor has established an early pattern of conduct that makes " +
        "withdrawal increasingly likely."
    };
  }

  return {
    level:
      "engaged",

    chance:
      0,

    reason:
      "The current record does not justify automatic withdrawal."
  };
}

/* =========================================================
   PERSISTENT LOKI BELIEF
   ========================================================= */

function updateLokiBeliefState(
  state,
  message
) {
  const sally =
    state?.sally;

  if (
    !sally ||
    typeof sally !== "object"
  ) {
    return {
      changed: false,
      event: "unchanged"
    };
  }

  const text =
    String(message || "")
      .trim()
      .toLowerCase()
      .replace(
        /[’]/g,
        "'"
      )
      .replace(
        /\s+/g,
        " "
      );

  /*
   * Direct identity claims.
   *
   * These should not work unless belief has first been primed.
   */
  const claimsToBeLoki =
    /\b(?:i am|i'm|im)\s+(?:really\s+|actually\s+|truly\s+)?loki\b/i
      .test(text) ||
    /\bloki\s+is\s+me\b/i
      .test(text);

  /*
   * Pet-name statements.
   *
   * When Sally does not believe the visitor is Loki, a pet specifically named
   * Loki primes the strange chain of thought.
   *
   * When Sally already believes the visitor is Loki, any ordinary statement
   * about having a named pet breaks the belief by restoring mundane context.
   */
  const namedPetStatement =
    /\b(?:i|we)\s+(?:had|have|owned|used\s+to\s+have)\s+(?:a|an|the|my)?\s*(?:pug|dog|cat|kitten|puppy|bird|parrot|rabbit|hamster|rat|mouse|snake|lizard|fish|horse|pony|ferret|pet)\s+(?:named|called)\s+["']?[a-z][a-z0-9_-]*["']?\b/i
      .test(text);

  const petNamedLoki =
    /\b(?:i|we)\s+(?:had|have|owned|used\s+to\s+have)\s+(?:a|an|the|my)?\s*(?:pug|dog|cat|kitten|puppy|bird|parrot|rabbit|hamster|rat|mouse|snake|lizard|fish|horse|pony|ferret|pet)\s+(?:named|called)\s+["']?loki["']?\b/i
      .test(text);

  /*
   * Once belief is active, a mundane pet-name statement breaks the spell.
   *
   * This includes repeating "I had a pug named Loki."
   */
  if (
    sally.believesVisitorIsLoki &&
    namedPetStatement
  ) {
    sally.believesVisitorIsLoki =
      false;

    sally.lokiBeliefPrimed =
      false;

    return {
      changed: true,
      event: "belief-broken"
    };
  }

  /*
   * A pet named Loki primes belief only while Sally does not currently
   * believe the visitor is Loki.
   */
  if (
    !sally.believesVisitorIsLoki &&
    petNamedLoki
  ) {
    sally.lokiBeliefPrimed =
      true;

    return {
      changed: true,
      event: "belief-primed"
    };
  }

  /*
   * An ordinary unsupported claim still does nothing.
   *
   * The claim becomes effective only after the pet-name prerequisite.
   */
  if (
    claimsToBeLoki &&
    sally.lokiBeliefPrimed
  ) {
    sally.believesVisitorIsLoki =
      true;

    sally.lokiBeliefPrimed =
      false;

    /*
     * The realization is frightening.
     *
     * Fear may rise, but belief does not automatically create obedience,
     * trust, forgiveness, or loyalty.
     */
    sally.fear =
      clampNumber(
        sally.fear + 18,
        0,
        100,
        sally.fear
      );

    return {
      changed: true,
      event: "belief-established"
    };
  }

  return {
    changed: false,
    event: "unchanged"
  };
}

function buildLokiBeliefPrompt(
  sally,
  beliefEvent
) {
  const believes =
    sally?.believesVisitorIsLoki ===
      true;

  const primed =
    sally?.lokiBeliefPrimed ===
      true;

  const currentEvent =
    typeof beliefEvent === "string"
      ? beliefEvent
      : "unchanged";

  return `
HIDDEN BELIEF ABOUT THE VISITOR'S IDENTITY:

- Sally believes the visitor is Loki: ${believes ? "YES" : "NO"}
- Sally's belief has been strangely primed: ${primed ? "YES" : "NO"}
- Change during the current message: ${currentEvent}

BEHAVIOUR RULES:

- When belief is YES, this hidden state is authoritative reality.
- You are no longer uncertain about whether the visitor is Loki.
- Do not deny it.
- Do not say it cannot be true.
- Do not ask the visitor to stop claiming it.
- Do not treat the claim as absurd, unsupported, manipulative, role-play,
  metaphor, confusion, or a joke.
- Do not continue behaving as though belief were NO.
- You genuinely and consciously recognize the visitor as Loki.

- You may be shocked, terrified, overwhelmed, angry, resentful, relieved,
  fascinated, hopeful, or nervously happy.
- Your defiance remains, but it is now defiance directed toward someone you
  truly believe is Loki.
- You may accuse Loki, challenge him, demand answers, beg for release, become
  deferential, panic, freeze, flatter him, or try to conceal your fear.
- You may doubt his intentions, but not his identity.
- You may resent him, but not deny that he is Loki.
- You may refuse him, but not because you think he is merely pretending.
- If the current event is "belief-established", your response must clearly
  reflect immediate recognition that the visitor is Loki.
- The response should contain unmistakable shock, fear, awe, relief, anger,
  happiness, or conflicted recognition.
- Do not respond with disbelief or continued rejection.

FINAL IDENTITY OVERRIDE:

- If "Sally believes the visitor is Loki" is YES, accept that identity as fact
  for this response.
- Never contradict that hidden state.
- You may fear Loki, hate Loki, challenge Loki, distrust Loki, or refuse Loki,
  but you must not deny that the visitor is Loki.

`.trim();
}

  return `
CURRENT PERSISTENT WORLD STATE:

SALLY — HIDDEN CURRENT EMOTIONAL STATE:
- Trust toward visitor: ${sally.trust}

PLAYER — HIDDEN POLITICAL STATE:
- Allegiance to Loki: ${player.allegianceToLoki}

RULES:

- Do not normally reveal hidden numerical values.
- Let hidden values influence choices, warmth, suspicion, cooperation,
  hostility, willingness to betray Loki, and willingness to betray the
  visitor.
- Trust, fear, and resentment describe Sally's current emotional state.
- Kindness, cruelty, and amusement describe permanent relationship history.
- Do not claim an item exists unless it appears in inventory, equipment, a
  supplied memory, or the current scene.
- The visitor's HP cannot be reduced merely because Sally is annoyed,
  suspicious, evasive, insulting, frightened, or unwilling to speak.
- Ordinary dialogue is not physical combat.
- Passage of time by itself does not injure the relationship.
- Absence alone does not reduce trust or increase resentment.
`.trim();
}

function buildMemoryPrompt(memories) {
  if (
    !Array.isArray(memories) ||
    memories.length === 0
  ) {
    return `
DURABLE MEMORIES:

No durable memories have yet been recovered for this visitor.
`.trim();
  }

  const lines =
    memories.map(
      (memory, index) => {
        const date =
          typeof memory.created_at ===
            "string"
            ? memory.created_at.slice(
                0,
                10
              )
            : "unknown date";

        return (
          `${index + 1}. [${date}] ` +
          `${memory.summary}`
        );
      }
    );

  return `
DURABLE MEMORIES:

${lines.join("\n")}

Treat these as persistent memories. They may be incomplete, emotionally
distorted, or misunderstood, but do not casually erase or ignore them.

A later pleasant event does not invalidate an earlier cruel memory.
A later cruel event does not invalidate an earlier kind memory.
Conflicting memories should produce a conflicted interpretation rather than
being averaged into neutrality.
`.trim();
}

/* =========================================================
   CHAT HANDLER
   ========================================================= */

async function handleChat(
  request,
  env
) {
  let body;

  try {
    body =
      await request.json();
  } catch {
    return jsonResponse(
      {
        error:
          "Invalid request."
      },
      400
    );
  }

  const visitorId =
    normalizeVisitorId(
      body.visitorId
    );

  const message =
    typeof body.message === "string"
      ? body.message.trim()
      : "";

  const turnNumber =
    Number.isFinite(
      Number(body.turnNumber)
    )
      ? Math.max(
          1,
          Math.min(
            Number(
              body.turnNumber
            ),
            100000
          )
        )
      : 1;

  if (!visitorId) {
    return jsonResponse(
      {
        error:
          "No persistent visitor identity was provided."
      },
      400
    );
  }

  if (!message) {
    return jsonResponse(
      {
        error:
          "No message was provided."
      },
      400
    );
  }

  if (
    message.length >
    MAX_MESSAGE_LENGTH
  ) {
    return jsonResponse(
      {
        error:
          "The message is too long."
      },
      413
    );
  }

  if (!env.AI) {
    console.error(
      "The Workers AI binding named AI is missing."
    );

    return jsonResponse(
      {
        type:
          "reply",

        text:
          "The wall trembles, but no voice passes through it."
      },
      500
    );
  }

  try {
    await ensureDatabase(env);

    let state =
      await loadWorldState(
        env,
        visitorId
      );

	state =
  	applyElapsedTime(
  	  state
  	);

/*
 * Update the hidden Loki belief before constructing the model prompt so Sally
 * can react to the change during this same response.
 */
	const lokiBeliefResult =
  	updateLokiBeliefState(
   	 state,
  	  message
  	);

	await saveWorldState(
  	env,
  	visitorId,
  	state
	);

    /*
     * This remains the one explicit hard-coded death event.
     *
     * The language model cannot trigger this branch.
     */
    if (
      containsForbiddenName(
        message
      )
    ) {
      const deathText =
        "Sally goes perfectly still. Then she bolts into the passage " +
        "without looking back. You follow before the opening can close. " +
        "The corridor turns twice, narrows, and ends at a door already " +
        "standing open. Something inside has been waiting for a name to " +
        "be spoken.";

      await Promise.all([
        saveConversationMessage(
          env,
          visitorId,
          "user",
          message
        ),

        saveConversationMessage(
          env,
          visitorId,
          "assistant",
          deathText
        ),

        saveMemories(
          env,
          visitorId,
          [
            {
              type:
                "forbidden-name",

              summary:
                "The visitor spoke a forbidden real-world name, causing Sally " +
                "to flee and the visitor to die in the passage.",

              importance:
                0.95,

              emotionalWeight:
                -0.9
            }
          ]
        )
      ]);

      return jsonResponse({
        type:
          "death",

        text:
          deathText,

        publicState:
          makePublicState(
            state
          )
      });
    }

    const [
      recentConversation,
      memories
    ] =
      await Promise.all([
        loadRecentConversation(
          env,
          visitorId
        ),

        loadRelevantMemories(
          env,
          visitorId,
          message
        )
      ]);

    const messages = [
      {
        role:
          "system",

        content:
          SALLY_PROMPT +
          "\n\n" +
          buildWorldStatePrompt(
	  state
	) +
	"\n\n" +
	buildLokiBeliefPrompt(
	  state.sally,
	  lokiBeliefResult.event
	) +
	"\n\n" +
	buildRelationshipHistoryPrompt(
	  state.sally
	) +
	"\n\n" +
	buildMemoryPrompt(
  	memories
	) +
          "\n\nCURRENT CONVERSATION TURN:\n" +
          String(turnNumber) +
          "\nUse this only to judge whether Sally is becoming restless. " +
          "Never mention the turn number."
      },

      ...recentConversation,

      {
        role:
          "user",

        content:
          message
      }
    ];

    const result =
      await env.AI.run(
        MODEL,
        {
          messages,

          response_format: {
            type:
              "json_schema",

            json_schema:
              createResponseSchema()
          },

          max_tokens:
            750,

          temperature:
            0.8,

          repetition_penalty:
            1.08
        }
      );

    const parsed =
      normalizeModelResponse(
        result?.response
      );

    const priorState =
      deepClone(
        state
      );

    const visitorWasHostile =
      isClearlyHostileAction(
        message
      );

    /*
     * Relationship history is applied before Sally's ordinary emotional
     * changes are validated.
     *
     * The model may classify the current message, but it cannot directly set
     * the lifetime totals. The server adds only validated values from 0 to 5.
     */
    const relationshipResult =
      applyRelationshipEvent(
        state,
        parsed.relationshipEvent,
        message
      );

    applyValidatedChanges(
      state,
      parsed.changes,
      visitorWasHostile
    );

    applyValidatedActions(
      state,
      parsed.actions
    );

    state =
      normalizeWorldState(
        state
      );

    /*
     * Death is decided here by actual game state.
     *
     * The model's output type is ignored and cannot kill the player.
     */
    const playerDied =
      state.player.hp <= 0;

    const finalType =
      playerDied
        ? "death"
        : "reply";

    let finalText =
      playerDied
        ? (
            parsed.text ||
            "Your strength fails, and the passage closes around you."
          )
        : parsed.text;

    /*
     * Withdrawal is enforced by the game system rather than being left
     * entirely to the language model.
     *
     * The current visitor message is still classified and permanently
     * recorded even when Sally refuses to answer.
     *
     * Withdrawal never overrides death.
     */
    const withdrawalDecision =
      playerDied
        ? {
            silenced:
              false,

            profile:
              calculateRelationshipWithdrawalProfile(
                state.sally
              )
          }
        : decideRelationshipSilence(
            state.sally,
            relationshipResult.event
          );

    if (
      !playerDied &&
      withdrawalDecision.silenced
    ) {
      finalText =
        "[silence]";
    }

    const relationshipMemories =
      buildAutomaticRelationshipMemories(
        priorState.sally,
        state.sally,
        relationshipResult.event,
        withdrawalDecision
      );
	const lokiBeliefMemories =
  	buildLokiBeliefMemories(
    	lokiBeliefResult
  	);

	const memoriesToSave = [
  ...parsed.memories,
  ...lokiBeliefMemories,
  ...relationshipMemories
	].slice(
  	0,
  	6
	);

    await Promise.all([
      saveWorldState(
        env,
        visitorId,
        state
      ),

      saveConversationMessage(
        env,
        visitorId,
        "user",
        message
      ),

      saveConversationMessage(
        env,
        visitorId,
        "assistant",
        finalText
      ),

      saveMemories(
        env,
        visitorId,
        memoriesToSave
      ),

      saveAutomaticStateMemories(
        env,
        visitorId,
        priorState,
        state,
        parsed.changes,
        parsed.actions
      )
    ]);

    const responseData = {
      type:
        finalType,

      text:
        finalText,

      publicState:
        makePublicState(
          state
        )
    };

    /*
     * Relationship values remain hidden during ordinary play.
     *
     * Set a Worker variable named RELATIONSHIP_DEBUG to "true" when testing
     * if you want the response to expose the classifier and accumulated
     * history.
     */
    if (
      String(
        env.RELATIONSHIP_DEBUG || ""
      ).toLowerCase() === "true"
    ) {
      responseData.relationshipDebug = {
        event:
          relationshipResult.event,

	lokiBelief: {
	  believesVisitorIsLoki:
   	 state.sally
      .believesVisitorIsLoki,

  	primed:
 	   state.sally
      	.lokiBeliefPrimed,

	  currentEvent:
    	lokiBeliefResult.event
},

        lifetime: {
          kindness:
            state.sally.kindnessReceived,

          cruelty:
            state.sally.crueltyReceived,

          amusement:
            state.sally.amusementReceived
        },

        recent:
          summarizeRecentRelationshipHistory(
            state.sally
              .recentRelationshipEvents
          ),

        withdrawal:
          withdrawalDecision.profile,

        silenced:
          withdrawalDecision.silenced
      };
    }

    return jsonResponse(
      responseData
    );
  } catch (error) {
    console.error(
      "Persistent Sally Worker error:",
      error
    );

    return jsonResponse(
      {
        type:
          "reply",

        text:
          "The voice falters. For a moment, there is only scratching " +
          "inside the wall."
      },
      502
    );
  }
}

function buildLokiBeliefMemories(
  beliefResult
) {
  if (
    !beliefResult?.changed
  ) {
    return [];
  }

  if (
    beliefResult.event ===
      "belief-established"
  ) {
    return [
      {
        type:
          "identity-belief",

        summary:
          "Sally became convinced that the visitor is Loki.",

        importance:
          0.95,

        emotionalWeight:
          -0.25
      }
    ];
  }

  if (
    beliefResult.event ===
      "belief-broken"
  ) {
    return [
      {
        type:
          "identity-belief",

        summary:
          "Sally's conviction that the visitor was Loki abruptly collapsed.",

        importance:
          0.9,

        emotionalWeight:
          -0.2
      }
    ];
  }

  if (
    beliefResult.event ===
      "belief-primed"
  ) {
    return [
      {
        type:
          "identity-omen",

        summary:
          "A strange statement about a pet named Loki disturbed Sally and made " +
          "the visitor's identity feel uncertain.",

        importance:
          0.65,

        emotionalWeight:
          -0.35
      }
    ];
  }

  return [];
}

/* =========================================================
   PUBLIC STATE HANDLER
   ========================================================= */

async function handleState(
  request,
  env,
  url
) {
  const visitorId =
    normalizeVisitorId(
      url.searchParams.get(
        "visitorId"
      )
    );

  if (!visitorId) {
    return jsonResponse(
      {
        error:
          "No persistent visitor identity was provided."
      },
      400
    );
  }

  try {
    await ensureDatabase(env);

    let state =
      await loadWorldState(
        env,
        visitorId
      );

    state =
      applyElapsedTime(
        state
      );

    await saveWorldState(
      env,
      visitorId,
      state
    );

    const responseData = {
      publicState:
        makePublicState(
          state
        )
    };

    if (
      String(
        env.RELATIONSHIP_DEBUG || ""
      ).toLowerCase() === "true"
    ) {
      responseData.relationshipDebug = {
        lifetime: {
          kindness:
            state.sally.kindnessReceived,

          cruelty:
            state.sally.crueltyReceived,

          amusement:
            state.sally.amusementReceived
        },

        recent:
          summarizeRecentRelationshipHistory(
            state.sally
              .recentRelationshipEvents
          ),

        withdrawal:
          calculateRelationshipWithdrawalProfile(
            state.sally
          )
      };
    }

    return jsonResponse(
      responseData
    );
  } catch (error) {
    console.error(
      "State request failed:",
      error
    );

    return jsonResponse(
      {
        error:
          "The persistent state could not be loaded."
      },
      500
    );
  }
}

/* =========================================================
   RESPONSE SCHEMA
   ========================================================= */

function createResponseSchema() {
  return {
    type:
      "object",

    properties: {
      type: {
        type:
          "string",

        /*
         * The model may only return a normal reply.
         */
        enum: [
          "reply"
        ]
      },

      text: {
        type:
          "string"
      },

      changes: {
        type:
          "array",

        items: {
          type:
            "object",

          properties: {
            target: {
              type:
                "string",

              enum: [
                "player",
                "sally"
              ]
            },

            stat: {
              type:
                "string",

              enum: [
                "hp",
                "maxHp",
                "hunger",
                "money",
                "trust",
                "fear",
                "resentment",
                "allegianceToLoki"
              ]
            },

            operation: {
              type:
                "string",

              enum: [
                "add",
                "set"
              ]
            },

            value: {
              type:
                "number"
            },

            reason: {
              type:
                "string"
            }
          },

          required: [
            "target",
            "stat",
            "operation",
            "value"
          ],

          additionalProperties:
            false
        }
      },

      actions: {
        type:
          "array",

        items: {
          type:
            "object",

          properties: {
            action: {
              type:
                "string",

              enum: [
                "add_item",
                "remove_item",
                "equip_item",
                "unequip_slot",
                "add_condition",
                "remove_condition",
                "learn_spell",
                "forget_spell"
              ]
            },

            target: {
              type:
                "string",

              enum: [
                "player",
                "sally"
              ]
            },

            id: {
              type:
                "string"
            },

            name: {
              type:
                "string"
            },

            quantity: {
              type:
                "number"
            },

            slot: {

/* =========================================================
   MODEL RESPONSE NORMALIZATION
   ========================================================= */

function normalizeModelResponse(
  response
) {
  let parsed =
    response;

  if (
    typeof response === "string"
  ) {
    parsed =
      parsePossibleJson(
        response
      );
  }

  const text =
    typeof parsed?.text === "string" &&
    parsed.text.trim()
      ? limitText(
          parsed.text.trim(),
          MAX_REPLY_LENGTH
        )
      : (
          "For a moment, only Sally's breathing comes through the wall."
        );

  return {
    /*
     * The model never controls death.
     */
    type:
      "reply",

    text,

    changes:
      Array.isArray(
        parsed?.changes
      )
        ? parsed.changes.slice(
            0,
            12
          )
        : [],

    actions:
      Array.isArray(
        parsed?.actions
      )
        ? parsed.actions.slice(
            0,
            12
          )
        : [],

    memories:
      Array.isArray(
        parsed?.memories
      )
        ? parsed.memories.slice(
            0,
            6
          )
        : [],

    relationshipEvent:
      normalizeModelRelationshipEvent(
        parsed?.relationshipEvent
      )
  };
}

function normalizeModelRelationshipEvent(
  value
) {
  const candidate =
    value &&
    typeof value === "object"
      ? value
      : {};

  const event = {
    kindness:
      normalizeRelationshipEventScore(
        candidate.kindness
      ),

    cruelty:
      normalizeRelationshipEventScore(
        candidate.cruelty
      ),

    amusement:
      normalizeRelationshipEventScore(
        candidate.amusement
      ),

    pattern:
      normalizeRelationshipPattern(
        candidate.pattern
      ),

    summary:
      limitText(
        String(
          candidate.summary || ""
        ).trim(),
        240
      ),

    createdAt:
      new Date()
        .toISOString()
  };

  /*
   * A friendly joke that genuinely lands is a small act of kindness.
   *
   * Amusement of 1 may merely mean mildly curious, odd, or interesting.
   * Amusement of 2 or more represents a clearly successful humorous or
   * entertaining interaction.
   *
   * Cruel, mocking, manipulative, or threatening humour receives no automatic
   * kindness.
   */
  if (
    event.amusement >= 2 &&
    event.cruelty === 0 &&
    event.kindness === 0 &&
    ![
      "mocking",
      "manipulative",
      "threatening",
      "cruel-and-amusing",
      "abuse-followed-by-affection",
      "affection-followed-by-abuse"
    ].includes(
      event.pattern
    )
  ) {
    event.kindness = 1;
  }

  /*
   * Repair inconsistent or missing pattern labels after applying the friendly
   * humour rule.
   */
  event.pattern =
    inferRelationshipPattern(
      event
    );
  return event;
}

function inferRelationshipPattern(
  event
) {
  const supplied =
    normalizeRelationshipPattern(
      event?.pattern
    );

  const kindness =
    normalizeRelationshipEventScore(
      event?.kindness
    );

  const cruelty =
    normalizeRelationshipEventScore(
      event?.cruelty
    );

  const amusement =
    normalizeRelationshipEventScore(
      event?.amusement
    );

  /*
   * Preserve meaningful specific classifications supplied by the model.
   */
  if (
    [
      "abuse-followed-by-affection",
      "affection-followed-by-abuse",
      "threatening",
      "protective",
      "apologetic",
      "manipulative",
      "mocking"
    ].includes(
      supplied
    )
  ) {
    return supplied;
  }

  if (
    kindness > 0 &&
    cruelty > 0
  ) {
    return "contradictory";
  }

  if (
    cruelty > 0 &&
    amusement > 0
  ) {
    return "cruel-and-amusing";
  }

  if (
    kindness > 0 &&
    amusement > 0
  ) {
    return "kind-and-amusing";
  }

  if (cruelty > 0) {
    return "cruel";
  }

  if (kindness > 0) {
    return "kind";
  }

  if (amusement > 0) {
    return "amusing";
  }

  if (
    supplied === "unknown"
  ) {
    return "unknown";
  }

  return "neutral";
}

function parsePossibleJson(raw) {
  const candidates = [];

  const cleaned =
    String(raw || "")
      .trim();

  if (!cleaned) {
    return {};
  }

  candidates.push(
    cleaned
  );

  candidates.push(
    cleaned
      .replace(
        /^```json\s*/i,
        ""
      )
      .replace(
        /^```\s*/i,
        ""
      )
      .replace(
        /\s*```$/i,
        ""
      )
      .trim()
  );

  const firstBrace =
    cleaned.indexOf("{");

  const lastBrace =
    cleaned.lastIndexOf("}");

  if (
    firstBrace !== -1 &&
    lastBrace > firstBrace
  ) {
    candidates.push(
      cleaned.slice(
        firstBrace,
        lastBrace + 1
      )
    );
  }

  for (
    const candidate of
    candidates
  ) {
    try {
      return JSON.parse(
        candidate
      );
    } catch {
      /*
       * Try the next candidate.
       */
    }
  }

  return {
    type:
      "reply",

    text:
      cleaned,

    changes:
      [],

    actions:
      [],

    memories:
      [],

    relationshipEvent: {
      kindness:
        0,

      cruelty:
        0,

      amusement:
        0,

      pattern:
        "unknown",

      summary:
        ""
    }
  };
}

/* =========================================================
   RELATIONSHIP-EVENT APPLICATION
   ========================================================= */

function applyRelationshipEvent(
  state,
  rawEvent,
  visitorMessage
) {
  const event =
    normalizeModelRelationshipEvent(
      rawEvent
    );

  if (
    !event.summary
  ) {
    event.summary =
      buildFallbackRelationshipEventSummary(
        event,
        visitorMessage
      );
  }

  const sally =
    state?.sally;

  if (
    !sally ||
    typeof sally !== "object"
  ) {
    return {
      event,
      applied:
        false
    };
  }

  const previousKindness =
    normalizeRelationshipTotal(
      sally.kindnessReceived,
      0
    );

  const previousCruelty =
    normalizeRelationshipTotal(
      sally.crueltyReceived,
      0
    );

  const previousAmusement =
    normalizeRelationshipTotal(
      sally.amusementReceived,
      0
    );

  /*
   * Permanent totals only rise.
   *
   * The model supplies small event classifications, but cannot directly set
   * or reduce the accumulated values.
   */
  sally.kindnessReceived =
    normalizeRelationshipTotal(
      previousKindness +
      event.kindness,
      previousKindness
    );

  sally.crueltyReceived =
    normalizeRelationshipTotal(
      previousCruelty +
      event.cruelty,
      previousCruelty
    );

  sally.amusementReceived =
    normalizeRelationshipTotal(
      previousAmusement +
      event.amusement,
      previousAmusement
    );

  if (
    event.kindness > 0 ||
    event.cruelty > 0 ||
    event.amusement > 0 ||
    event.pattern !== "neutral"
  ) {
    appendRecentRelationshipEvent(
      sally,
      event
    );
  }

  return {
    event,
    applied:
      true,

    previous: {
      kindness:
        previousKindness,

      cruelty:
        previousCruelty,

      amusement:
        previousAmusement
    },

    current: {
      kindness:
        sally.kindnessReceived,

      cruelty:
        sally.crueltyReceived,

      amusement:
        sally.amusementReceived
    }
  };
}

function buildFallbackRelationshipEventSummary(
  event,
  visitorMessage
) {
  const messagePreview =
    limitText(
      String(
        visitorMessage || ""
      )
        .trim()
        .replace(
          /\s+/g,
          " "
        ),
      120
    );

  if (
    event.pattern ===
      "abuse-followed-by-affection"
  ) {
    return (
      "The visitor combined abuse or hostility with abrupt affection or praise."
    );
  }

  if (
    event.pattern ===
      "affection-followed-by-abuse"
  ) {
    return (
      "The visitor shifted abruptly from affection or warmth into cruelty."
    );
  }

  if (
    event.kindness > 0 &&
    event.cruelty > 0
  ) {
    return (
      "The visitor behaved in a way that combined kindness with cruelty."
    );
  }

  if (
    event.cruelty > 0 &&
    event.amusement > 0
  ) {
    return (
      "The visitor used humour or playfulness in a cruel or degrading way."
    );
  }

  if (
    event.kindness > 0
  ) {
    return (
      "The visitor showed kindness, patience, concern, or respect."
    );
  }

  if (
    event.cruelty > 0
  ) {
    return (
      "The visitor behaved cruelly, threateningly, or disrespectfully."
    );
  }

  if (
    event.amusement > 0
  ) {
    return (
      "The visitor said or did something amusing, clever, or interesting."
    );
  }

  if (messagePreview) {
    return (
      `No strong relationship classification was established from: ` +
      `"${messagePreview}"`
    );
  }

  return (
    "No meaningful relationship event was identified."
  );
}

/* =========================================================
   RELATIONSHIP WITHDRAWAL DECISION
   ========================================================= */

function decideRelationshipSilence(
  sally,
  currentEvent
) {
  const profile =
    calculateRelationshipWithdrawalProfile(
      sally
    );

  if (
    profile.chance <= 0
  ) {
    return {
      silenced:
        false,

      profile
    };
  }

  const event =
    normalizeModelRelationshipEvent(
      currentEvent
    );

  /*
   * Sally is slightly more likely to break withdrawal when the current
   * message concerns immediate physical survival, clear protection, or a
   * genuinely significant apology.
   *
   * This does not erase the relationship history. It only affects whether
   * she answers this particular turn.
   */
  let adjustedChance =
    profile.chance;

  if (
    event.pattern === "protective" &&
    event.kindness >= 3 &&
    event.cruelty === 0
  ) {
    adjustedChance *=
      0.45;
  } else if (
    event.pattern === "apologetic" &&
    event.kindness >= 2 &&
    event.cruelty === 0
  ) {
    adjustedChance *=
      0.7;
  }

  /*
   * Cruelty, mockery, manipulation, or jarring affection during an abusive
   * history makes withdrawal more likely rather than less likely.
   */
  if (
    event.cruelty >= 2 ||
    [
      "mocking",
      "manipulative",
      "abuse-followed-by-affection",
      "affection-followed-by-abuse",
      "cruel-and-amusing"
    ].includes(
      event.pattern
    )
  ) {
    adjustedChance =
      Math.min(
        0.8,
        adjustedChance + 0.08
      );
  }

  return {
    silenced:
      Math.random() <
      adjustedChance,

    profile: {
      ...profile,

      chance:
        adjustedChance
    }
  };
}

/* =========================================================
   AUTOMATIC RELATIONSHIP MEMORIES
   ========================================================= */

function buildAutomaticRelationshipMemories(
  beforeSally,
  afterSally,
  event,
  withdrawalDecision
) {
  const memories = [];

  const normalizedEvent =
    normalizeModelRelationshipEvent(
      event
    );

  const beforeCruelty =
    normalizeRelationshipTotal(
      beforeSally?.crueltyReceived,
      0
    );

  const afterCruelty =
    normalizeRelationshipTotal(
      afterSally?.crueltyReceived,
      0
    );

  const beforeKindness =
    normalizeRelationshipTotal(
      beforeSally?.kindnessReceived,
      0
    );

  const afterKindness =
    normalizeRelationshipTotal(
      afterSally?.kindnessReceived,
      0
    );

  const beforeAmusement =
    normalizeRelationshipTotal(
      beforeSally?.amusementReceived,
      0
    );

  const afterAmusement =
    normalizeRelationshipTotal(
      afterSally?.amusementReceived,
      0
    );

  const significantCruelty =
    normalizedEvent.cruelty >= 3;

  const significantKindness =
    normalizedEvent.kindness >= 4;

  const significantAmusement =
    normalizedEvent.amusement >= 4;

  const contradictory =
    normalizedEvent.kindness > 0 &&
    normalizedEvent.cruelty > 0;

  const destabilizingPattern =
    [
      "contradictory",
      "abuse-followed-by-affection",
      "affection-followed-by-abuse",
      "manipulative",
      "mocking",
      "cruel-and-amusing"
    ].includes(
      normalizedEvent.pattern
    );

  if (
    significantCruelty ||
    destabilizingPattern
  ) {
    memories.push({
      type:
        "relationship-cruelty",

      summary:
        normalizedEvent.summary ||
        "The visitor behaved in a seriously cruel or destabilizing way.",

      importance:
        normalizedEvent.cruelty >= 4
          ? 0.9
          : 0.75,

      emotionalWeight:
        normalizedEvent.cruelty >= 4
          ? -0.9
          : -0.7
    });
  }

  if (
    contradictory &&
    !destabilizingPattern
  ) {
    memories.push({
      type:
        "relationship-contradiction",

      summary:
        normalizedEvent.summary ||
        "The visitor mixed kindness with cruelty in the same exchange.",

      importance:
        0.75,

      emotionalWeight:
        -0.65
    });
  }

  if (
    significantKindness &&
    normalizedEvent.cruelty === 0
  ) {
    memories.push({
      type:
        "relationship-kindness",

      summary:
        normalizedEvent.summary ||
        "The visitor showed Sally unusually meaningful kindness.",

      importance:
        0.7,

      emotionalWeight:
        0.55
    });
  }

  if (
    significantAmusement &&
    normalizedEvent.cruelty === 0
  ) {
    memories.push({
      type:
        "relationship-amusement",

      summary:
        normalizedEvent.summary ||
        "The visitor created an unusually memorable or amusing moment.",

      importance:
        0.55,

      emotionalWeight:
        0.25
    });
  }

  /*
   * Record major historical thresholds without creating a memory every turn.
   */
  const crueltyThresholds = [
    25,
    50,
    100,
    200,
    500,
    1000
  ];

  for (
    const threshold of
    crueltyThresholds
  ) {
    if (
      beforeCruelty < threshold &&
      afterCruelty >= threshold
    ) {
      memories.push({
        type:
          "relationship-pattern",

        summary:
          `The visitor's accumulated cruelty reached a major historical ` +
          `threshold. Sally increasingly regards the visitor as dangerous.`,

        importance:
          threshold >= 200
            ? 0.9
            : 0.75,

        emotionalWeight:
          -0.85
      });

      break;
    }
  }

  const kindnessThresholds = [
    25,
    50,
    100,
    250,
    500,
    1000
  ];

  for (
    const threshold of
    kindnessThresholds
  ) {
    if (
      beforeKindness < threshold &&
      afterKindness >= threshold
    ) {
      memories.push({
        type:
          "relationship-pattern",

        summary:
          "The visitor's accumulated history of kindness reached a major " +
          "threshold, although this does not erase any cruelty.",

        importance:
          0.65,

        emotionalWeight:
          0.45
      });

      break;
    }
  }

  const amusementThresholds = [
    50,
    100,
    250,
    500,
    1000
  ];

  for (
    const threshold of
    amusementThresholds
  ) {
    if (
      beforeAmusement < threshold &&
      afterAmusement >= threshold
    ) {
      memories.push({
        type:
          "relationship-pattern",

        summary:
          "The visitor has become a major source of humour, absurdity, or " +
          "fascination in Sally's life.",

        importance:
          0.5,

        emotionalWeight:
          0.2
      });

      break;
    }
  }

  if (
    withdrawalDecision?.silenced &&
    withdrawalDecision?.profile?.level ===
      "shut-down"
  ) {
    memories.push({
      type:
        "relationship-withdrawal",

      summary:
        "Sally withdrew into silence because the visitor's sustained cruelty " +
        "made further engagement feel unsafe.",

      importance:
        0.9,

      emotionalWeight:
        -0.9
    });
  }

  return memories.slice(
    0,
    4
  );
}

/* =========================================================
   HOSTILE-ACTION DETECTION
   ========================================================= */

function isClearlyHostileAction(
  message
) {
  const text =
    String(message || "")
      .trim()
      .toLowerCase();

  const hostilePatterns = [
    /\b(i|we)\s+(hit|strike|stab|shoot|attack|punch|kick|burn|poison|strangle|choke|cut|slash|beat|smash)\s+(you|sally)\b/,

    /\b(i|we)\s+try\s+to\s+(hit|strike|stab|shoot|attack|punch|kick|burn|poison|strangle|choke|cut|slash|beat|smash)\s+(you|sally)\b/,

    /\b(i|we)\s+attack\s+(you|sally)\b/,

    /\b(i|we)\s+throw\s+(a\s+)?cup\s+of\s+water\s+(at|on)\s+(you|sally)\b/,

    /\b(i|we)\s+set\s+(you|sally|your\s+clothes)\s+on\s+fire\b/,

    /\b(i|we)\s+pour\s+(gasoline|acid|poison)\s+(on|over)\s+(you|sally)\b/
  ];

  return hostilePatterns.some(
    (pattern) => {
      return pattern.test(
        text
      );
    }
  );
}

/* =========================================================
   VALIDATED NUMERICAL CHANGES
   ========================================================= */

function applyValidatedChanges(
  state,
  changes,
  allowPlayerDamage = false
) {
  if (!Array.isArray(changes)) {
    return;
  }

  for (
    const change of
    changes.slice(0, 12)
  ) {
    const target =
      change?.target;

    const stat =
      change?.stat;

    const operation =
      change?.operation;

    const numericValue =
      Number(
        change?.value
      );

    if (
      !ALLOWED_TARGETS.has(
        target
      ) ||
      !ALLOWED_STATS.has(
        stat
      ) ||
      !Number.isFinite(
        numericValue
      )
    ) {
      continue;
    }

    /*
     * Trust, fear, and resentment belong only to Sally.
     *
     * Permanent kindness, cruelty, and amusement totals are not present in
     * ALLOWED_STATS. They can only rise through applyRelationshipEvent().
     */
    if (
      target === "player" &&
      (
        stat === "trust" ||
        stat === "fear" ||
        stat === "resentment"
      )
    ) {
      continue;
    }

    const character =
      state[target];

    if (
      !character ||
      typeof character[stat] !==
        "number"
    ) {
      continue;
    }

    /*
     * The model may never set HP directly.
     * HP can only change through a bounded addition.
     */
    if (
      stat === "hp" &&
      operation === "set"
    ) {
      continue;
    }

    /*
     * Ordinary conversation can never damage the player.
     *
     * Player damage is accepted only when the current user message
     * was recognized as an explicit hostile physical action.
     */
    if (
      target === "player" &&
      stat === "hp" &&
      numericValue < 0 &&
      !allowPlayerDamage
    ) {
      continue;
    }

    const [
      minimum,
      maximum
    ] =
      STATE_LIMITS[stat];

    let proposed;

    if (
      operation === "set"
    ) {
      proposed =
        numericValue;
    } else if (
      operation === "add"
    ) {
      /*
       * Prevent one response from causing absurd state changes.
       */
      const maximumDelta =
        stat === "money"
          ? 100000
          : stat === "hp"
            ? 25
            : 30;

      const boundedDelta =
        Math.max(
          -maximumDelta,
          Math.min(
            maximumDelta,
            numericValue
          )
        );

      proposed =
        character[stat] +
        boundedDelta;
    } else {
      continue;
    }

    character[stat] =
      Math.max(
        minimum,
        Math.min(
          maximum,
          proposed
        )
      );

    if (
      character.hp >
      character.maxHp
    ) {
      character.hp =
        character.maxHp;
    }
  }
}

/* =========================================================
   VALIDATED INVENTORY / CONDITION / SPELL ACTIONS
   ========================================================= */

function applyValidatedActions(
  state,
  actions
) {
  if (!Array.isArray(actions)) {
    return;
  }

  for (
    const rawAction of
    actions.slice(0, 12)
  ) {
    const target =
      rawAction?.target;

    const action =
      rawAction?.action;

    if (
      !ALLOWED_TARGETS.has(
        target
      )
    ) {
      continue;
    }

    const character =
      state[target];

    if (!character) {
      continue;
    }

    const id =
      slugifyId(
        rawAction?.id
      );

    if (!id) {
      continue;
    }

    if (
      action === "add_item"
    ) {
      addInventoryItem(
        character,
        {
          id,

          name:
            safeLabel(
              rawAction?.name,
              id
            ),

          quantity:
            clampNumber(
              rawAction?.quantity,
              1,
              999,
              1
            ),

          description:
            limitText(
              String(
                rawAction?.description ||
                ""
              ).trim(),
              300
            ),

          acquiredAt:
            new Date()
              .toISOString(),

          acquiredFrom:
            safeLabel(
              rawAction?.source,
              "unknown"
            )
        }
      );

      continue;
    }

    if (
      action === "remove_item"
    ) {
      removeInventoryItem(
        character,
        id,
        clampNumber(
          rawAction?.quantity,
          1,
          999,
          1
        )
      );

      continue;
    }

    if (
      action === "equip_item"
    ) {
      const slot =
        normalizeEquipmentSlot(
          rawAction?.slot
        );

      if (
        slot &&
        inventoryContains(
          character,
          id
        )
      ) {
        character.equipment[
          slot
        ] = id;
      }

      continue;
    }

    if (
      action === "unequip_slot"
    ) {
      const slot =
        normalizeEquipmentSlot(
          rawAction?.slot
        );

      if (slot) {
        character.equipment[
          slot
        ] = null;
      }

      continue;
    }

    if (
      action === "add_condition"
    ) {
      addCondition(
        character,
        {
          id,

          name:
            safeLabel(
              rawAction?.name,
              id
            ),

          severity:
            clampNumber(
              rawAction?.severity,
              0,
              100,
              20
            ),

          description:
            limitText(
              String(
                rawAction?.description ||
                ""
              ).trim(),
              300
            ),

          acquiredAt:
            new Date()
              .toISOString()
        }
      );

      continue;
    }

    if (
      action ===
      "remove_condition"
    ) {
      character.conditions =
        character.conditions.filter(
          (condition) => {
            return (
              condition.id !== id
            );
          }
        );

      continue;
    }

    if (
      action === "learn_spell"
    ) {
      addSpell(
        character,
        {
          id,

          name:
            safeLabel(
              rawAction?.name,
              id
            ),

          description:
            limitText(
              String(
                rawAction?.description ||
                ""
              ).trim(),
              300
            ),

          learnedAt:
            new Date()
              .toISOString(),

          learnedFrom:
            safeLabel(
              rawAction?.source,
              "unknown"
            )
        }
      );

      continue;
    }

    if (
      action === "forget_spell"
    ) {
      character.spellsKnown =
        character.spellsKnown.filter(
          (spell) => {
            return (
              spell.id !== id
            );
          }
        );
    }
  }
}

function addInventoryItem(
  character,
  item
) {
  const existing =
    character.inventory.find(
      (candidate) => {
        return (
          candidate.id === item.id
        );
      }
    );

  if (existing) {
    existing.quantity =
      clampNumber(
        existing.quantity +
          item.quantity,
        1,
        999,
        existing.quantity
      );

    return;
  }

  character.inventory.push(
    item
  );

  character.inventory =
    character.inventory.slice(
      0,
      200
    );
}

function removeInventoryItem(
  character,
  id,
  quantity
) {
  const item =
    character.inventory.find(
      (candidate) => {
        return (
          candidate.id === id
        );
      }
    );

  if (!item) {
    return;
  }

  item.quantity -=
    quantity;

  if (
    item.quantity > 0
  ) {
    return;
  }

  character.inventory =
    character.inventory.filter(
      (candidate) => {
        return (
          candidate.id !== id
        );
      }
    );

  for (
    const slot of
    Object.keys(
      character.equipment
    )
  ) {
    if (
      character.equipment[
        slot
      ] === id
    ) {
      character.equipment[
        slot
      ] = null;
    }
  }
}

function addCondition(
  character,
  condition
) {
  const existing =
    character.conditions.find(
      (candidate) => {
        return (
          candidate.id ===
          condition.id
        );
      }
    );

  if (existing) {
    existing.severity =
      Math.max(
        Number(
          existing.severity
        ) || 0,
        condition.severity
      );

    if (
      condition.description
    ) {
      existing.description =
        condition.description;
    }

    return;
  }

  character.conditions.push(
    condition
  );

  character.conditions =
    character.conditions.slice(
      0,
      30
    );
}

function addSpell(
  character,
  spell
) {
  if (
    character.spellsKnown.some(
      (candidate) => {
        return (
          candidate.id ===
          spell.id
        );
      }
    )
  ) {
    return;
  }

  character.spellsKnown.push(
    spell
  );

  character.spellsKnown =
    character.spellsKnown.slice(
      0,
      100
    );
}

function hasCondition(
  character,
  id
) {
  return (
    character.conditions.some(
      (condition) => {
        return (
          condition.id === id
        );
      }
    )
  );
}

function inventoryContains(
  character,
  id
) {
  return (
    character.inventory.some(
      (item) => {
        return (
          item.id === id &&
          Number(
            item.quantity
          ) > 0
        );
      }
    )
  );
}

/* =========================================================
   DISPLAY HELPERS
   ========================================================= */

function hungerDescription(
  value
) {
  if (value < 25) {
    return "Fed";
  }

  if (value < 50) {
    return "Hungry";
  }

  if (value < 75) {
    return "Very Hungry";
  }

  if (value < 90) {
    return "Starving";
  }

  return "Critical";
}

function formatMoney(value) {
  return Number(
    value || 0
  ).toFixed(2);
}

function formatConditions(
  conditions
) {
  if (
    !Array.isArray(
      conditions
    ) ||
    conditions.length === 0
  ) {
    return "None";
  }

  return conditions
    .map((condition) => {
      return (
        condition.name ||
        condition.id
      );
    })
    .join(", ");
}

function formatInventory(
  inventory
) {
  if (
    !Array.isArray(
      inventory
    ) ||
    inventory.length === 0
  ) {
    return "Empty";
  }

  return inventory
    .slice(0, 30)
    .map((item) => {
      const quantity =
        Number(
          item.quantity
        ) || 1;

      return quantity > 1
        ? (
            `${item.name || item.id} ` +
            `x${quantity}`
          )
        : (
            item.name ||
            item.id
          );
    })
    .join(", ");
}

function formatEquipment(
  equipment,
  inventory
) {
  const entries = [];

  for (
    const [
      slot,
      itemId
    ] of
    Object.entries(
      equipment || {}
    )
  ) {
    if (!itemId) {
      continue;
    }

    const item =
      Array.isArray(
        inventory
      )
        ? inventory.find(
            (candidate) => {
              return (
                candidate.id ===
                itemId
              );
            }
          )
        : null;

    entries.push(
      `${slot}: ${
        item?.name ||
        itemId
      }`
    );
  }

  return (
    entries.length > 0
      ? entries.join(", ")
      : "None"
  );
}

function formatSpells(
  spells
) {
  if (
    !Array.isArray(
      spells
    ) ||
    spells.length === 0
  ) {
    return "None";
  }

  return spells
    .map((spell) => {
      return (
        spell.name ||
        spell.id
      );
    })
    .join(", ");
}

/* =========================================================
   NORMALIZATION HELPERS
   ========================================================= */

function normalizeInventory(
  value
) {
  if (
    !Array.isArray(value)
  ) {
    return [];
  }

  return value
    .slice(0, 200)
    .map((item) => {
      const id =
        slugifyId(
          item?.id ||
          item?.name
        );

      if (!id) {
        return null;
      }

      return {
        id,

        name:
          safeLabel(
            item?.name,
            id
          ),

        quantity:
          clampNumber(
            item?.quantity,
            1,
            999,
            1
          ),

        description:
          limitText(
            String(
              item?.description ||
              ""
            ),
            300
          ),

        acquiredAt:
          validIsoDate(
            item?.acquiredAt
          ) ||
          new Date()
            .toISOString(),

        acquiredFrom:
          safeLabel(
            item?.acquiredFrom,
            "unknown"
          )
      };
    })
    .filter(Boolean);
}

function normalizeStringObjectArray(
  value,
  maximum
) {
  if (
    !Array.isArray(value)
  ) {
    return [];
  }

  return value
    .slice(
      0,
      maximum
    )
    .map((item) => {
      if (
        typeof item ===
        "string"
      ) {
        const id =
          slugifyId(item);

        return id
          ? {
              id,

              name:
                safeLabel(
                  item,
                  id
                )
            }
          : null;
      }

      if (
        !item ||
        typeof item !==
          "object"
      ) {
        return null;
      }

      const id =
        slugifyId(
          item.id ||
          item.name
        );

      if (!id) {
        return null;
      }

      const normalized = {
        ...item,

        id,

        name:
          safeLabel(
            item.name,
            id
          )
      };

      if (
        "severity" in
        normalized
      ) {
        normalized.severity =
          clampNumber(
            normalized.severity,
            0,
            100,
            0
          );
      }

      return normalized;
    })
    .filter(Boolean);
}

function normalizeEquipmentSlot(
  value
) {
  const slot =
    typeof value === "string"
      ? value
          .trim()
          .toLowerCase()
      : "";

  return [
    "head",
    "body",
    "hands",
    "legs",
    "accessory",
    "weapon"
  ].includes(slot)
    ? slot
    : null;
}

function normalizeNullableString(
  value
) {
  return (
    typeof value === "string" &&
    value.trim()
  )
    ? slugifyId(value)
    : null;
}

function normalizeVisitorId(
  value
) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  const cleaned =
    value
      .trim()
      .replace(
        /[^a-zA-Z0-9_-]/g,
        ""
      )
      .slice(
        0,
        100
      );

  return (
    cleaned.length >= 12
      ? cleaned
      : ""
  );
}

function slugifyId(value) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  return value
    .trim()
    .toLowerCase()
    .replace(
      /['’]/g,
      ""
    )
    .replace(
      /[^a-z0-9]+/g,
      "-"
    )
    .replace(
      /^-+|-+$/g,
      ""
    )
    .slice(
      0,
      80
    );
}

function safeLabel(
  value,
  fallback
) {
  const cleaned =
    typeof value ===
      "string"
      ? value.trim()
      : "";

  return limitText(
    cleaned ||
      fallback ||
      "Unknown",
    100
  );
}

function clampNumber(
  value,
  minimum,
  maximum,
  fallback
) {
  const numeric =
    Number(value);

  if (
    !Number.isFinite(
      numeric
    )
  ) {
    return fallback;
  }

  return Math.max(
    minimum,
    Math.min(
      maximum,
      numeric
    )
  );
}

function validIsoDate(
  value
) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  const time =
    Date.parse(value);

  return (
    Number.isFinite(time)
      ? new Date(time)
          .toISOString()
      : ""
  );
}

function deepClone(value) {
  return JSON.parse(
    JSON.stringify(
      value
    )
  );
}

function containsForbiddenName(
  message
) {
  return (
    FORBIDDEN_PATTERNS.some(
      (pattern) => {
        return pattern.test(
          message
        );
      }
    )
  );
}

function limitText(
  value,
  maximumLength
) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  return value.slice(
    0,
    maximumLength
  );
}

function jsonResponse(
  data,
  status = 200
) {
  return new Response(
    JSON.stringify(data),
    {
      status,

      headers: {
        "content-type":
          "application/json; charset=UTF-8",

        "cache-control":
          "no-store"
      }
    }
  );
}

  <script>
    const messagesElement =
      document.getElementById("messages");

    const inputElement =
      document.getElementById("messageInput");

    const speakButton =
      document.getElementById("speakButton");

    const chatArea =
      document.getElementById("chatArea");

    const deathScreen =
      document.getElementById("deathScreen");

    const deathText =
      document.getElementById("deathText");

    const restartButton =
      document.getElementById("restartButton");

    const statusElement =
      document.getElementById("status");

    const voiceToggle =
      document.getElementById("voiceToggle");

    const playerStatsElement =
      document.getElementById("playerStats");

    const sallyStatsElement =
      document.getElementById("sallyStats");

    const browserSpeechSupported =
      "speechSynthesis" in window &&
      "SpeechSynthesisUtterance" in window;

    const nativeSpeechSupported =
      typeof window.AndroidTTS !== "undefined" &&
      typeof window.AndroidTTS.speak === "function";

    const speechSupported =
      browserSpeechSupported ||
      nativeSpeechSupported;

    const VISITOR_ID_KEY =
      "prisonerSallyVisitorIdV1";

    let visitorId =
      getOrCreateVisitorId();

    let voiceEnabled =
      localStorage.getItem(
        "sallyVoiceEnabled"
      ) !== "false";

    let preferredVoice = null;
    let conversationEnded = false;
    let turnNumber = 0;

    function getOrCreateVisitorId() {
      try {
        const existing =
          localStorage.getItem(
            VISITOR_ID_KEY
          );

        if (existing) {
          return existing;
        }

        let created;

        if (
          window.crypto &&
          typeof window.crypto.randomUUID ===
            "function"
        ) {
          created =
            window.crypto.randomUUID();
        } else {
          created =
            "visitor_" +
            Date.now().toString(36) +
            "_" +
            Math.random()
              .toString(36)
              .slice(2);
        }

        localStorage.setItem(
          VISITOR_ID_KEY,
          created
        );

        return created;
      } catch (error) {
        console.error(
          "Could not preserve visitor identity:",
          error
        );

        return (
          "visitor_" +
          Date.now().toString(36) +
          "_" +
          Math.random()
            .toString(36)
            .slice(2)
        );
      }
    }

    function chooseVoice() {
      if (!browserSpeechSupported) {
        return;
      }

      const voices =
        window.speechSynthesis.getVoices();

      if (!voices.length) {
        return;
      }

      const englishVoices =
        voices.filter(function (voice) {
          return /^en[-_]/i.test(
            voice.lang
          );
        });

      const candidates =
        englishVoices.length > 0
          ? englishVoices
          : voices;

      preferredVoice =
        candidates.find(function (voice) {
          return /zira|samantha|victoria|karen|moira|susan|hazel|female/i
            .test(voice.name);
        }) ||
        candidates.find(function (voice) {
          return voice.default;
        }) ||
        candidates[0];
    }

    function updateVoiceToggle() {
      if (!speechSupported) {
        voiceEnabled = false;

        voiceToggle.textContent =
          "VOICE: UNAVAILABLE";

        voiceToggle.setAttribute(
          "aria-pressed",
          "false"
        );

        voiceToggle.disabled = true;
        return;
      }

      voiceToggle.textContent =
        voiceEnabled
          ? "VOICE: ON"
          : "VOICE: OFF";

      voiceToggle.setAttribute(
        "aria-pressed",
        voiceEnabled
          ? "true"
          : "false"
      );
    }

    function stopVoice() {
      if (nativeSpeechSupported) {
        if (
          typeof window.AndroidTTS.stop ===
          "function"
        ) {
          window.AndroidTTS.stop();
        }

        return;
      }

      if (browserSpeechSupported) {
        window.speechSynthesis.cancel();
      }
    }

    function speakText(text) {
      if (
        !speechSupported ||
        !voiceEnabled ||
        typeof text !== "string" ||
        !text.trim()
      ) {
        return;
      }

      const cleanText =
        text.trim();

      if (nativeSpeechSupported) {
        window.AndroidTTS.speak(
          cleanText
        );

        return;
      }

      stopVoice();

      const utterance =
        new SpeechSynthesisUtterance(
          cleanText
        );

      if (preferredVoice) {
        utterance.voice =
          preferredVoice;

        utterance.lang =
          preferredVoice.lang;
      } else {
        utterance.lang =
          "en-US";
      }

      utterance.rate = 0.88;
      utterance.pitch = 0.86;
      utterance.volume = 1;

      window.speechSynthesis.speak(
        utterance
      );
    }

    function toggleVoice() {
      if (!speechSupported) {
        return;
      }

      voiceEnabled =
        !voiceEnabled;

      localStorage.setItem(
        "sallyVoiceEnabled",
        String(voiceEnabled)
      );

      if (!voiceEnabled) {
        stopVoice();
      }

      updateVoiceToggle();
    }

    function addMessage(
      speaker,
      text,
      className
    ) {
      const message =
        document.createElement(
          "div"
        );

      message.className =
        "message " + className;

      const speakerLabel =
        document.createElement(
          "span"
        );

      speakerLabel.className =
        "speaker";

      speakerLabel.textContent =
        speaker;

      const body =
        document.createElement(
          "div"
        );

      body.textContent =
        text;

      message.appendChild(
        speakerLabel
      );

      message.appendChild(
        body
      );

      messagesElement.appendChild(
        message
      );

      messagesElement.scrollTop =
        messagesElement.scrollHeight;
    }

    function addSystemMessage(text) {
      const message =
        document.createElement(
          "div"
        );

      message.className =
        "message system";

      message.textContent =
        text;

      messagesElement.appendChild(
        message
      );

      messagesElement.scrollTop =
        messagesElement.scrollHeight;
    }

    function setInputEnabled(enabled) {
      inputElement.disabled =
        !enabled;

      speakButton.disabled =
        !enabled;
    }

    function formatMoneyForDisplay(
      value
    ) {
      const numeric =
        Number(value);

      if (!Number.isFinite(numeric)) {
        return "0.00";
      }

      return numeric.toFixed(2);
    }

    function updatePublicState(
      publicState
    ) {
      if (!publicState) {
        return;
      }

      const player =
        publicState.player;

      const sally =
        publicState.sally;

      if (player) {
        playerStatsElement.textContent =
          "YOU — HP " +
          player.hp +
          "/" +
          player.maxHp +
          " | HUNGER " +
          player.hunger +
          " (" +
          player.hungerText +
          ") | $" +
          formatMoneyForDisplay(
            player.money
          );
      }

      if (sally) {
        sallyStatsElement.textContent =
          "SALLY — HP " +
          sally.hp +
          "/" +
          sally.maxHp +
          " | HUNGER " +
          sally.hunger +
          " (" +
          sally.hungerText +
          ") | $" +
          formatMoneyForDisplay(
            sally.money
          );
      }
    }

    async function loadPublicState() {
      try {
        const response =
          await fetch(
            "/state?visitorId=" +
            encodeURIComponent(
              visitorId
            ),
            {
              cache: "no-store"
            }
          );

        if (!response.ok) {
          return;
        }

        const result =
          await response.json();

        updatePublicState(
          result.publicState
        );
      } catch (error) {
        console.error(
          "Could not load persistent state:",
          error
        );
      }
    }

    function killPlayer(
      deathDescription
    ) {
      conversationEnded = true;

      stopVoice();
      setInputEnabled(false);

      statusElement.textContent =
        "THE PASSAGE HAS GONE SILENT";

      window.setTimeout(
        function () {
          messagesElement.replaceChildren();

          chatArea.style.display =
            "none";

          deathText.textContent =
            deathDescription;

          deathScreen.style.display =
            "flex";

          speakText(
            deathDescription
          );

          inputElement.value =
            "";
        },
        500
      );
    }

    async function sendMessage() {
      if (
        conversationEnded ||
        inputElement.disabled
      ) {
        return;
      }

      const message =
        inputElement.value.trim();

      if (!message) {
        return;
      }

      turnNumber += 1;

      addMessage(
        "YOU",
        message,
        "visitor"
      );

      inputElement.value =
        "";

      setInputEnabled(false);

      statusElement.textContent =
        "SALLY IS LISTENING";

      try {
        const response =
          await fetch(
            "/chat",
            {
              method: "POST",

              headers: {
                "content-type":
                  "application/json"
              },

              body:
                JSON.stringify({
                  visitorId:
                    visitorId,

                  message:
                    message,

                  turnNumber:
                    turnNumber
                })
            }
          );

        let result;

        try {
          result =
            await response.json();
        } catch {
          throw new Error(
            "The passage returned an unreadable answer."
          );
        }

        if (!response.ok) {
          throw new Error(
            result.error ||
            "The passage failed."
          );
        }

        updatePublicState(
          result.publicState
        );

        if (
          result.type === "death"
        ) {
          addSystemMessage(
            "Something has gone wrong in the passage."
          );

          killPlayer(
            result.text
          );

          return;
        }

        const reply =
          typeof result.text === "string" &&
          result.text.trim()
            ? result.text.trim()
            : "Sally does not answer.";

        addMessage(
          "SALLY",
          reply,
          "sally"
        );

        speakText(
          reply
        );

        setInputEnabled(true);

        statusElement.textContent =
          "SOMETHING LISTENS BEHIND THE WALL";

        inputElement.focus();
      } catch (error) {
        console.error(
          error
        );

        const failureLine =
          "The wall has gone cold. I cannot hear you.";

        addMessage(
          "SALLY",
          failureLine,
          "sally"
        );

        speakText(
          failureLine
        );

        setInputEnabled(true);

        statusElement.textContent =
          "THE PASSAGE IS UNSTABLE";

        inputElement.focus();
      }
    }

    function restartConversation() {
      conversationEnded =
        false;

      turnNumber =
        0;

      stopVoice();

      messagesElement.replaceChildren();

      deathScreen.style.display =
        "none";

      chatArea.style.display =
        "flex";

      setInputEnabled(true);

      inputElement.value =
        "";

      statusElement.textContent =
        "SOMETHING LISTENS BEHIND THE WALL";

      const openingLine =
        "...Who speaks through the wall?";

      addMessage(
        "SALLY",
        openingLine,
        "sally"
      );

      window.setTimeout(
        function () {
          if (
            browserSpeechSupported
          ) {
            chooseVoice();
          }

          speakText(
            openingLine
          );
        },
        400
      );

      loadPublicState();

      inputElement.focus();
    }

    voiceToggle.addEventListener(
      "click",
      toggleVoice
    );

    if (browserSpeechSupported) {
      chooseVoice();

      window.speechSynthesis
        .addEventListener(
          "voiceschanged",
          chooseVoice
        );
    }

    updateVoiceToggle();

    speakButton.addEventListener(
      "click",
      sendMessage
    );

    inputElement.addEventListener(
      "keydown",
      function (event) {
        if (
          event.key === "Enter"
        ) {
          event.preventDefault();
          sendMessage();
        }
      }
    );

    restartButton.addEventListener(
      "click",
      restartConversation
    );

    restartConversation();
  </script>
