export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    if (
      request.method === "POST" &&
      url.pathname === "/chat"
    ) {
      return handleChat(request, env);
    }

    if (
      request.method === "GET" &&
      url.pathname === "/state"
    ) {
      return handleState(request, env, url);
    }

    return new Response(HTML, {
      headers: {
        "content-type": "text/html; charset=UTF-8",
        "cache-control": "no-store"
      }
    });
  }
};

async function ensureDatabase(env) {
  if (!env.SALLY_DB) {
    throw new Error(
      'Missing D1 binding. Add a D1 binding named "SALLY_DB".'
    );
  }

  await env.SALLY_DB.batch([
    env.SALLY_DB.prepare(`
      CREATE TABLE IF NOT EXISTS world_state (
        visitor_id TEXT PRIMARY KEY,
        state_json TEXT NOT NULL,
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL
      )
    `),

    env.SALLY_DB.prepare(`
      CREATE TABLE IF NOT EXISTS conversation_log (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        visitor_id TEXT NOT NULL,
        role TEXT NOT NULL,
        content TEXT NOT NULL,
        created_at TEXT NOT NULL
      )
    `),

    env.SALLY_DB.prepare(`
      CREATE INDEX IF NOT EXISTS
      idx_conversation_visitor_created
      ON conversation_log(visitor_id, created_at)
    `),

    env.SALLY_DB.prepare(`
      CREATE TABLE IF NOT EXISTS memories (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        visitor_id TEXT NOT NULL,
        type TEXT NOT NULL,
        summary TEXT NOT NULL,
        importance REAL NOT NULL DEFAULT 0.5,
        emotional_weight REAL NOT NULL DEFAULT 0,
        created_at TEXT NOT NULL,
        last_recalled_at TEXT,
        recall_count INTEGER NOT NULL DEFAULT 0
      )
    `),

    env.SALLY_DB.prepare(`
      CREATE INDEX IF NOT EXISTS
      idx_memories_visitor_created
      ON memories(visitor_id, created_at)
    `)
  ]);
}

async function loadWorldState(env, visitorId) {
  const row = await env.SALLY_DB
    .prepare(`
      SELECT state_json
      FROM world_state
      WHERE visitor_id = ?
      LIMIT 1
    `)
    .bind(visitorId)
    .first();

  if (!row?.state_json) {
    const state = createDefaultWorldState();

    await saveWorldState(
      env,
      visitorId,
      state,
      true
    );

    return state;
  }

  try {
    const parsed = JSON.parse(
      row.state_json
    );

    return normalizeWorldState(parsed);
  } catch (error) {
    console.error(
      "Stored world state was unreadable:",
      error
    );

    const state = createDefaultWorldState();

    await saveWorldState(
      env,
      visitorId,
      state,
      false
    );

    return state;
  }
}

async function saveWorldState(
  env,
  visitorId,
  state,
  isNew = false
) {
  const now = new Date().toISOString();

  state.lastUpdatedAt = now;

  const serialized =
    JSON.stringify(
      normalizeWorldState(state)
    );

  if (isNew) {
    await env.SALLY_DB
      .prepare(`
        INSERT OR REPLACE INTO world_state (
          visitor_id,
          state_json,
          created_at,
          updated_at
        )
        VALUES (?, ?, ?, ?)
      `)
      .bind(
        visitorId,
        serialized,
        now,
        now
      )
      .run();

    return;
  }

  await env.SALLY_DB
    .prepare(`
      INSERT INTO world_state (
        visitor_id,
        state_json,
        created_at,
        updated_at
      )
      VALUES (?, ?, ?, ?)

      ON CONFLICT(visitor_id)
      DO UPDATE SET
        state_json = excluded.state_json,
        updated_at = excluded.updated_at
    `)
    .bind(
      visitorId,
      serialized,
      now,
      now
    )
    .run();
}

function normalizeWorldState(candidate) {
  const fallback =
    createDefaultWorldState();

  const player =
    normalizeCharacterState(
      candidate?.player,
      fallback.player,
      false
    );

  const sally =
    normalizeCharacterState(
      candidate?.sally,
      fallback.sally,
      true
    );

  return {
    player,
    sally,

    lastUpdatedAt:
      validIsoDate(
        candidate?.lastUpdatedAt
      ) ||
      fallback.lastUpdatedAt,

    lastHungerUpdateAt:
      validIsoDate(
        candidate?.lastHungerUpdateAt
      ) ||
      fallback.lastHungerUpdateAt
  };
}

function normalizeCharacterState(
  candidate,
  fallback,
  isSally
) {
  const equipment =
    candidate?.equipment &&
    typeof candidate.equipment === "object"
      ? candidate.equipment
      : {};

  const normalized = {
    hp: clampNumber(
      candidate?.hp,
      0,
      1000,
      fallback.hp
    ),

    maxHp: clampNumber(
      candidate?.maxHp,
      1,
      1000,
      fallback.maxHp
    ),

    hunger: clampNumber(
      candidate?.hunger,
      0,
      100,
      fallback.hunger
    ),

    money: clampNumber(
      candidate?.money,
      0,
      1_000_000_000,
      fallback.money
    ),

    conditions:
      normalizeStringObjectArray(
        candidate?.conditions,
        30
      ),

    inventory:
      normalizeInventory(
        candidate?.inventory
      ),

    equipment: {
      head:
        normalizeNullableString(
          equipment.head
        ),

      body:
        normalizeNullableString(
          equipment.body
        ),

      hands:
        normalizeNullableString(
          equipment.hands
        ),

      legs:
        normalizeNullableString(
          equipment.legs
        ),

      accessory:
        normalizeNullableString(
          equipment.accessory
        ),

      weapon:
        normalizeNullableString(
          equipment.weapon
        )
    },

    spellsKnown:
      normalizeStringObjectArray(
        candidate?.spellsKnown,
        50
      ),

    allegianceToLoki:
      clampNumber(
        candidate?.allegianceToLoki,
        -100,
        100,
        fallback.allegianceToLoki
      )
  };

  normalized.hp = Math.min(
    normalized.hp,
    normalized.maxHp
  );

  if (isSally) {
    normalized.trust =
      clampNumber(
        candidate?.trust,
        -100,
        100,
        fallback.trust
      );

    normalized.fear =
      clampNumber(
        candidate?.fear,
        0,
        100,
        fallback.fear
      );

    normalized.resentment =
      clampNumber(
        candidate?.resentment,
        0,
        100,
        fallback.resentment
      );
  }

  return normalized;
}

/* =========================================================
   HUNGER AND PASSAGE OF TIME
   ========================================================= */

function applyElapsedTime(state) {
  const now = Date.now();

  const previous =
    Date.parse(
      state.lastHungerUpdateAt
    );

  if (!Number.isFinite(previous)) {
    state.lastHungerUpdateAt =
      new Date(now).toISOString();

    return state;
  }

  const elapsedHours =
    Math.max(
      0,
      (now - previous) / 3_600_000
    );

  /*
   * Hunger rises by one point every six hours.
   *
   * Offline growth is capped so a long absence does not
   * automatically destroy the entire game state.
   */
  const hungerIncrease =
    Math.min(
      40,
      Math.floor(
        elapsedHours / 6
      )
    );

  if (hungerIncrease <= 0) {
    return state;
  }

  state.player.hunger =
    clampNumber(
      state.player.hunger +
        hungerIncrease,
      0,
      100,
      state.player.hunger
    );

  state.sally.hunger =
    clampNumber(
      state.sally.hunger +
        hungerIncrease,
      0,
      100,
      state.sally.hunger
    );

  applyStarvationDamage(
    state.player
  );

  applyStarvationDamage(
    state.sally
  );

  state.lastHungerUpdateAt =
    new Date(now).toISOString();

  return state;
}

function applyStarvationDamage(
  character
) {
  if (character.hunger < 75) {
    return;
  }

  let damage = 0;

  if (character.hunger >= 95) {
    damage = 8;
  } else if (
    character.hunger >= 90
  ) {
    damage = 5;
  } else {
    damage = 2;
  }

  character.hp =
    Math.max(
      0,
      character.hp - damage
    );

  if (
    character.hunger >= 90 &&
    !hasCondition(
      character,
      "starvation"
    )
  ) {
    character.conditions.push({
      id: "starvation",
      name: "Starvation",
      severity:
        character.hunger,
      description:
        "Prolonged hunger is weakening the body.",
      acquiredAt:
        new Date().toISOString()
    });
  }
}

async function saveConversationMessage(
  env,
  visitorId,
  role,
  content
) {
  const cleaned =
    limitText(
      String(content || "").trim(),
      3000
    );

  if (!cleaned) {
    return;
  }

  await env.SALLY_DB
    .prepare(`
      INSERT INTO conversation_log (
        visitor_id,
        role,
        content,
        created_at
      )
      VALUES (?, ?, ?, ?)
    `)
    .bind(
      visitorId,
      role,
      cleaned,
      new Date().toISOString()
    )
    .run();
}

async function loadRecentConversation(
  env,
  visitorId
) {
  const result =
    await env.SALLY_DB
      .prepare(`
        SELECT
          role,
          content
        FROM conversation_log
        WHERE visitor_id = ?
        ORDER BY id DESC
        LIMIT ?
      `)
      .bind(
        visitorId,
        MAX_RECENT_MESSAGES
      )
      .all();

  const rows =
    Array.isArray(
      result?.results
    )
      ? result.results
      : [];

  return rows
    .reverse()
    .filter((row) => {
      return (
        row &&
        (
          row.role === "user" ||
          row.role === "assistant"
        ) &&
        typeof row.content === "string"
      );
    })
    .map((row) => ({
      role: row.role,

      content:
        limitText(
          row.content,
          MAX_MESSAGE_LENGTH
        )
    }));
}

/* =========================================================
   MEMORY STORAGE
   ========================================================= */

async function saveMemories(
  env,
  visitorId,
  memories
) {
  if (!Array.isArray(memories)) {
    return;
  }

  const statements = [];

  for (
    const memory of
    memories.slice(0, 6)
  ) {
    const summary =
      typeof memory?.summary ===
        "string"
        ? limitText(
            memory.summary.trim(),
            MAX_STORED_MEMORY_LENGTH
          )
        : "";

    if (!summary) {
      continue;
    }

    const type =
      typeof memory?.type ===
        "string"
        ? (
            limitText(
              memory.type.trim(),
              50
            ) || "event"
          )
        : "event";

    const importance =
      clampNumber(
        memory?.importance,
        0,
        1,
        0.5
      );

    const emotionalWeight =
      clampNumber(
        memory?.emotionalWeight,
        -1,
        1,
        0
      );

    statements.push(
      env.SALLY_DB
        .prepare(`
          INSERT INTO memories (
            visitor_id,
            type,
            summary,
            importance,
            emotional_weight,
            created_at
          )
          VALUES (?, ?, ?, ?, ?, ?)
        `)
        .bind(
          visitorId,
          type,
          summary,
          importance,
          emotionalWeight,
          new Date().toISOString()
        )
    );
  }

  if (statements.length > 0) {
    await env.SALLY_DB.batch(
      statements
    );
  }
}

  }

  const messageWords =
    meaningfulWords(message);

  const scored =
    rows.map((row) => {
      const memoryWords =
        meaningfulWords(
          row.summary
        );

      let overlap = 0;

      for (
        const word of
        messageWords
      ) {
        if (
          memoryWords.has(word)
        ) {
          overlap += 1;
        }
      }

      const importance =
        Number(
          row.importance
        ) || 0;

      const emotional =
        Math.abs(
          Number(
            row.emotional_weight
          ) || 0
        );

      const recallBonus =
        Math.min(
          0.15,
          (
            Number(
              row.recall_count
            ) || 0
          ) * 0.005
        );

      const score =
        overlap * 2 +
        importance * 3 +
        emotional +
        recallBonus;

      return {
        ...row,
        score
      };
    });

  scored.sort((a, b) => {
    if (b.score !== a.score) {
      return b.score - a.score;
    }

    return (
      Number(b.id) -
      Number(a.id)
    );
  });

  const selected =
    scored.slice(
      0,
      MAX_CONTEXT_MEMORIES
    );

  if (selected.length > 0) {
    const now =
      new Date().toISOString();

    const updates =
      selected.map(
        (memory) => {
          return env.SALLY_DB
            .prepare(`
              UPDATE memories
              SET
                last_recalled_at = ?,
                recall_count =
                  recall_count + 1
              WHERE id = ?
            `)
            .bind(
              now,
              memory.id
            );
        }
      );

    await env.SALLY_DB.batch(
      updates
    );
  }

  return selected;
}

function meaningfulWords(value) {
  const stopWords = new Set([
    "the",
    "and",
    "that",
    "this",
    "with",
    "from",
    "have",
    "your",
    "you",
    "are",
    "was",
    "were",
    "for",
    "but",
    "not",
    "what",
    "when",
    "where",
    "who",
    "why",
    "how",
    "into",
    "about",
    "they",
    "them",
    "their",
    "there",
    "then",
    "than",
    "will",
    "would",
    "could",
    "should"
  ]);

  return new Set(
    String(value || "")
      .toLowerCase()
      .replace(
        /[^\p{L}\p{N}'-]+/gu,
        " "
      )
      .split(/\s+/)
      .filter((word) => {
        return (
          word.length >= 3 &&
          !stopWords.has(word)
        );
      })
  );
}

function buildMemoryPrompt(memories) {
  if (
    !Array.isArray(memories) ||
    memories.length === 0
  ) {
    return `
DURABLE MEMORIES:

No durable memories have yet been recovered for this visitor.
`.trim();
  }

  const lines =
    memories.map(
      (memory, index) => {
        const date =
          typeof memory.created_at === "string"
            ? memory.created_at.slice(0, 10)
            : "unknown date";

        return (
          `${index + 1}. [${date}] ` +
          `${memory.summary}`
        );
      }
    );

  return `
DURABLE MEMORIES:

${lines.join("\n")}

Treat these as persistent memories. They may be incomplete, emotionally
distorted, or misunderstood, but do not casually erase or ignore them.
`.trim();
}

/* =========================================================
   CHAT HANDLER
   ========================================================= */

async function handleChat(
  request,
  env
) {
  let body;

  try {
    body =
      await request.json();
  } catch {
    return jsonResponse(
      {
        error:
          "Invalid request."
      },
      400
    );
  }

  const visitorId =
    normalizeVisitorId(
      body.visitorId
    );

  const message =
    typeof body.message === "string"
      ? body.message.trim()
      : "";

  const turnNumber =
    Number.isFinite(
      Number(body.turnNumber)
    )
      ? Math.max(
          1,
          Math.min(
            Number(
              body.turnNumber
            ),
            100000
          )
        )
      : 1;

  if (!visitorId) {
    return jsonResponse(
      {
        error:
          "No persistent visitor identity was provided."
      },
      400
    );
  }

  if (!message) {
    return jsonResponse(
      {
        error:
          "No message was provided."
      },
      400
    );
  }

  if (
    message.length >
    MAX_MESSAGE_LENGTH
  ) {
    return jsonResponse(
      {
        error:
          "The message is too long."
      },
      413
    );
  }

  if (!env.AI) {
    console.error(
      "The Workers AI binding named AI is missing."
    );

    return jsonResponse(
      {
        type: "reply",
        text:
          "The wall trembles, but no voice passes through it."
      },
      500
    );
  }

  try {
    await ensureDatabase(env);

    let state =
      await loadWorldState(
        env,
        visitorId
      );

    state =
      applyElapsedTime(state);

    await saveWorldState(
      env,
      visitorId,
      state
    );

    /*
     * This remains the one explicit hard-coded death event.
     *
     * The language model cannot trigger this branch.
     */
    if (
      containsForbiddenName(
        message
      )
    ) {
      const deathText =
        "Sally goes perfectly still. Then she bolts into the passage " +
        "without looking back. You follow before the opening can close. " +
        "The corridor turns twice, narrows, and ends at a door already " +
        "standing open. Something inside has been waiting for a name to " +
        "be spoken.";

      await Promise.all([
        saveConversationMessage(
          env,
          visitorId,
          "user",
          message
        ),

        saveConversationMessage(
          env,
          visitorId,
          "assistant",
          deathText
        ),

        saveMemories(
          env,
          visitorId,
          [
            {
              type:
                "forbidden-name",

              summary:
                "The visitor spoke a forbidden real-world name, causing Sally " +
                "to flee and the visitor to die in the passage.",

              importance:
                0.95,

              emotionalWeight:
                -0.9
            }
          ]
        )
      ]);

      return jsonResponse({
        type: "death",
        text: deathText,

        publicState:
          makePublicState(
            state
          )
      });
    }

    const [
      recentConversation,
      memories
    ] =
      await Promise.all([
        loadRecentConversation(
          env,
          visitorId
        ),

        loadRelevantMemories(
          env,
          visitorId,
          message
        )
      ]);

    const messages = [
      {
        role: "system",

        content:
          SALLY_PROMPT +
          "\n\n" +
          buildWorldStatePrompt(
            state
          ) +
          "\n\n" +
          buildMemoryPrompt(
            memories
          ) +
          "\n\nCURRENT CONVERSATION TURN:\n" +
          String(turnNumber) +
          "\nUse this only to judge whether Sally is becoming restless. " +
          "Never mention the turn number."
      },

      ...recentConversation,

      {
        role: "user",
        content: message
      }
    ];

    const result =
      await env.AI.run(
        MODEL,
        {
          messages,

          response_format: {
            type:
              "json_schema",

            json_schema:
              createResponseSchema()
          },

          max_tokens:
            650,

          temperature:
            0.8,

          repetition_penalty:
            1.08
        }
      );

    const parsed =
      normalizeModelResponse(
        result?.response
      );

    const priorState =
      deepClone(state);

    const visitorWasHostile =
      isClearlyHostileAction(
        message
      );

    applyValidatedChanges(
      state,
      parsed.changes,
      visitorWasHostile
    );

    applyValidatedActions(
      state,
      parsed.actions
    );

    state =
      normalizeWorldState(
        state
      );

    /*
     * Death is decided here by actual game state.
     *
     * The model's output type is ignored and cannot kill the player.
     */
    const playerDied =
      state.player.hp <= 0;

    const finalType =
      playerDied
        ? "death"
        : "reply";

    const finalText =
      playerDied
        ? (
            parsed.text ||
            "Your strength fails, and the passage closes around you."
          )
        : parsed.text;

    await Promise.all([
      saveWorldState(
        env,
        visitorId,
        state
      ),

      saveConversationMessage(
        env,
        visitorId,
        "user",
        message
      ),

      saveConversationMessage(
        env,
        visitorId,
        "assistant",
        finalText
      ),

      saveMemories(
        env,
        visitorId,
        parsed.memories
      ),

      saveAutomaticStateMemories(
        env,
        visitorId,
        priorState,
        state,
        parsed.changes,
        parsed.actions
      )
    ]);

    return jsonResponse({
      type:
        finalType,

      text:
        finalText,

      publicState:
        makePublicState(
          state
        )
    });
  } catch (error) {
    console.error(
      "Persistent Sally Worker error:",
      error
    );

    return jsonResponse(
      {
        type:
          "reply",

        text:
          "The voice falters. For a moment, there is only scratching " +
          "inside the wall."
      },
      502
    );
  }
}

   PUBLIC STATE HANDLER
   ========================================================= */

async function handleState(
  request,
  env,
  url
) {
  const visitorId =
    normalizeVisitorId(
      url.searchParams.get(
        "visitorId"
      )
    );

  if (!visitorId) {
    return jsonResponse(
      {
        error:
          "No persistent visitor identity was provided."
      },
      400
    );
  }

  try {
    await ensureDatabase(env);

    let state =
      await loadWorldState(
        env,
        visitorId
      );

    state =
      applyElapsedTime(
        state
      );

    await saveWorldState(
      env,
      visitorId,
      state
    );

    return jsonResponse({
      publicState:
        makePublicState(
          state
        )
    });
  } catch (error) {
    console.error(
      "State request failed:",
      error
    );

    return jsonResponse(
      {
        error:
          "The persistent state could not be loaded."
      },
      500
    );
  }
}

/* =========================================================
   MODEL RESPONSE NORMALIZATION
   ========================================================= */

function normalizeModelResponse(
  response
) {
  let parsed =
    response;

  if (
    typeof response === "string"
  ) {
    parsed =
      parsePossibleJson(
        response
      );
  }

  const text =
    typeof parsed?.text === "string" &&
    parsed.text.trim()
      ? limitText(
          parsed.text.trim(),
          MAX_REPLY_LENGTH
        )
      : (
          "For a moment, only Sally's breathing comes through the wall."
        );

  return {
    /*
     * The model never controls death.
     */
    type:
      "reply",

    text,

    changes:
      Array.isArray(
        parsed?.changes
      )
        ? parsed.changes.slice(
            0,
            12
          )
        : [],

    actions:
      Array.isArray(
        parsed?.actions
      )
        ? parsed.actions.slice(
            0,
            12
          )
        : [],

    memories:
      Array.isArray(
        parsed?.memories
      )
        ? parsed.memories.slice(
            0,
            6
          )
        : []
  };
}

function parsePossibleJson(raw) {
  const candidates = [];

  const cleaned =
    String(raw || "")
      .trim();

  if (!cleaned) {
    return {};
  }

  candidates.push(
    cleaned
  );

  candidates.push(
    cleaned
      .replace(
        /^```json\s*/i,
        ""
      )
      .replace(
        /^```\s*/i,
        ""
      )
      .replace(
        /\s*```$/i,
        ""
      )
      .trim()
  );

  const firstBrace =
    cleaned.indexOf("{");

  const lastBrace =
    cleaned.lastIndexOf("}");

  if (
    firstBrace !== -1 &&
    lastBrace > firstBrace
  ) {
    candidates.push(
      cleaned.slice(
        firstBrace,
        lastBrace + 1
      )
    );
  }

  for (
    const candidate of
    candidates
  ) {
    try {
      return JSON.parse(
        candidate
      );
    } catch {
      /*
       * Try the next candidate.
       */
    }
  }

  return {
    type:
      "reply",

    text:
      cleaned,

    changes:
      [],

    actions:
      [],

    memories:
      []
  };
}

/* =========================================================
   HOSTILE-ACTION DETECTION
   ========================================================= */

function isClearlyHostileAction(
  message
) {
  const text =
    String(message || "")
      .trim()
      .toLowerCase();

  const hostilePatterns = [
    /\b(i|we)\s+(hit|strike|stab|shoot|attack|punch|kick|burn|poison|strangle|choke|cut|slash|beat|smash)\s+(you|sally)\b/,

    /\b(i|we)\s+try\s+to\s+(hit|strike|stab|shoot|attack|punch|kick|burn|poison|strangle|choke|cut|slash|beat|smash)\s+(you|sally)\b/,

    /\b(i|we)\s+attack\s+(you|sally)\b/,

    /\b(i|we)\s+throw\s+(a\s+)?cup\s+of\s+water\s+(at|on)\s+(you|sally)\b/,

    /\b(i|we)\s+set\s+(you|sally|your\s+clothes)\s+on\s+fire\b/,

    /\b(i|we)\s+pour\s+(gasoline|acid|poison)\s+(on|over)\s+(you|sally)\b/
  ];

  return hostilePatterns.some(
    (pattern) => {
      return pattern.test(
        text
      );
    }
  );
}

/* =========================================================
   VALIDATED NUMERICAL CHANGES
   ========================================================= */

function applyValidatedChanges(
  state,
  changes,
  allowPlayerDamage = false
) {
  if (!Array.isArray(changes)) {
    return;
  }

  for (
    const change of
    changes.slice(0, 12)
  ) {
    const target =
      change?.target;

    const stat =
      change?.stat;

    const operation =
      change?.operation;

    const numericValue =
      Number(
        change?.value
      );

    if (
      !ALLOWED_TARGETS.has(
        target
      ) ||
      !ALLOWED_STATS.has(
        stat
      ) ||
      !Number.isFinite(
        numericValue
      )
    ) {
      continue;
    }

    /*
     * Trust, fear, and resentment belong only to Sally.
     */
    if (
      target === "player" &&
      (
        stat === "trust" ||
        stat === "fear" ||
        stat === "resentment"
      )
    ) {
      continue;
    }

    const character =
      state[target];

    if (
      !character ||
      typeof character[stat] !==
        "number"
    ) {
      continue;
    }

    /*
     * The model may never set HP directly.
     * HP can only change through a bounded addition.
     */
    if (
      stat === "hp" &&
      operation === "set"
    ) {
      continue;
    }

    /*
     * Ordinary conversation can never damage the player.
     *
     * Player damage is accepted only when the current user message
     * was recognized as an explicit hostile physical action.
     */
    if (
      target === "player" &&
      stat === "hp" &&
      numericValue < 0 &&
      !allowPlayerDamage
    ) {
      continue;
    }

    const [
      minimum,
      maximum
    ] =
      STATE_LIMITS[stat];

    let proposed;

    if (
      operation === "set"
    ) {
      proposed =
        numericValue;
    } else if (
      operation === "add"
    ) {
      /*
       * Prevent one response from causing absurd state changes.
       */
      const maximumDelta =
        stat === "money"
          ? 100000
          : stat === "hp"
            ? 25
            : 30;

      const boundedDelta =
        Math.max(
          -maximumDelta,
          Math.min(
            maximumDelta,
            numericValue
          )
        );

      proposed =
        character[stat] +
        boundedDelta;
    } else {
      continue;
    }

    character[stat] =
      Math.max(
        minimum,
        Math.min(
          maximum,
          proposed
        )
      );

    if (
      character.hp >
      character.maxHp
    ) {
      character.hp =
        character.maxHp;
    }
  }
}

/* =========================================================
   VALIDATED INVENTORY / CONDITION / SPELL ACTIONS
   ========================================================= */

function applyValidatedActions(
  state,
  actions
) {
  if (!Array.isArray(actions)) {
    return;
  }

  for (
    const rawAction of
    actions.slice(0, 12)
  ) {
    const target =
      rawAction?.target;

    const action =
      rawAction?.action;

    if (
      !ALLOWED_TARGETS.has(
        target
      )
    ) {
      continue;
    }

    const character =
      state[target];

    if (!character) {
      continue;
    }

    const id =
      slugifyId(
        rawAction?.id
      );

    if (!id) {
      continue;
    }

    if (
      action === "add_item"
    ) {
      addInventoryItem(
        character,
        {
          id,

          name:
            safeLabel(
              rawAction?.name,
              id
            ),

          quantity:
            clampNumber(
              rawAction?.quantity,
              1,
              999,
              1
            ),

          description:
            limitText(
              String(
                rawAction?.description ||
                ""
              ).trim(),
              300
            ),

          acquiredAt:
            new Date()
              .toISOString(),

          acquiredFrom:
            safeLabel(
              rawAction?.source,
              "unknown"
            )
        }
      );

      continue;
    }

    if (
      action === "remove_item"
    ) {
      removeInventoryItem(
        character,
        id,
        clampNumber(
          rawAction?.quantity,
          1,
          999,
          1
        )
      );

      continue;
    }

    if (
      action === "equip_item"
    ) {
      const slot =
        normalizeEquipmentSlot(
          rawAction?.slot
        );

      if (
        slot &&
        inventoryContains(
          character,
          id
        )
      ) {
        character.equipment[
          slot
        ] = id;
      }

      continue;
    }

    if (
      action === "unequip_slot"
    ) {
      const slot =
        normalizeEquipmentSlot(
          rawAction?.slot
        );

      if (slot) {
        character.equipment[
          slot
        ] = null;
      }

      continue;
    }

    if (
      action === "add_condition"
    ) {
      addCondition(
        character,
        {
          id,

          name:
            safeLabel(
              rawAction?.name,
              id
            ),

          severity:
            clampNumber(
              rawAction?.severity,
              0,
              100,
              20
            ),

          description:
            limitText(
              String(
                rawAction?.description ||
                ""
              ).trim(),
              300
            ),

          acquiredAt:
            new Date()
              .toISOString()
        }
      );

      continue;
    }

    if (
      action ===
      "remove_condition"
    ) {
      character.conditions =
        character.conditions.filter(
          (condition) => {
            return (
              condition.id !== id
            );
          }
        );

      continue;
    }

    if (
      action === "learn_spell"
    ) {
      addSpell(
        character,
        {
          id,

          name:
            safeLabel(
              rawAction?.name,
              id
            ),

          description:
            limitText(
              String(
                rawAction?.description ||
                ""
              ).trim(),
              300
            ),

          learnedAt:
            new Date()
              .toISOString(),

          learnedFrom:
            safeLabel(
              rawAction?.source,
              "unknown"
            )
        }
      );

      continue;
    }

    if (
      action === "forget_spell"
    ) {
      character.spellsKnown =
        character.spellsKnown.filter(
          (spell) => {
            return (
              spell.id !== id
            );
          }
        );
    }
  }
}

function addInventoryItem(
  character,
  item
) {
  const existing =
    character.inventory.find(
      (candidate) => {
        return (
          candidate.id === item.id
        );
      }
    );

  if (existing) {
    existing.quantity =
      clampNumber(
        existing.quantity +
          item.quantity,
        1,
        999,
        existing.quantity
      );

    return;
  }

  character.inventory.push(
    item
  );

  character.inventory =
    character.inventory.slice(
      0,
      200
    );
}

function removeInventoryItem(
  character,
  id,
  quantity
) {
  const item =
    character.inventory.find(
      (candidate) => {
        return (
          candidate.id === id
        );
      }
    );

  if (!item) {
    return;
  }

  item.quantity -=
    quantity;

  if (
    item.quantity > 0
  ) {
    return;
  }

  character.inventory =
    character.inventory.filter(
      (candidate) => {
        return (
          candidate.id !== id
        );
      }
    );

  for (
    const slot of
    Object.keys(
      character.equipment
    )
  ) {
    if (
      character.equipment[
        slot
      ] === id
    ) {
      character.equipment[
        slot
      ] = null;
    }
  }
}

function addCondition(
  character,
  condition
) {
  const existing =
    character.conditions.find(
      (candidate) => {
        return (
          candidate.id ===
          condition.id
        );
      }
    );

  if (existing) {
    existing.severity =
      Math.max(
        Number(
          existing.severity
        ) || 0,
        condition.severity
      );

    if (
      condition.description
    ) {
      existing.description =
        condition.description;
    }

    return;
  }

  character.conditions.push(
    condition
  );

  character.conditions =
    character.conditions.slice(
      0,
      30
    );
}

function addSpell(
  character,
  spell
) {
  if (
    character.spellsKnown.some(
      (candidate) => {
        return (
          candidate.id ===
          spell.id
        );
      }
    )
  ) {
    return;
  }

  character.spellsKnown.push(
    spell
  );

  character.spellsKnown =
    character.spellsKnown.slice(
      0,
      100
    );
}

function hasCondition(
  character,
  id
) {
  return (
    character.conditions.some(
      (condition) => {
        return (
          condition.id === id
        );
      }
    )
  );
}

function inventoryContains(
  character,
  id
) {
  return (
    character.inventory.some(
      (item) => {
        return (
          item.id === id &&
          Number(
            item.quantity
          ) > 0
        );
      }
    )
  );
}

/* =========================================================
   DISPLAY HELPERS
   ========================================================= */

function hungerDescription(
  value
) {
  if (value < 25) {
    return "Fed";
  }

  if (value < 50) {
    return "Hungry";
  }

  if (value < 75) {
    return "Very Hungry";
  }

  if (value < 90) {
    return "Starving";
  }

  return "Critical";
}

function formatMoney(value) {
  return Number(
    value || 0
  ).toFixed(2);
}

function formatConditions(
  conditions
) {
  if (
    !Array.isArray(
      conditions
    ) ||
    conditions.length === 0
  ) {
    return "None";
  }

  return conditions
    .map((condition) => {
      return (
        condition.name ||
        condition.id
      );
    })
    .join(", ");
}

function formatInventory(
  inventory
) {
  if (
    !Array.isArray(
      inventory
    ) ||
    inventory.length === 0
  ) {
    return "Empty";
  }

  return inventory
    .slice(0, 30)
    .map((item) => {
      const quantity =
        Number(
          item.quantity
        ) || 1;

      return quantity > 1
        ? (
            `${item.name || item.id} ` +
            `x${quantity}`
          )
        : (
            item.name ||
            item.id
          );
    })
    .join(", ");
}

function formatEquipment(
  equipment,
  inventory
) {
  const entries = [];

  for (
    const [
      slot,
      itemId
    ] of
    Object.entries(
      equipment || {}
    )
  ) {
    if (!itemId) {
      continue;
    }

    const item =
      Array.isArray(
        inventory
      )
        ? inventory.find(
            (candidate) => {
              return (
                candidate.id ===
                itemId
              );
            }
          )
        : null;

    entries.push(
      `${slot}: ${
        item?.name ||
        itemId
      }`
    );
  }

  return (
    entries.length > 0
      ? entries.join(", ")
      : "None"
  );
}

function formatSpells(
  spells
) {
  if (
    !Array.isArray(
      spells
    ) ||
    spells.length === 0
  ) {
    return "None";
  }

  return spells
    .map((spell) => {
      return (
        spell.name ||
        spell.id
      );
    })
    .join(", ");
}

/* =========================================================
   NORMALIZATION HELPERS
   ========================================================= */

function normalizeInventory(
  value
) {
  if (
    !Array.isArray(value)
  ) {
    return [];
  }

  return value
    .slice(0, 200)
    .map((item) => {
      const id =
        slugifyId(
          item?.id ||
          item?.name
        );

      if (!id) {
        return null;
      }

      return {
        id,

        name:
          safeLabel(
            item?.name,
            id
          ),

        quantity:
          clampNumber(
            item?.quantity,
            1,
            999,
            1
          ),

        description:
          limitText(
            String(
              item?.description ||
              ""
            ),
            300
          ),

        acquiredAt:
          validIsoDate(
            item?.acquiredAt
          ) ||
          new Date()
            .toISOString(),

        acquiredFrom:
          safeLabel(
            item?.acquiredFrom,
            "unknown"
          )
      };
    })
    .filter(Boolean);
}

function normalizeStringObjectArray(
  value,
  maximum
) {
  if (
    !Array.isArray(value)
  ) {
    return [];
  }

  return value
    .slice(
      0,
      maximum
    )
    .map((item) => {
      if (
        typeof item ===
        "string"
      ) {
        const id =
          slugifyId(item);

        return id
          ? {
              id,

              name:
                safeLabel(
                  item,
                  id
                )
            }
          : null;
      }

      if (
        !item ||
        typeof item !==
          "object"
      ) {
        return null;
      }

      const id =
        slugifyId(
          item.id ||
          item.name
        );

      if (!id) {
        return null;
      }

      const normalized = {
        ...item,

        id,

        name:
          safeLabel(
            item.name,
            id
          )
      };

      if (
        "severity" in
        normalized
      ) {
        normalized.severity =
          clampNumber(
            normalized.severity,
            0,
            100,
            0
          );
      }

      return normalized;
    })
    .filter(Boolean);
}

function normalizeEquipmentSlot(
  value
) {
  const slot =
    typeof value === "string"
      ? value
          .trim()
          .toLowerCase()
      : "";

  return [
    "head",
    "body",
    "hands",
    "legs",
    "accessory",
    "weapon"
  ].includes(slot)
    ? slot
    : null;
}

function normalizeNullableString(
  value
) {
  return (
    typeof value === "string" &&
    value.trim()
  )
    ? slugifyId(value)
    : null;
}

function normalizeVisitorId(
  value
) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  const cleaned =
    value
      .trim()
      .replace(
        /[^a-zA-Z0-9_-]/g,
        ""
      )
      .slice(
        0,
        100
      );

  return (
    cleaned.length >= 12
      ? cleaned
      : ""
  );
}

function slugifyId(value) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  return value
    .trim()
    .toLowerCase()
    .replace(
      /['’]/g,
      ""
    )
    .replace(
      /[^a-z0-9]+/g,
      "-"
    )
    .replace(
      /^-+|-+$/g,
      ""
    )
    .slice(
      0,
      80
    );
}

function safeLabel(
  value,
  fallback
) {
  const cleaned =
    typeof value ===
      "string"
      ? value.trim()
      : "";

  return limitText(
    cleaned ||
      fallback ||
      "Unknown",
    100
  );
}

function clampNumber(
  value,
  minimum,
  maximum,
  fallback
) {
  const numeric =
    Number(value);

  if (
    !Number.isFinite(
      numeric
    )
  ) {
    return fallback;
  }

  return Math.max(
    minimum,
    Math.min(
      maximum,
      numeric
    )
  );
}

function validIsoDate(
  value
) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  const time =
    Date.parse(value);

  return (
    Number.isFinite(time)
      ? new Date(time)
          .toISOString()
      : ""
  );
}

function deepClone(value) {
  return JSON.parse(
    JSON.stringify(
      value
    )
  );
}

function containsForbiddenName(
  message
) {
  return (
    FORBIDDEN_PATTERNS.some(
      (pattern) => {
        return pattern.test(
          message
        );
      }
    )
  );
}

function limitText(
  value,
  maximumLength
) {
  if (
    typeof value !==
    "string"
  ) {
    return "";
  }

  return value.slice(
    0,
    maximumLength
  );
}

function jsonResponse(
  data,
  status = 200
) {
  return new Response(
    JSON.stringify(data),
    {
      status,

      headers: {
        "content-type":
          "application/json; charset=UTF-8",

        "cache-control":
          "no-store"
      }
    }
  );
}

  <script>
    const messagesElement =
      document.getElementById("messages");

    const inputElement =
      document.getElementById("messageInput");

    const speakButton =
      document.getElementById("speakButton");

    const chatArea =
      document.getElementById("chatArea");

    const deathScreen =
      document.getElementById("deathScreen");

    const deathText =
      document.getElementById("deathText");

    const restartButton =
      document.getElementById("restartButton");

    const statusElement =
      document.getElementById("status");

    const voiceToggle =
      document.getElementById("voiceToggle");

    const playerStatsElement =
      document.getElementById("playerStats");

    const sallyStatsElement =
      document.getElementById("sallyStats");

    const browserSpeechSupported =
      "speechSynthesis" in window &&
      "SpeechSynthesisUtterance" in window;

    const nativeSpeechSupported =
      typeof window.AndroidTTS !== "undefined" &&
      typeof window.AndroidTTS.speak === "function";

    const speechSupported =
      browserSpeechSupported ||
      nativeSpeechSupported;

    const VISITOR_ID_KEY =
      "prisonerSallyVisitorIdV1";

    let visitorId =
      getOrCreateVisitorId();

    let voiceEnabled =
      localStorage.getItem(
        "sallyVoiceEnabled"
      ) !== "false";

    let preferredVoice = null;
    let conversationEnded = false;
    let turnNumber = 0;

    function getOrCreateVisitorId() {
      try {
        const existing =
          localStorage.getItem(
            VISITOR_ID_KEY
          );

        if (existing) {
          return existing;
        }

        let created;

        if (
          window.crypto &&
          typeof window.crypto.randomUUID ===
            "function"
        ) {
          created =
            window.crypto.randomUUID();
        } else {
          created =
            "visitor_" +
            Date.now().toString(36) +
            "_" +
            Math.random()
              .toString(36)
              .slice(2);
        }

        localStorage.setItem(
          VISITOR_ID_KEY,
          created
        );

        return created;
      } catch (error) {
        console.error(
          "Could not preserve visitor identity:",
          error
        );

        return (
          "visitor_" +
          Date.now().toString(36) +
          "_" +
          Math.random()
            .toString(36)
            .slice(2)
        );
      }
    }

    function chooseVoice() {
      if (!browserSpeechSupported) {
        return;
      }

      const voices =
        window.speechSynthesis.getVoices();

      if (!voices.length) {
        return;
      }

      const englishVoices =
        voices.filter(function (voice) {
          return /^en[-_]/i.test(
            voice.lang
          );
        });

      const candidates =
        englishVoices.length > 0
          ? englishVoices
          : voices;

      preferredVoice =
        candidates.find(function (voice) {
          return /zira|samantha|victoria|karen|moira|susan|hazel|female/i
            .test(voice.name);
        }) ||
        candidates.find(function (voice) {
          return voice.default;
        }) ||
        candidates[0];
    }

    function updateVoiceToggle() {
      if (!speechSupported) {
        voiceEnabled = false;

        voiceToggle.textContent =
          "VOICE: UNAVAILABLE";

        voiceToggle.setAttribute(
          "aria-pressed",
          "false"
        );

        voiceToggle.disabled = true;
        return;
      }

      voiceToggle.textContent =
        voiceEnabled
          ? "VOICE: ON"
          : "VOICE: OFF";

      voiceToggle.setAttribute(
        "aria-pressed",
        voiceEnabled
          ? "true"
          : "false"
      );
    }

    function stopVoice() {
      if (nativeSpeechSupported) {
        if (
          typeof window.AndroidTTS.stop ===
          "function"
        ) {
          window.AndroidTTS.stop();
        }

        return;
      }

      if (browserSpeechSupported) {
        window.speechSynthesis.cancel();
      }
    }

    function speakText(text) {
      if (
        !speechSupported ||
        !voiceEnabled ||
        typeof text !== "string" ||
        !text.trim()
      ) {
        return;
      }

      const cleanText =
        text.trim();

      if (nativeSpeechSupported) {
        window.AndroidTTS.speak(
          cleanText
        );

        return;
      }

      stopVoice();

      const utterance =
        new SpeechSynthesisUtterance(
          cleanText
        );

      if (preferredVoice) {
        utterance.voice =
          preferredVoice;

        utterance.lang =
          preferredVoice.lang;
      } else {
        utterance.lang =
          "en-US";
      }

      utterance.rate = 0.88;
      utterance.pitch = 0.86;
      utterance.volume = 1;

      window.speechSynthesis.speak(
        utterance
      );
    }

    function toggleVoice() {
      if (!speechSupported) {
        return;
      }

      voiceEnabled =
        !voiceEnabled;

      localStorage.setItem(
        "sallyVoiceEnabled",
        String(voiceEnabled)
      );

      if (!voiceEnabled) {
        stopVoice();
      }

      updateVoiceToggle();
    }

    function addMessage(
      speaker,
      text,
      className
    ) {
      const message =
        document.createElement(
          "div"
        );

      message.className =
        "message " + className;

      const speakerLabel =
        document.createElement(
          "span"
        );

      speakerLabel.className =
        "speaker";

      speakerLabel.textContent =
        speaker;

      const body =
        document.createElement(
          "div"
        );

      body.textContent =
        text;

      message.appendChild(
        speakerLabel
      );

      message.appendChild(
        body
      );

      messagesElement.appendChild(
        message
      );

      messagesElement.scrollTop =
        messagesElement.scrollHeight;
    }

    function addSystemMessage(text) {
      const message =
        document.createElement(
          "div"
        );

      message.className =
        "message system";

      message.textContent =
        text;

      messagesElement.appendChild(
        message
      );

      messagesElement.scrollTop =
        messagesElement.scrollHeight;
    }

    function setInputEnabled(enabled) {
      inputElement.disabled =
        !enabled;

      speakButton.disabled =
        !enabled;
    }

    function formatMoneyForDisplay(
      value
    ) {
      const numeric =
        Number(value);

      if (!Number.isFinite(numeric)) {
        return "0.00";
      }

      return numeric.toFixed(2);
    }

    function updatePublicState(
      publicState
    ) {
      if (!publicState) {
        return;
      }

      const player =
        publicState.player;

      const sally =
        publicState.sally;

      if (player) {
        playerStatsElement.textContent =
          "YOU — HP " +
          player.hp +
          "/" +
          player.maxHp +
          " | HUNGER " +
          player.hunger +
          " (" +
          player.hungerText +
          ") | $" +
          formatMoneyForDisplay(
            player.money
          );
      }

      if (sally) {
        sallyStatsElement.textContent =
          "SALLY — HP " +
          sally.hp +
          "/" +
          sally.maxHp +
          " | HUNGER " +
          sally.hunger +
          " (" +
          sally.hungerText +
          ") | $" +
          formatMoneyForDisplay(
            sally.money
          );
      }
    }

    async function loadPublicState() {
      try {
        const response =
          await fetch(
            "/state?visitorId=" +
            encodeURIComponent(
              visitorId
            ),
            {
              cache: "no-store"
            }
          );

        if (!response.ok) {
          return;
        }

        const result =
          await response.json();

        updatePublicState(
          result.publicState
        );
      } catch (error) {
        console.error(
          "Could not load persistent state:",
          error
        );
      }
    }

    function killPlayer(
      deathDescription
    ) {
      conversationEnded = true;

      stopVoice();
      setInputEnabled(false);

      statusElement.textContent =
        "THE PASSAGE HAS GONE SILENT";

      window.setTimeout(
        function () {
          messagesElement.replaceChildren();

          chatArea.style.display =
            "none";

          deathText.textContent =
            deathDescription;

          deathScreen.style.display =
            "flex";

          speakText(
            deathDescription
          );

          inputElement.value =
            "";
        },
        500
      );
    }

    async function sendMessage() {
      if (
        conversationEnded ||
        inputElement.disabled
      ) {
        return;
      }

      const message =
        inputElement.value.trim();

      if (!message) {
        return;
      }

      turnNumber += 1;

      addMessage(
        "YOU",
        message,
        "visitor"
      );

      inputElement.value =
        "";

      setInputEnabled(false);

      statusElement.textContent =
        "SALLY IS LISTENING";

      try {
        const response =
          await fetch(
            "/chat",
            {
              method: "POST",

              headers: {
                "content-type":
                  "application/json"
              },

              body:
                JSON.stringify({
                  visitorId:
                    visitorId,

                  message:
                    message,

                  turnNumber:
                    turnNumber
                })
            }
          );

        let result;

        try {
          result =
            await response.json();
        } catch {
          throw new Error(
            "The passage returned an unreadable answer."
          );
        }

        if (!response.ok) {
          throw new Error(
            result.error ||
            "The passage failed."
          );
        }

        updatePublicState(
          result.publicState
        );

        if (
          result.type === "death"
        ) {
          addSystemMessage(
            "Something has gone wrong in the passage."
          );

          killPlayer(
            result.text
          );

          return;
        }

        const reply =
          typeof result.text === "string" &&
          result.text.trim()
            ? result.text.trim()
            : "Sally does not answer.";

        addMessage(
          "SALLY",
          reply,
          "sally"
        );

        speakText(
          reply
        );

        setInputEnabled(true);

        statusElement.textContent =
          "SOMETHING LISTENS BEHIND THE WALL";

        inputElement.focus();
      } catch (error) {
        console.error(
          error
        );

        const failureLine =
          "The wall has gone cold. I cannot hear you.";

        addMessage(
          "SALLY",
          failureLine,
          "sally"
        );

        speakText(
          failureLine
        );

        setInputEnabled(true);

        statusElement.textContent =
          "THE PASSAGE IS UNSTABLE";

        inputElement.focus();
      }
    }

    function restartConversation() {
      conversationEnded =
        false;

      turnNumber =
        0;

      stopVoice();

      messagesElement.replaceChildren();

      deathScreen.style.display =
        "none";

      chatArea.style.display =
        "flex";

      setInputEnabled(true);

      inputElement.value =
        "";

      statusElement.textContent =
        "SOMETHING LISTENS BEHIND THE WALL";

      const openingLine =
        "...Who speaks through the wall?";

      addMessage(
        "SALLY",
        openingLine,
        "sally"
      );

      window.setTimeout(
        function () {
          if (
            browserSpeechSupported
          ) {
            chooseVoice();
          }

          speakText(
            openingLine
          );
        },
        400
      );

      loadPublicState();

      inputElement.focus();
    }

    voiceToggle.addEventListener(
      "click",
      toggleVoice
    );

    if (browserSpeechSupported) {
      chooseVoice();

      window.speechSynthesis
        .addEventListener(
          "voiceschanged",
          chooseVoice
        );
    }

    updateVoiceToggle();

    speakButton.addEventListener(
      "click",
      sendMessage
    );

    inputElement.addEventListener(
      "keydown",
      function (event) {
        if (
          event.key === "Enter"
        ) {
          event.preventDefault();
          sendMessage();
        }
      }
    );

    restartButton.addEventListener(
      "click",
      restartConversation
    );

    restartConversation();
  </script>
</body>
