  function getOrCreateVisitorId(storageKey) {
    try {
      const savedVisitorId = localStorage.getItem(storageKey);

      if (savedVisitorId) {
        return savedVisitorId;
      }

      const newVisitorId = crypto.randomUUID().replace(/-/g, '');
      localStorage.setItem(storageKey, newVisitorId);
      return newVisitorId;
    } catch (error) {
      console.error('Could not save the anonymous visitor ID:', error);
      return crypto.randomUUID().replace(/-/g, '');
    }
  }

function readSharedWorldCookie() {
  const prefix =
    encodeURIComponent(
      WORLD_VISITOR_ID_STORAGE_KEY
    ) + "=";

  const cookies =
    document.cookie.split(";");

  for (const cookie of cookies) {
    const trimmed =
      cookie.trim();

    if (
      trimmed.startsWith(prefix)
    ) {
      return decodeURIComponent(
        trimmed.substring(
          prefix.length
        )
      );
    }
  }

  return null;
}


function writeSharedWorldCookie(
  visitorId
) {
  document.cookie =
    encodeURIComponent(
      WORLD_VISITOR_ID_STORAGE_KEY
    ) +
    "=" +
    encodeURIComponent(
      visitorId
    ) +
    "; Max-Age=315360000" +
    "; Path=/" +
    "; Domain=lokiskingdom.com" +
    "; SameSite=Lax" +
    "; Secure";
}


function getOrCreateWorldVisitorId() {
  try {
    /*
     * Prefer the shared cookie created
     * by the Fortuneteller.
     */
    const sharedVisitorId =
      readSharedWorldCookie();

    if (sharedVisitorId) {
      /*
       * Overwrite the jukebox's old
       * local world ID with the shared ID.
       */
      localStorage.setItem(
        WORLD_VISITOR_ID_STORAGE_KEY,
        sharedVisitorId
      );

      return sharedVisitorId;
    }

    /*
     * Fallback only when no shared
     * cookie exists.
     */
    let visitorId =
      localStorage.getItem(
        WORLD_VISITOR_ID_STORAGE_KEY
      );

    if (!visitorId) {
      visitorId =
        crypto.randomUUID()
          .replace(/-/g, "");

      localStorage.setItem(
        WORLD_VISITOR_ID_STORAGE_KEY,
        visitorId
      );
    }

    writeSharedWorldCookie(
      visitorId
    );

    return visitorId;

  } catch (error) {
    console.error(
      "Could not establish shared world identity:",
      error
    );

    return crypto.randomUUID()
      .replace(/-/g, "");
  }
}

  function loadEmbeddedArtwork(songUrl) {
    const requestNumber = ++artworkRequestNumber;

    /*
     * Keep the existing image visible while the next image is loading.
     * This prevents the player from flashing to an empty black area
     * during song changes.
     */
    if (!window.jsmediatags) {
      console.error(
        'The jsmediatags library did not load. Embedded artwork cannot be read.'
      );

      hideArtwork();
      return;
    }

    window.jsmediatags.read(songUrl, {
      onSuccess(tag) {
        /*
         * A newer song was selected before this request finished.
         * Ignore this older result.
         */
        if (requestNumber !== artworkRequestNumber) {
          return;
        }

        const picture = tag?.tags?.picture;

        if (!picture?.data?.length) {
          hideArtwork();
          return;
        }

        try {
          const bytes = new Uint8Array(picture.data);

          const artworkBlob = new Blob([bytes], {
            type: picture.format || 'image/jpeg'
          });

          /*
           * Create the new URL before revoking the old one.
           * This helps keep the visual transition seamless.
           */
          const newArtworkUrl =
            URL.createObjectURL(artworkBlob);

          const oldArtworkUrl = currentArtworkUrl;

          currentArtworkUrl = newArtworkUrl;

          songArtworkEl.onload = () => {
            /*
             * Only revoke the old URL after the replacement image
             * has successfully loaded.
             */
            if (oldArtworkUrl) {
              URL.revokeObjectURL(oldArtworkUrl);
            }

            songArtworkEl.onload = null;
          };

          songArtworkEl.onerror = () => {
            console.error(
              'The embedded artwork was found but could not be displayed.'
            );

            if (currentArtworkUrl === newArtworkUrl) {
              URL.revokeObjectURL(newArtworkUrl);
              currentArtworkUrl = oldArtworkUrl;
            }

            songArtworkEl.onerror = null;
          };

          songArtworkEl.src = newArtworkUrl;
          songArtworkEl.hidden = false;
        } catch (error) {
          console.error(
            'Could not convert the embedded artwork:',
            error
          );

          hideArtwork();
        }
      },

      onError(error) {
        if (requestNumber !== artworkRequestNumber) {
          return;
        }

        console.error(
          'Could not read embedded artwork from the MP3:',
          error
        );

        hideArtwork();
      }
    });
  }

	function readSynchsafeInteger(bytes, offset) {
	  return (
		(bytes[offset] << 21) |
		(bytes[offset + 1] << 14) |
		(bytes[offset + 2] << 7) |
		bytes[offset + 3]
	  );
	}

	function readBigEndianInteger(bytes, offset) {
	  return (
		(bytes[offset] * 0x1000000) +
		(bytes[offset + 1] << 16) +
		(bytes[offset + 2] << 8) +
		bytes[offset + 3]
	  );
	}

	function findNullTerminator(bytes, start, encoding) {
	  /*
	   * UTF-16 strings end with two zero bytes.
	   * Latin-1 and UTF-8 strings end with one.
	   */
	  if (encoding === 1 || encoding === 2) {
		for (let i = start; i + 1 < bytes.length; i += 2) {
		  if (bytes[i] === 0 && bytes[i + 1] === 0) {
			return i;
		  }
		}

		return bytes.length;
	  }

	  for (let i = start; i < bytes.length; i++) {
		if (bytes[i] === 0) {
		  return i;
		}
	  }

	  return bytes.length;
	}

	function decodeId3Text(bytes, encoding) {
	  if (!bytes?.length) {
		return '';
	  }

	  try {
		if (encoding === 0) {
		  return new TextDecoder('windows-1252')
			.decode(bytes)
			.replace(/\0+$/g, '')
			.trim();
		}

		if (encoding === 1) {
		  /*
		   * Encoding 1 contains a byte-order mark.
		   */
		  if (bytes[0] === 0xff && bytes[1] === 0xfe) {
			return new TextDecoder('utf-16le')
			  .decode(bytes.subarray(2))
			  .replace(/\0+$/g, '')
			  .trim();
		  }

		  if (bytes[0] === 0xfe && bytes[1] === 0xff) {
			return new TextDecoder('utf-16be')
			  .decode(bytes.subarray(2))
			  .replace(/\0+$/g, '')
			  .trim();
		  }

		  return new TextDecoder('utf-16')
			.decode(bytes)
			.replace(/\0+$/g, '')
			.trim();
		}

		if (encoding === 2) {
		  return new TextDecoder('utf-16be')
			.decode(bytes)
			.replace(/\0+$/g, '')
			.trim();
		}

		/*
		 * Encoding 3 is UTF-8.
		 */
		return new TextDecoder('utf-8')
		  .decode(bytes)
		  .replace(/\0+$/g, '')
		  .trim();
	  } catch (error) {
		console.error('Could not decode an ID3 text field:', error);
		return '';
	  }
	}

	async function fetchEmbeddedLyrics(songUrl) {
	  /*
	   * First retrieve the ten-byte ID3 header so we know exactly
	   * how large the metadata section is.
	   */
	  const headerResponse = await fetch(songUrl, {
		headers: {
		  Range: 'bytes=0-9'
		}
	  });

	  if (!headerResponse.ok) {
		throw new Error(
		  `ID3 header request failed with status ${headerResponse.status}`
		);
	  }

	  const headerBuffer = await headerResponse.arrayBuffer();
	  const headerBytes = new Uint8Array(headerBuffer);

	  if (
		headerBytes.length < 10 ||
		headerBytes[0] !== 0x49 ||
		headerBytes[1] !== 0x44 ||
		headerBytes[2] !== 0x33
	  ) {
		return '';
	  }

	  const tagSize = readSynchsafeInteger(headerBytes, 6);
	  const completeTagSize = 10 + tagSize;

	  /*
	   * Protect against malformed metadata requesting an absurdly
	   * large amount of data.
	   */
	  if (completeTagSize > 8 * 1024 * 1024) {
		throw new Error(
		  `ID3 metadata is unexpectedly large: ${completeTagSize} bytes`
		);
	  }

	  const tagResponse = await fetch(songUrl, {
		headers: {
		  Range: `bytes=0-${completeTagSize - 1}`
		}
	  });

	  if (!tagResponse.ok) {
		throw new Error(
		  `ID3 metadata request failed with status ${tagResponse.status}`
		);
	  }

	  const tagBuffer = await tagResponse.arrayBuffer();

	  return extractLyricsEngFromId3(tagBuffer);
	}

	async function loadEmbeddedLyrics(songUrl, song, album) {
	  const requestNumber = ++lyricsRequestNumber;
	  const cacheKey = song.file;

	  albumLyricsEl.textContent = 'Loading lyrics...';

	  if (embeddedLyricsCache.has(cacheKey)) {
		const cachedLyrics = embeddedLyricsCache.get(cacheKey);

		if (cachedLyrics) {
		  albumLyricsEl.textContent = cachedLyrics;
		  return;
		}

		loadAlbumLyrics(album);
		return;
	  }

	  /*
	   * This custom lyrics tag is inside ID3 metadata, so WAV files
	   * should use the existing album LyricsFile directly.
	   */
	  if (!song.file.toLowerCase().endsWith('.mp3')) {
		embeddedLyricsCache.set(cacheKey, null);
		loadAlbumLyrics(album);
		return;
	  }

	  try {
		const embeddedLyrics =
		  await fetchEmbeddedLyrics(songUrl);

		/*
		 * The listener may have selected another song while the
		 * metadata request was running.
		 */
		if (requestNumber !== lyricsRequestNumber) {
		  return;
		}

		if (embeddedLyrics) {
		  embeddedLyricsCache.set(cacheKey, embeddedLyrics);
		  albumLyricsEl.textContent = embeddedLyrics;
		  return;
		}

		embeddedLyricsCache.set(cacheKey, null);
		loadAlbumLyrics(album);
	  } catch (error) {
		if (requestNumber !== lyricsRequestNumber) {
		  return;
		}

		console.warn(
		  `Could not read lyrics-eng from "${song.title}". Using LyricsFile:`,
		  error
		);

		embeddedLyricsCache.set(cacheKey, null);
		loadAlbumLyrics(album);
	  }
	}

  async function loadAlbumLyrics(album) {
    const requestNumber = ++lyricsRequestNumber;

    if (!album.lyricsFile) {
      albumLyricsEl.textContent = 'Lyrics Unknown';
      return;
    }

    albumLyricsEl.textContent = 'Loading lyrics...';

    try {
      const response = await fetch(album.lyricsFile);

      if (!response.ok) {
        throw new Error(
          `Lyrics request failed with status ${response.status}`
        );
      }

      const lyricsText = await response.text();

      /*
       * Ignore this result if another album was selected while
       * the text file was still loading.
       */
      if (requestNumber !== lyricsRequestNumber) {
        return;
      }

      albumLyricsEl.textContent =
        lyricsText.trim().length > 0
          ? lyricsText
          : 'Lyrics Unknown';
    } catch (error) {
      console.error(
        `Could not load lyrics for "${album.title}":`,
        error
      );

      if (requestNumber === lyricsRequestNumber) {
        albumLyricsEl.textContent = 'Lyrics Unknown';
      }
    }
  }
	  
	function extractLyricsEngFromId3(arrayBuffer) {
	  const bytes = new Uint8Array(arrayBuffer);

	  if (
		bytes.length < 10 ||
		bytes[0] !== 0x49 ||
		bytes[1] !== 0x44 ||
		bytes[2] !== 0x33
	  ) {
		console.log('No ID3 header found.');
		return '';
	  }

	  const majorVersion = bytes[3];
	  const tagSize = readSynchsafeInteger(bytes, 6);
	  const tagEnd = Math.min(bytes.length, 10 + tagSize);

	  console.log('ID3 version:', majorVersion);
	  console.log('ID3 tag size:', tagSize);

	  let offset = 10;

	  while (offset + 10 <= tagEnd) {
		const frameId = new TextDecoder('ascii').decode(
		  bytes.subarray(offset, offset + 4)
		);

		if (/^\0+$/.test(frameId)) {
		  break;
		}

		const frameSize =
		  majorVersion === 4
			? readSynchsafeInteger(bytes, offset + 4)
			: readBigEndianInteger(bytes, offset + 4);

		console.log('ID3 frame:', frameId, 'size:', frameSize);

		if (
		  frameSize <= 0 ||
		  offset + 10 + frameSize > tagEnd
		) {
		  console.warn(
			'Invalid ID3 frame size:',
			frameId,
			frameSize
		  );
		  break;
		}

		const frameData = bytes.subarray(
		  offset + 10,
		  offset + 10 + frameSize
		);

		/*
		 * Standard unsynchronised lyrics frame.
		 */
		if (frameId === 'USLT' && frameData.length >= 5) {
		  console.log('Found USLT lyrics frame.');

		  const encoding = frameData[0];

		  const language = new TextDecoder('ascii')
			.decode(frameData.subarray(1, 4))
			.toLowerCase();

		  const descriptionStart = 4;

		  const descriptionEnd = findNullTerminator(
			frameData,
			descriptionStart,
			encoding
		  );

		  const terminatorLength =
			encoding === 1 || encoding === 2 ? 2 : 1;

		  const lyricsStart = Math.min(
			frameData.length,
			descriptionEnd + terminatorLength
		  );

		  const lyrics = decodeId3Text(
			frameData.subarray(lyricsStart),
			encoding
		  );

		  console.log('USLT language:', language);
		  console.log('USLT lyrics length:', lyrics.length);

		  if (lyrics) {
			return lyrics;
		  }
		}

		/*
		 * Custom text field named lyrics-eng.
		 */
		if (frameId === 'TXXX' && frameData.length > 1) {
		  const encoding = frameData[0];

		  const descriptionEnd = findNullTerminator(
			frameData,
			1,
			encoding
		  );

		  const description = decodeId3Text(
			frameData.subarray(1, descriptionEnd),
			encoding
		  );

		  console.log('TXXX description:', description);

		  const terminatorLength =
			encoding === 1 || encoding === 2 ? 2 : 1;

		  const valueStart = Math.min(
			frameData.length,
			descriptionEnd + terminatorLength
		  );

		  if (description.toLowerCase() === 'lyrics-eng') {
			const lyrics = decodeId3Text(
			  frameData.subarray(valueStart),
			  encoding
			);

			console.log(
			  'lyrics-eng text length:',
			  lyrics.length
			);

			return lyrics;
		  }
		}

		offset += 10 + frameSize;
	  }

	  console.log('No readable embedded lyrics frame found.');
	  return '';
	}

  function updateSelectedAlbumButton() {
    const albumButtons =
      albumsEl.querySelectorAll('.album-button');

    albumButtons.forEach((button, index) => {
      const isSelected = index === currentAlbumIndex;

      button.classList.toggle('selected', isSelected);

      button.setAttribute(
        'aria-pressed',
        isSelected ? 'true' : 'false'
      );
    });
  }

  function updateSelectedSongButton() {
    const songButtons =
      songsEl.querySelectorAll('.song-button');

    songButtons.forEach((button, index) => {
      const isSelected = index === currentSongIndex;

      button.classList.toggle('selected', isSelected);

      if (isSelected) {
        button.setAttribute('aria-current', 'true');
      } else {
        button.removeAttribute('aria-current');
      }
    });
  }

  function renderAlbums() {
    albumsEl.innerHTML = '';

    albums.forEach((album, index) => {
      const button = document.createElement('button');

      button.type = 'button';
      button.textContent = album.title;
      button.className = 'album-button';
      button.setAttribute('aria-pressed', 'false');

      button.addEventListener('click', () => {
        currentAlbumIndex = index;
        currentSongIndex = 0;

        renderSongs();
        updateSelectedAlbumButton();
      });

      albumsEl.appendChild(button);
    });

    updateSelectedAlbumButton();
  }

  function renderSongs() {
    const album = albums[currentAlbumIndex];

    if (!album) {
      albumTitleEl.textContent = 'No albums available';
      nowPlayingEl.textContent = 'Nothing playing';
      songsEl.innerHTML = '';
      albumLyricsEl.textContent = 'Lyrics Unknown';
      return;
    }

	albumTitleEl.textContent = album.title;
	nowPlayingEl.textContent = 'Nothing playing';
	songsEl.innerHTML = '';
	albumLyricsEl.textContent = 'Select a song to view lyrics.';

	if (!album.songs?.length) {
	  const message = document.createElement('p');

	  message.textContent = 'No songs available.';
	  message.className = 'empty-message';

	  songsEl.appendChild(message);
	  return;
	}

    album.songs.forEach((song, index) => {
      const row = document.createElement('div');
      const songButton = document.createElement('button');
      const likeButton = document.createElement('button');

      row.className = 'song-row';

      songButton.type = 'button';
      songButton.textContent = song.title;
      songButton.className = 'song-button';

      songButton.addEventListener('click', () => {
        playSong(index);
      });

      likeButton.type = 'button';
      likeButton.className = 'like-button';
      likeButton.dataset.songKey = getSongLikeKey(song);
      updateLikeButton(likeButton, song);

      likeButton.addEventListener('click', (event) => {
        event.preventDefault();
        event.stopPropagation();

        toggleSongLike(song, likeButton);
      });

      row.appendChild(songButton);
      row.appendChild(likeButton);
      songsEl.appendChild(row);
    });

    requestLikeStatuses(album);
  }

  function playSong(index) {
    const album = albums[currentAlbumIndex];

    if (!album?.songs?.length) {
      return;
    }

    const song = album.songs[index];

    if (!song) {
      return;
    }

    currentSongIndex = index;

    const songUrl = getSongUrl(song);

    audio.src = songUrl;
    startRewardSession(song);

    nowPlayingEl.textContent =
      `Now Playing: ${song.title}`;

    updateSelectedSongButton();
    loadEmbeddedArtwork(songUrl);
	loadEmbeddedLyrics(songUrl, song, album);

    audio.play().catch((error) => {
      console.error(
        'Audio playback could not begin:',
        error
      );
    });
  }

  function nextSong() {
    const album = albums[currentAlbumIndex];

    if (!album?.songs?.length) {
      return;
    }

    currentSongIndex =
      (currentSongIndex + 1) %
      album.songs.length;

    playSong(currentSongIndex);
  }

  function prevSong() {
    const album = albums[currentAlbumIndex];

    if (!album?.songs?.length) {
      return;
    }

    currentSongIndex =
      (
        currentSongIndex -
        1 +
        album.songs.length
      ) % album.songs.length;

    playSong(currentSongIndex);
  }

  nextBtn.addEventListener('click', nextSong);
  prevBtn.addEventListener('click', prevSong);

  audio.addEventListener('loadedmetadata', () => {
    if (!rewardSessionId) {
      return;
    }

    sendRewardAction('metadata', {
      duration: Number.isFinite(audio.duration) ? audio.duration : null
    }).catch((error) => {
      console.error('Could not update reward-session metadata:', error);
    });
  });

  audio.addEventListener('play', () => {
    lastPlaybackTime = audio.currentTime;
  });

  audio.addEventListener('pause', () => {
    lastPlaybackTime = null;
  });

  audio.addEventListener('seeking', () => {
    lastPlaybackTime = null;
  });

  audio.addEventListener('timeupdate', () => {
    if (audio.paused || audio.seeking || !rewardSessionId) {
      return;
    }

    const currentTime = audio.currentTime;

    if (lastPlaybackTime === null) {
      lastPlaybackTime = currentTime;
      return;
    }

    const delta = currentTime - lastPlaybackTime;
    lastPlaybackTime = currentTime;

    // Count ordinary forward playback only. Large jumps are treated as seeks.
    if (delta > 0 && delta <= 1.5) {
      listeningSeconds += delta;
    }

    if (listeningSeconds >= 15 && !thresholdRewardClaimed) {
      claimThresholdReward();
    }
  });

  audio.addEventListener('ended', async () => {
    await claimCompletionReward();
    nextSong();
  });

  /*
   * Release the artwork object URL if the page is unloaded.
   */
  window.addEventListener('pagehide', () => {
    clearPreviousArtworkUrl();
  });

  renderAlbums();
  renderSongs();
  loadWorldMoney();
</script>
