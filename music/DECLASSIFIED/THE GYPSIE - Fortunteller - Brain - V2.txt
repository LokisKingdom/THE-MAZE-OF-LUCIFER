// THE GYPSIE — Loki's Kingdom resident Worker
// Cloudflare Worker + Workers AI + shared D1 world state + optional KV lore corpus.
//
// Required bindings:
//   AI        Workers AI binding
//   WORLD_DB  D1 binding pointing to the SAME database used by Sally
//
// Optional binding:
//   LORE_KV   KV namespace containing key: lokis-kingdom:lore-bible
//
// Recommended variables:
//   CHARACTER_DEBUG=false
//
// The Worker reads and updates the player's existing `world_state` row, while
// storing the Gypsie's private state, conversations, memories, tarot history,
// and payments in separate tables.

export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    if (request.method === "POST" && url.pathname === "/chat") {
      return handleChat(request, env);
    }

    if (request.method === "GET" && url.pathname === "/state") {
      return handleState(request, env, url);
    }

    if (request.method === "GET" && url.pathname === "/health") {
      return jsonResponse({ ok: true, resident: "the-gypsie" });
    }

    return new Response(HTML, {
      headers: {
        "content-type": "text/html; charset=UTF-8",
        "cache-control": "no-store"
      }
    });
  }
};

/* =========================================================
   DATABASE
   ========================================================= */

async function ensureDatabase(env) {
  if (!env.WORLD_DB) {
    throw new Error(
      'Missing D1 binding. Add a D1 binding named "WORLD_DB".'
    );
  }

  await env.WORLD_DB.batch([
    env.WORLD_DB.prepare(`
      CREATE TABLE IF NOT EXISTS gypsie_state (
        visitor_id TEXT PRIMARY KEY,
        state_json TEXT NOT NULL,
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL
      )
    `),

    env.WORLD_DB.prepare(`
      CREATE TABLE IF NOT EXISTS gypsie_conversation_log (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        visitor_id TEXT NOT NULL,
        role TEXT NOT NULL,
        content TEXT NOT NULL,
        created_at TEXT NOT NULL
      )
    `),

    env.WORLD_DB.prepare(`
      CREATE INDEX IF NOT EXISTS
      idx_gypsie_conversation_visitor
      ON gypsie_conversation_log(visitor_id, id)
    `),

    env.WORLD_DB.prepare(`
      CREATE TABLE IF NOT EXISTS gypsie_memories (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        visitor_id TEXT NOT NULL,
        type TEXT NOT NULL,
        summary TEXT NOT NULL,
        importance REAL NOT NULL DEFAULT 0.5,
        emotional_weight REAL NOT NULL DEFAULT 0,
        created_at TEXT NOT NULL,
        last_recalled_at TEXT,
        recall_count INTEGER NOT NULL DEFAULT 0
      )
    `),

    env.WORLD_DB.prepare(`
      CREATE INDEX IF NOT EXISTS
      idx_gypsie_memories_visitor
      ON gypsie_memories(visitor_id, id)
    `),

    env.WORLD_DB.prepare(`
      CREATE TABLE IF NOT EXISTS gypsie_tarot_history (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        visitor_id TEXT NOT NULL,
        card_name TEXT NOT NULL,
        orientation TEXT NOT NULL,
        question TEXT NOT NULL,
        interpretation TEXT NOT NULL,
        fee_paid INTEGER NOT NULL DEFAULT 0,
        created_at TEXT NOT NULL
      )
    `),

    env.WORLD_DB.prepare(`
      CREATE INDEX IF NOT EXISTS
      idx_gypsie_tarot_visitor
      ON gypsie_tarot_history(visitor_id, id)
    `),

    env.WORLD_DB.prepare(`
      CREATE TABLE IF NOT EXISTS gypsie_payments (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        visitor_id TEXT NOT NULL,
        amount INTEGER NOT NULL,
        purpose TEXT NOT NULL,
        created_at TEXT NOT NULL
      )
    `)
  ]);
}

   ========================================================= */

async function loadSharedWorldState(env, visitorId) {
  const row = await env.WORLD_DB
    .prepare(`
      SELECT state_json
      FROM world_state
      WHERE visitor_id = ?
      LIMIT 1
    `)
    .bind(visitorId)
    .first();

  if (!row?.state_json) {
    const now = new Date().toISOString();

    const fallback = {
      player: {
        hp: 100,
        maxHp: 100,
        hunger: 5,
        money: 0,
        conditions: [],
        inventory: [],
        equipment: {
          head: null,
          body: null,
          hands: null,
          legs: null,
          accessory: null,
          weapon: null
        },
        spellsKnown: [],
        allegianceToLoki: 0
      },
      lastUpdatedAt: now,
      lastHungerUpdateAt: now
    };

normalizeSharedPlayer(fallback);
applyDailyGold(fallback);

await env.WORLD_DB
  .prepare(`
    INSERT INTO world_state (
      visitor_id,
      state_json,
      created_at,
      updated_at
    )
    VALUES (?, ?, ?, ?)
  `)
  .bind(
    visitorId,
    JSON.stringify(fallback),
    now,
    now
  )
  .run();

return fallback;  }

try {
  const worldState =
    JSON.parse(row.state_json);

  normalizeSharedPlayer(worldState);

  const receivedDailyGold =
    applyDailyGold(worldState);

  if (receivedDailyGold) {
    await saveSharedWorldState(
      env,
      visitorId,
      worldState
    );
  }

  return worldState;
} catch (error) {
    console.error("Unreadable shared world state:", error);

    throw new Error(
      "The visitor's shared world state could not be read."
    );
  }
}

async function saveSharedWorldState(
  env,
  visitorId,
  worldState
) {
  const now = new Date().toISOString();

  worldState.lastUpdatedAt = now;

  await env.WORLD_DB
    .prepare(`
      UPDATE world_state
      SET
        state_json = ?,
        updated_at = ?
      WHERE visitor_id = ?
    `)
    .bind(
      JSON.stringify(worldState),
      now,
      visitorId
    )
    .run();
}

function normalizeSharedPlayer(worldState) {
  if (
    !worldState.player ||
    typeof worldState.player !== "object"
  ) {
    worldState.player = {};
  }

  const player = worldState.player;

  player.hp = clampNumber(
    player.hp,
    0,
    1000,
    100
  );

  player.maxHp = clampNumber(
    player.maxHp,
    1,
    1000,
    100
  );

  player.hp = Math.min(
    player.hp,
    player.maxHp
  );

  player.hunger = clampNumber(
    player.hunger,
    0,
    100,
    5
  );

  player.money = Math.floor(
    clampNumber(
      player.money,
      0,
      1_000_000_000,
      0
    )
  );

player.lastDailyGoldDate =
  typeof player.lastDailyGoldDate === "string"
    ? player.lastDailyGoldDate
    : null;

  player.conditions = Array.isArray(
    player.conditions
  )
    ? player.conditions
    : [];

  player.inventory = Array.isArray(
    player.inventory
  )
    ? player.inventory
    : [];

  player.equipment =
    player.equipment &&
    typeof player.equipment === "object"
      ? player.equipment
      : {
          head: null,
          body: null,
          hands: null,
          legs: null,
          accessory: null,
          weapon: null
        };

  player.spellsKnown = Array.isArray(
    player.spellsKnown
  )
    ? player.spellsKnown
    : [];

  player.allegianceToLoki = clampNumber(
    player.allegianceToLoki,
    -100,
    100,
    0
  );

  return player;
}

/* =========================================================
   GYPSIE STATE
   ========================================================= */

async function loadGypsieState(env, visitorId) {
  const row = await env.WORLD_DB
    .prepare(`
      SELECT state_json
      FROM gypsie_state
      WHERE visitor_id = ?
      LIMIT 1
    `)
    .bind(visitorId)
    .first();

  if (!row?.state_json) {
    const state = createDefaultGypsieState();

    await saveGypsieState(
      env,
      visitorId,
      state,
      true
    );

    return state;
  }

  try {
    return normalizeGypsieState(
      JSON.parse(row.state_json)
    );
  } catch (error) {
    console.error(
      "Unreadable Gypsie state:",
      error
    );

    const state = createDefaultGypsieState();

    await saveGypsieState(
      env,
      visitorId,
      state,
      false
    );

    return state;
  }
}

async function saveGypsieState(
  env,
  visitorId,
  state,
  isNew = false
) {
  const now = new Date().toISOString();

  state.updatedAt = now;

  const serialized = JSON.stringify(
    normalizeGypsieState(state)
  );

  if (isNew) {
    await env.WORLD_DB
      .prepare(`
        INSERT OR REPLACE INTO gypsie_state (
          visitor_id,
          state_json,
          created_at,
          updated_at
        )
        VALUES (?, ?, ?, ?)
      `)
      .bind(
        visitorId,
        serialized,
        state.createdAt || now,
        now
      )
      .run();

    return;
  }

  await env.WORLD_DB
    .prepare(`
      INSERT INTO gypsie_state (
        visitor_id,
        state_json,
        created_at,
        updated_at
      )
      VALUES (?, ?, ?, ?)

      ON CONFLICT(visitor_id)
      DO UPDATE SET
        state_json = excluded.state_json,
        updated_at = excluded.updated_at
    `)
    .bind(
      visitorId,
      serialized,
      state.createdAt || now,
      now
    )
    .run();
}

function normalizeGypsieState(candidate) {
  const fallback = createDefaultGypsieState();

  const state =
    candidate &&
    typeof candidate === "object"
      ? candidate
      : {};

  return {
    hp: clampNumber(
      state.hp,
      0,
      1000,
      fallback.hp
    ),

    maxHp: clampNumber(
      state.maxHp,
      1,
      1000,
      fallback.maxHp
    ),

    hunger: clampNumber(
      state.hunger,
      0,
      100,
      fallback.hunger
    ),

    money: Math.floor(
      clampNumber(
        state.money,
        0,
        1_000_000_000,
        fallback.money
      )
    ),

    conditions: Array.isArray(
      state.conditions
    )
      ? state.conditions.slice(0, 30)
      : fallback.conditions,

    inventory: Array.isArray(
      state.inventory
    )
      ? state.inventory.slice(0, 100)
      : fallback.inventory,

    equipment:
      state.equipment &&
      typeof state.equipment === "object"

              state.equipment.body || null,

              state.equipment.legs || null,
            accessory:
              state.equipment.accessory || null,
            weapon:
              state.equipment.weapon || null
          }
        : fallback.equipment,

    spellsKnown: Array.isArray(
      state.spellsKnown
    )
      ? state.spellsKnown.slice(0, 50)
      : fallback.spellsKnown,

    trust: clampNumber(
      state.trust,
      -100,
      100,
      fallback.trust
    ),

    fear: clampNumber(
      state.fear,
      0,
      100,
      fallback.fear
    ),

    resentment: clampNumber(
      state.resentment,
      0,
      100,
      fallback.resentment
    ),

    curiosity: clampNumber(
      state.curiosity,
      0,
      100,
      fallback.curiosity
    ),

    irritation: clampNumber(
      state.irritation,
      0,
      100,
      fallback.irritation
    ),

    kindnessReceived:
      normalizeRelationshipTotal(
        state.kindnessReceived
      ),

    crueltyReceived:
      normalizeRelationshipTotal(
        state.crueltyReceived
      ),

    amusementReceived:
      normalizeRelationshipTotal(
        state.amusementReceived
      ),

    recentRelationshipEvents:
      normalizeRecentEvents(

    paidReadings: Math.max(
      0,

      0,
      Math.floor(
        Number(state.totalCoinsTaken) || 0
      )

      0,
      Math.floor(
        Number(state.unpaidVisits) || 0
      )
    ),

    currentFee: Math.floor(
      clampNumber(
        state.currentFee,
        BASE_READING_FEE,
        MAX_READING_FEE,
        fallback.currentFee
      )
    ),

    lastCard:
      typeof state.lastCard === "string"
        ? state.lastCard
        : null,

    lastCardOrientation:
      state.lastCardOrientation === "upright" ||
      state.lastCardOrientation === "reversed"
        ? state.lastCardOrientation
        : null,

    lastReadingAt:
      validIsoDate(state.lastReadingAt),

    allegianceToLoki: clampNumber(
      state.allegianceToLoki,
      -100,
      100,
      fallback.allegianceToLoki
    ),

    createdAt:
      validIsoDate(state.createdAt) ||
      fallback.createdAt,

    updatedAt:
      validIsoDate(state.updatedAt) ||
      fallback.updatedAt
  };
}

function normalizeRelationshipTotal(value) {
  const numeric = Number(value);

  if (!Number.isFinite(numeric)) {
    return 0;
  }

  return Math.max(
    0,
    Math.min(
      MAX_RELATIONSHIP_SCORE,
      Math.floor(numeric)
    )
  );
}

function normalizeRecentEvents(value) {
  if (!Array.isArray(value)) {
    return [];
  }

  return value
    .slice(-30)
    .map((event) => {
      if (
        !event ||
        typeof event !== "object"
      ) {
        return null;
      }

      return {
        kindness: clampInteger(
          event.kindness,
          0,
          5,
          0
        ),
        cruelty: clampInteger(
          event.cruelty,
          0,
          5,
          0
        ),
        amusement: clampInteger(
          event.amusement,
          0,
          5,
          0
        ),
        pattern:
          normalizeRelationshipPattern(
            event.pattern
          ),
        summary: limitText(
          String(event.summary || ""),
          240
        ),
        createdAt:
          validIsoDate(event.createdAt) ||
          new Date().toISOString()
      };
    })
    .filter(Boolean);
}

/* =========================================================
   PAYMENT
   ========================================================= */

function calculateReadingFee(gypsieState) {
  const depth = Math.floor(
    gypsieState.paidReadings / 3
  );

  return Math.min(
    MAX_READING_FEE,
    BASE_READING_FEE + depth * 2
  );
}

function shouldChargeForMessage(
  message,
  recentConversation
) {
  const normalized =
    String(message || "")
      .trim()
      .toLowerCase();

  const freeSocialPatterns = [
    /^(hello|hi|hey|greetings|good morning|good evening)[.!? ]*$/,
    /^(who are you|what is your name)[?! ]*$/,
    /^(goodbye|bye|farewell)[.!? ]*$/
  ];

  if (
    freeSocialPatterns.some((pattern) =>
      pattern.test(normalized)
    )
  ) {
    return false;
  }

  const visitorMessages =
    recentConversation.filter(
      (messageItem) =>
        messageItem.role === "user"
    ).length;

  return visitorMessages >= 1 ||
    normalized.length > 30;
}

async function collectReadingFee(
  env,
  visitorId,
  worldState,
  gypsieState,
  fee
) {
  const player =
    normalizeSharedPlayer(worldState);

  if (player.money < fee) {
    return {
      paid: false,
      amount: 0
    };
  }

  player.money -= fee;
  gypsieState.money += fee;
  gypsieState.paidReadings += 1;
  gypsieState.totalCoinsTaken += fee;
  gypsieState.currentFee =
    calculateReadingFee(gypsieState);

  await env.WORLD_DB
    .prepare(`
      INSERT INTO gypsie_payments (
        visitor_id,
        amount,
        purpose,
        created_at
      )
      VALUES (?, ?, ?, ?)
    `)
    .bind(
      visitorId,
      fee,
      "conversation and divination",
      new Date().toISOString()
    )
    .run();

  return {
    paid: true,
    amount: fee
  };
}

   TAROT
   ========================================================= */

function messageRequestsDivination(message) {
  return /\b(tarot|card|cards|fortune|future|fate|omen|omens|divination|reading|prophecy|predict|prediction)\b/i.test(
    String(message || "")
  );
}

function shouldDrawTarot(message, paid) {
  if (!paid) {
    return false;
  }

  if (messageRequestsDivination(message)) {
    return true;
  }

  return Math.random() < 0.22;
}

function drawTarotCard() {
  const card =
    MAJOR_ARCANA[
      Math.floor(
        Math.random() *
        MAJOR_ARCANA.length
      )
    ];

  const orientation =
    Math.random() < 0.28
      ? "reversed"
      : "upright";

  return {
    ...card,
    orientation,
    meaning:
      orientation === "upright"
        ? card.upright
        : card.reversed
  };
}

/* =========================================================
   MEMORIES
   ========================================================= */

async function saveMemories(
  env,
  visitorId,
  memories
) {
  if (!Array.isArray(memories)) {
    return;
  }

  const statements = [];

  for (
    const memory of memories.slice(0, 6)
  ) {
    const summary =
      typeof memory?.summary === "string"
        ? limitText(
            memory.summary.trim(),
            MAX_STORED_MEMORY_LENGTH
          )
        : "";

    if (!summary) {
      continue;
    }

    const type =
      typeof memory?.type === "string"
        ? limitText(
            memory.type.trim(),
            50
          ) || "event"
        : "event";

    const importance = clampNumber(
      memory.importance,
      0,
      1,
      0.5
    );

    const emotionalWeight =
      clampNumber(
        memory.emotionalWeight,
        -1,
        1,
        0
      );

    statements.push(
      env.WORLD_DB
        .prepare(`
          INSERT INTO gypsie_memories (
            visitor_id,
            type,
            summary,
            importance,
            emotional_weight,
            created_at
          )
          VALUES (?, ?, ?, ?, ?, ?)
        `)
        .bind(
          visitorId,
          type,
          summary,
          importance,
          emotionalWeight,
          new Date().toISOString()
        )
    );
  }

  if (statements.length > 0) {
    await env.WORLD_DB.batch(
      statements
    );
  }
}

async function loadRelevantMemories(
  env,
  visitorId,
  message
) {
  const result = await env.WORLD_DB
    .prepare(`
      SELECT
        id,
        type,
        summary,
        importance,
        emotional_weight,
        created_at,
        recall_count
      FROM gypsie_memories
      WHERE visitor_id = ?
      ORDER BY
        importance DESC,
        id DESC
      LIMIT 250
    `)
    .bind(visitorId)
    .all();

  const rows = Array.isArray(
    result?.results
  )
    ? result.results
    : [];

  if (rows.length === 0) {
    return [];
  }

  const messageWords =
    meaningfulWords(message);

  const scored = rows.map((row) => {
    const memoryWords =
      meaningfulWords(row.summary);

    let overlap = 0;

    for (const word of messageWords) {
      if (memoryWords.has(word)) {
        overlap += 1;
      }
    }

    const importance =
      Number(row.importance) || 0;

    const emotional =
      Math.abs(
        Number(row.emotional_weight) || 0
      );

    const recallBonus = Math.min(
      0.15,
      (Number(row.recall_count) || 0) *
      0.005
    );

    return {
      ...row,
      score:
        overlap * 2 +
        importance * 3 +
        emotional +
        recallBonus
    };
  });

  scored.sort((a, b) => {
    if (b.score !== a.score) {
      return b.score - a.score;
    }

    return Number(b.id) - Number(a.id);
  });

  const selected = scored.slice(
    0,
    MAX_CONTEXT_MEMORIES
  );

  if (selected.length > 0) {
    const now = new Date().toISOString();

    await env.WORLD_DB.batch(
      selected.map((memory) =>
        env.WORLD_DB
          .prepare(`
            UPDATE gypsie_memories
            SET
              last_recalled_at = ?,
              recall_count =
                recall_count + 1
            WHERE id = ?
          `)
          .bind(now, memory.id)
      )
    );
  }

  return selected;
}

function selectRelevantLore(
  corpus,
  message
) {
  const sections = parseLoreSections(
    corpus
  );

  if (sections.length === 0) {
    return limitText(
      corpus,
      MAX_LORE_CONTEXT_LENGTH
    );
  }

  const messageWords =
    meaningfulWords(message);

  const expandedWords =
    expandLoreKeywords(messageWords);

  const scored = sections.map(
    (section, index) => {
      const sectionWords =
        meaningfulWords(
          `${section.title} ${section.body}`
        );

      let overlap = 0;

      for (const word of expandedWords) {
        if (sectionWords.has(word)) {
          overlap += 1;
        }
      }

      const alwaysUseful =
        /world|loki|labyrinth|visitor|tone/i
          .test(section.title)
          ? 0.7
          : 0;

      return {
        ...section,
        index,
        score: overlap + alwaysUseful
      };
    }
  );

  scored.sort((a, b) => {
    if (b.score !== a.score) {
      return b.score - a.score;
    }

    return a.index - b.index;
  });

  const selected = [];
  let currentLength = 0;

  for (const section of scored) {
    if (
      section.score <= 0 &&
      selected.length >= 4
    ) {
      continue;
    }

    const rendered =
      `[${section.title}]\n${section.body}`;

    if (
      currentLength +
      rendered.length >
      MAX_LORE_CONTEXT_LENGTH
    ) {
      continue;
    }

    selected.push(rendered);
    currentLength += rendered.length;

    if (selected.length >= 10) {
      break;
    }
  }

  return selected.join("\n\n");
}

function parseLoreSections(corpus) {
  const lines =
    String(corpus || "")
      .replace(/\r\n/g, "\n")
      .split("\n");

  const sections = [];
  let current = null;

  for (const line of lines) {
    const bracketMatch =
      line.match(/^\s*\[([^\]]+)\]\s*$/);

    const markdownMatch =
      line.match(
        /^\s*#{1,4}\s+(.+?)\s*$/
      );

    if (bracketMatch || markdownMatch) {
      if (
        current &&
        current.body.trim()
      ) {
        sections.push({
          title: current.title,
          body: current.body.trim()
        });
      }

      current = {
        title:
          bracketMatch?.[1] ||
          markdownMatch?.[1] ||
          "Lore",
        body: ""
      };

      continue;
    }

    if (!current) {
      current = {
        title: "General Lore",
        body: ""
      };
    }

    current.body += `${line}\n`;
  }

  if (
    current &&
    current.body.trim()
  ) {
    sections.push({
      title: current.title,
      body: current.body.trim()
    });
  }

  return sections;
}

      group.some((word) =>
        words.has(word)
      )
    ) {
      for (const word of group) {
        expanded.add(word);
      }
    }
  }

  return expanded;
}

/* =========================================================
   MODEL OUTPUT
   ========================================================= */

function normalizeModelOutput(raw) {
  let candidate = raw;

  // Workers AI may wrap the useful result one or more times.
  for (let depth = 0; depth < 4; depth += 1) {
    if (
      candidate &&
      typeof candidate === "object" &&
      !Array.isArray(candidate)
    ) {
      // Already looks like the character's final object.
      if (
        typeof candidate.text === "string" ||
        typeof candidate.reply === "string"
      ) {
        break;
      }

      if (candidate.response !== undefined) {
        candidate = candidate.response;
        continue;
      }

      if (candidate.result !== undefined) {
        candidate = candidate.result;
        continue;
      }

      if (candidate.output !== undefined) {
        candidate = candidate.output;
        continue;
      }

      if (candidate.generated_text !== undefined) {
        candidate = candidate.generated_text;
        continue;
      }
    }

    break;
  }

  let parsed = null;

  if (
    candidate &&
    typeof candidate === "object" &&
    !Array.isArray(candidate)
  ) {
    parsed = candidate;
  } else if (typeof candidate === "string") {
    const cleaned = candidate
      .trim()
      .replace(/^```(?:json)?\s*/i, "")
      .replace(/\s*```$/i, "");

    try {
      parsed = JSON.parse(cleaned);
    } catch {
      const start = cleaned.indexOf("{");
      const end = cleaned.lastIndexOf("}");

      if (start >= 0 && end > start) {
        try {
          parsed = JSON.parse(
            cleaned.slice(start, end + 1)
          );
        } catch {
          parsed = null;
        }
      }
    }

    // Preserve usable plain text rather than showing the card fallback.
    if (!parsed && cleaned) {
      return {
        type: "reply",
        text: limitText(
          cleaned,
          MAX_REPLY_LENGTH
        ),
        changes: [],
        actions: [],
        memories: [],
        relationshipEvent:
          createNeutralRelationshipEvent()
      };
    }
  }

  if (!parsed) {
    console.error(
      "Could not normalize AI output:",
      JSON.stringify(raw)
    );

    return {
      type: "reply",
      text:
        "The smoke shifts across the cards, but whatever answer formed there has slipped beyond my reach.",
      changes: [],
      actions: [],
      memories: [],
      relationshipEvent:
        createNeutralRelationshipEvent()
    };
  }

  const text = limitText(
    String(
      parsed.text ??
      parsed.reply ??
      parsed.response ??
      parsed.message ??
      ""
    ).trim(),
    MAX_REPLY_LENGTH
  );

  return {
    type: "reply",

    text:
      text ||
      "The smoke shifts across the cards, but whatever answer formed there has slipped beyond my reach.",

    changes: Array.isArray(parsed.changes)
      ? parsed.changes.slice(0, 20)
      : [],

    actions: Array.isArray(parsed.actions)
      ? parsed.actions.slice(0, 20)
      : [],

    memories: normalizeMemories(
      parsed.memories
    ),

    relationshipEvent:
      normalizeRelationshipEvent(
        parsed.relationshipEvent
      )
  };
}

function normalizeMemories(value) {
  if (!Array.isArray(value)) {
    return [];
  }

  return value
    .slice(0, 6)
    .map((memory) => {
      if (
        !memory ||
        typeof memory !== "object"
      ) {
        return null;
      }

      const summary = limitText(
        String(memory.summary || "").trim(),
        MAX_STORED_MEMORY_LENGTH
      );

      if (!summary) {
        return null;
      }

      return {
        type: limitText(
          String(memory.type || "event"),
          50
        ),
        summary,
        importance: clampNumber(
          memory.importance,
          0,
          1,
          0.5
        ),
        emotionalWeight:
          clampNumber(
            memory.emotionalWeight,
            -1,
            1,
            0
          )
      };
    })
    .filter(Boolean);
}

function normalizeRelationshipEvent(
  value
) {
  if (
    !value ||
    typeof value !== "object"
  ) {
    return createNeutralRelationshipEvent();
  }

  return {
    kindness: clampInteger(
      value.kindness,
      0,
      5,
      0
    ),

    cruelty: clampInteger(
      value.cruelty,
      0,
      5,
      0
    ),

    amusement: clampInteger(
      value.amusement,
      0,
      5,
      0
    ),

    pattern:
      normalizeRelationshipPattern(
        value.pattern
      ),

    summary: limitText(
      String(value.summary || ""),
      240
    )
  };
}

function createNeutralRelationshipEvent() {
  return {
    kindness: 0,
    cruelty: 0,
    amusement: 0,
    pattern: "neutral",
    summary: ""
  };
}

function normalizeRelationshipPattern(
  value
) {
  const normalized =
    String(value || "")
      .trim()
      .toLowerCase();

  const allowed = new Set([
    "neutral",
    "kind",
    "cruel",
    "amusing",
    "kind-and-amusing",
    "cruel-and-amusing",
    "contradictory",
    "threatening",
    "protective",
    "apologetic",
    "manipulative",
    "mocking",
    "commercial",
    "insulting",
    "generous",
    "suspicious",
    "unknown"
  ]);

  return allowed.has(normalized)
    ? normalized
    : "unknown";
}

/* =========================================================
   STATE CHANGES
   ========================================================= */

function applyRelationshipEvent(
  gypsieState,
  event
) {
  gypsieState.kindnessReceived =
    normalizeRelationshipTotal(
      gypsieState.kindnessReceived +
      event.kindness
    );

  gypsieState.crueltyReceived =
    normalizeRelationshipTotal(
      gypsieState.crueltyReceived +
      event.cruelty
    );

  gypsieState.amusementReceived =
    normalizeRelationshipTotal(
      gypsieState.amusementReceived +
      event.amusement
    );

  if (event.kindness > 0) {
    gypsieState.trust = clampNumber(
      gypsieState.trust +
      event.kindness,
      -100,
      100,
      gypsieState.trust
    );

    gypsieState.irritation = clampNumber(
      gypsieState.irritation -
      event.kindness,
      0,
      100,
      gypsieState.irritation
    );
  }

  if (event.cruelty > 0) {
    gypsieState.resentment =
      clampNumber(
        gypsieState.resentment +
        event.cruelty * 2,
        0,
        100,
        gypsieState.resentment
      );

    gypsieState.irritation =
      clampNumber(
        gypsieState.irritation +
        event.cruelty * 2,
        0,
        100,
        gypsieState.irritation
      );

    gypsieState.trust = clampNumber(
      gypsieState.trust -
      event.cruelty,
      -100,
      100,
      gypsieState.trust
    );
  }

  if (event.amusement > 0) {
    gypsieState.curiosity =
      clampNumber(
        gypsieState.curiosity +
        event.amusement,
        0,
        100,
        gypsieState.curiosity
      );
  }

  gypsieState.recentRelationshipEvents =
    [
      ...gypsieState.recentRelationshipEvents,
      {
        ...event,
        createdAt:
          new Date().toISOString()
      }
    ].slice(-30);
}

function applyRequestedChanges(
  worldState,
  gypsieState,
  changes
) {
  if (!Array.isArray(changes)) {
    return;
  }

  const player =
    normalizeSharedPlayer(worldState);

  for (const change of changes) {
    if (
      !change ||
      typeof change !== "object"
    ) {
      continue;
    }

    const target =
      change.target === "player"
        ? player
        : change.target === "gypsie"
          ? gypsieState
          : null;

    if (!target) {
      continue;
    }

    const stat = String(
      change.stat || ""
    );

    const amount = Number(
      change.amount
    );

    if (!Number.isFinite(amount)) {
      continue;
    }

    if (stat === "hp") {
      target.hp = clampNumber(
        target.hp + amount,
        0,
        target.maxHp || 100,
        target.hp
      );
    }

    if (stat === "hunger") {
      target.hunger = clampNumber(
        target.hunger + amount,
        0,
        100,
        target.hunger
      );
    }

    if (
      target === gypsieState &&
      [
        "trust",
        "fear",
        "resentment",
        "curiosity",
        "irritation",
        "allegianceToLoki"
      ].includes(stat)
    ) {
      const minimum =
        stat === "trust" ||
        stat === "allegianceToLoki"
          ? -100
          : 0;

      target[stat] = clampNumber(
        target[stat] + amount,
        minimum,
        100,
        target[stat]
      );
    }
  }
}

  ];

  const index = Math.min(
    responses.length - 1,

    }
  };
}

/* =========================================================
   CHAT
   ========================================================= */

async function handleChat(request, env) {
  try {
    await ensureDatabase(env);

    const body = await request.json();

    const visitorId = normalizeVisitorId(
      body?.visitorId
    );

    const message = limitText(
      String(body?.message || "").trim(),
      MAX_MESSAGE_LENGTH
    );

    if (!visitorId) {
      return jsonResponse(
        {
          error: "Missing visitor identity."
        },
        400
      );
    }

    if (!message) {
      return jsonResponse(
        {
          error: "Write something first."
        },
        400
      );
    }

    const [
      worldState,
      gypsieState,
      recentConversation
    ] = await Promise.all([
      loadSharedWorldState(
        env,
        visitorId
      ),
      loadGypsieState(
        env,
        visitorId
      ),
      loadRecentConversation(
        env,
        visitorId
      )
    ]);

const player =
  normalizeSharedPlayer(worldState);

const charge =
  shouldChargeForMessage(
    message,
    recentConversation
  );

const fee =
  calculateReadingFee(gypsieState);

gypsieState.currentFee = fee;

let payment = {
  paid: false,
  amount: 0
};

if (charge) {
  if (player.money < fee) {
        const refusal =
          buildUnpaidResponse(
            gypsieState,
            player.money,
            fee
          );

        applyRelationshipEvent(
          gypsieState,
          refusal.relationshipEvent
        );

        await Promise.all([
          saveConversationMessage(
            env,
            visitorId,
            "user",
            message
          ),
          saveConversationMessage(
            env,
            visitorId,
            "assistant",
            refusal.text
          ),
          saveMemories(
            env,
            visitorId,
            refusal.memories
          ),
          saveGypsieState(
            env,
            visitorId,
            gypsieState
          )
        ]);

        return jsonResponse({
          ...refusal,
          publicState:
            buildPublicState(
              worldState,
              gypsieState
            )
        });
      }

      payment = await collectReadingFee(
        env,
        visitorId,
        worldState,
        gypsieState,
        fee
      );
    }

    const tarotCard =
      shouldDrawTarot(
        message,
        payment.paid
      )
        ? drawTarotCard()
        : null;

    if (tarotCard) {
      gypsieState.lastCard =
        tarotCard.name;

      gypsieState.lastCardOrientation =
        tarotCard.orientation;

      gypsieState.lastReadingAt =
        new Date().toISOString();
    }

    const [
      memories,
      loreCorpus
    ] = await Promise.all([
      loadRelevantMemories(
        env,
        visitorId,
        message
      ),
      loadLoreCorpus(env)
    ]);

    const relevantLore =
      selectRelevantLore(
        loreCorpus,
        message
      );

    const systemContext =
      buildSystemContext({
        worldState,
        gypsieState,
        memories,
        lore: relevantLore,
        fee,
        payment,
        tarotCard
      });

    const messages = [
      {
        role: "system",
        content: GYPSIE_PROMPT
      },
      {
        role: "system",
        content: systemContext
      },
      ...recentConversation,
      {
        role: "user",
        content: message
      }
    ];

    const aiResult = await env.AI.run(
      MODEL,
      {
        messages,
        max_tokens: 1400,
        temperature: 0.82,
        top_p: 0.92,
        response_format: {
          type: "json_object"
        }
      }
    );

    const rawResponse =
      aiResult?.response ??
      aiResult?.result ??
      aiResult;

    const output =
      normalizeModelOutput(rawResponse);

    applyRelationshipEvent(
      gypsieState,
      output.relationshipEvent
    );

    applyRequestedChanges(
      worldState,
      gypsieState,
      output.changes
    );

    await saveConversationMessage(
      env,
      visitorId,
      "user",
      message
    );

    await saveConversationMessage(
      env,
      visitorId,
      "assistant",
      output.text
    );

    await saveMemories(
      env,
      visitorId,
      output.memories
    );

    if (tarotCard) {
      await saveTarotReading(
        env,
        visitorId,
        tarotCard,
        message,
        output.text,
        payment.amount
      );
    }

    await Promise.all([
      saveSharedWorldState(
        env,
        visitorId,
        worldState
      ),
      saveGypsieState(
        env,
        visitorId,
        gypsieState
      )
    ]);

    return jsonResponse({
      ...output,

      tarot: tarotCard
        ? {
            name: tarotCard.name,
            number: tarotCard.number,
            orientation:
              tarotCard.orientation
          }
        : null,

      payment: {
        charged: payment.amount,
        nextFee:
          gypsieState.currentFee
      },

      publicState:
        buildPublicState(
          worldState,
          gypsieState
        )
    });
  } catch (error) {
    console.error(
      "Gypsie chat failed:",
      error
    );

    return jsonResponse(
      {
        error:
          "The lamps dimmed and the reading failed.",
        details:
          String(error?.message || error)
      },
      500
    );
  }
}

/* =========================================================
   HELPERS
   ========================================================= */

function meaningfulWords(value) {
  const stopWords = new Set([
    "the",
    "and",
    "that",
    "this",
    "with",
    "from",
    "have",
    "your",
    "you",
    "are",
    "was",
    "were",
    "for",
    "but",
    "not",
    "what",
    "when",
    "where",
    "who",
    "why",
    "how",
    "into",
    "about",
    "they",
    "them",
    "their",
    "there",
    "then",
    "than",
    "will",
    "would",
    "could",
    "should"
  ]);

  return new Set(
    String(value || "")
      .toLowerCase()
      .replace(
        /[^\p{L}\p{N}'-]+/gu,
        " "
      )
      .split(/\s+/)
      .filter(
        (word) =>
          word.length >= 3 &&
          !stopWords.has(word)
      )
  );
}

function normalizeVisitorId(value) {
  const cleaned =
    String(value || "")
      .trim()
      .replace(
        /[^a-zA-Z0-9_-]/g,
        ""
      )
      .slice(0, 128);

  return cleaned || null;
}

function clampNumber(
  value,
  minimum,
  maximum,
  fallback
) {
  const numeric = Number(value);

  if (!Number.isFinite(numeric)) {
    return fallback;
  }

  return Math.max(
    minimum,
    Math.min(maximum, numeric)
  );
}

function clampInteger(
  value,
  minimum,
  maximum,
  fallback
) {
  return Math.round(
    clampNumber(
      value,
      minimum,
      maximum,
      fallback
    )
  );
}

function limitText(value, maximum) {
  const text = String(value || "");

  if (text.length <= maximum) {
    return text;
  }

  return text.slice(0, maximum);
}

function validIsoDate(value) {
  if (
    typeof value !== "string" ||
    !Number.isFinite(Date.parse(value))
  ) {
    return null;
  }

  return new Date(value).toISOString();
}

function describeHunger(value) {
  if (value < 15) {
    return "Well Fed";
  }

  if (value < 35) {
    return "Fed";
  }

  if (value < 55) {
    return "Hungry";
  }

  if (value < 75) {
    return "Very Hungry";
  }

  if (value < 90) {
    return "Starving";
  }

  return "Near Death";
}

function formatInventory(inventory) {
  if (
    !Array.isArray(inventory) ||
    inventory.length === 0
  ) {
    return "none";
  }

  return inventory
    .slice(0, 30)
    .map((item) => {
      if (
        typeof item === "string"
      ) {
        return item;
      }

      return `${item?.name || item?.id || "unknown item"} x${item?.quantity || 1}`;
    })
    .join(", ");
}

function formatEquipment(equipment) {
  if (
    !equipment ||
    typeof equipment !== "object"
  ) {
    return "none";
  }

  const equipped =
    Object.entries(equipment)
      .filter(
        ([, value]) => Boolean(value)
      )
      .map(
        ([slot, value]) =>
          `${slot}: ${value}`
      );

  return equipped.length > 0
    ? equipped.join(", ")
    : "none";
}

function formatNamedList(list) {
  if (
    !Array.isArray(list) ||
    list.length === 0
  ) {
    return "none";
  }

  return list
    .slice(0, 30)
    .map((item) => {
      if (
        typeof item === "string"
      ) {
        return item;
      }

      return item?.name ||
        item?.id ||
        "unknown";
    })
    .join(", ");
}

function jsonResponse(
  value,
  status = 200
) {
  return new Response(
    JSON.stringify(value),
    {
      status,
      headers: {
        "content-type":
          "application/json; charset=UTF-8",
        "cache-control": "no-store",
        "access-control-allow-origin": "*"
      }
    }
  );
}

function getEasternDailyGoldDate() {
  const parts = new Intl.DateTimeFormat(
    "en-CA",
    {
      timeZone: DAILY_GOLD_TIME_ZONE,
      year: "numeric",
      month: "2-digit",
      day: "2-digit",
      hour: "2-digit",
      minute: "2-digit",
      hourCycle: "h23"
    }
  ).formatToParts(new Date());

  const values = {};

  for (const part of parts) {
    if (part.type !== "literal") {
      values[part.type] = part.value;
    }
  }

  return {
    dateKey:
      `${values.year}-${values.month}-${values.day}`,

    hour: Number(values.hour),
    minute: Number(values.minute)
  };
}

function applyDailyGold(worldState) {
  const player =
    normalizeSharedPlayer(worldState);

  const eastern =
    getEasternDailyGoldDate();

  // Gold becomes available at 12:01 AM Eastern.
  const allowanceIsAvailable =
    eastern.hour > 0 ||
    (
      eastern.hour === 0 &&
      eastern.minute >= 1
    );

  if (!allowanceIsAvailable) {
    return false;
  }

  if (
    player.lastDailyGoldDate ===
    eastern.dateKey
  ) {
    return false;
  }

  player.money = Math.min(
    1_000_000_000,
    player.money + DAILY_GOLD_AMOUNT
  );

  player.lastDailyGoldDate =
    eastern.dateKey;

  return true;
}

    .replace(/\\[[^\\]]+\\]/g, " ")
    .replace(/\\*+/g, "")
    .replace(/\\s+/g, " ")

  if (
    !voiceEnabled ||
    !("speechSynthesis" in window)
  ) {
    return;

    cleanSpeechText(text);

  if (!spokenText) {
    return;
  }

  window.speechSynthesis.cancel();

  const utterance =
    new SpeechSynthesisUtterance(
      spokenText
    );

  if (gypsieVoice) {
    utterance.voice = gypsieVoice;
  }

  utterance.lang =
    gypsieVoice?.lang ||
    "en-US";

  // Slightly slower and lower than Sally.
utterance.rate = 0.90;
utterance.pitch = 0.98;
  utterance.volume = 1;

  window.speechSynthesis.speak(
    utterance
  );
}
