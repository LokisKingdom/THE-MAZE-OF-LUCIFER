        // =====================================================

        function getVisitorId() {

          /*
           * Inside the Android app, the native player owns the
           * authoritative MOBILE visitor ID. Prefer it.
           *
           * Desktop browsers do not have window.AndroidGame,
           * so they continue using their own localStorage ID.
           */
          try {

            if (
              window.AndroidGame &&
              typeof window.AndroidGame.getVisitorId ===
                "function"
            ) {

              const androidVisitorId =
                String(
                  window.AndroidGame.getVisitorId() ||
                  ""
                ).trim();


              if (
                androidVisitorId &&
                androidVisitorId.length <= 200
              ) {

                localStorage.setItem(
                  VISITOR_ID_KEY,
                  androidVisitorId
                );

                return androidVisitorId;
              }
            }

          } catch (error) {

            console.warn(
              "Android visitor bridge unavailable:",
              error
            );
          }


          let visitorId =
            localStorage.getItem(
              VISITOR_ID_KEY
            );


          if (!visitorId) {

            visitorId =
              (
                typeof crypto !==
                  "undefined" &&
                crypto.randomUUID
              )
                ? crypto.randomUUID()
                : (
                    "visitor-" +
                    Date.now() +
                    "-" +
                    Math.random()
                      .toString(36)
                      .slice(2)
                  );


            localStorage.setItem(
              VISITOR_ID_KEY,
              visitorId
            );
          }


          return visitorId;
        }

        async function advance() {

          if (locked) {
            return;
          }


          // Refused.
          if (step === 99) {

            window.location.href =
              "/";

            return;
          }


          // The Tower itself never handles payment.
          // The Wizard's office performs the real admission check.
          if (step === 100) {

            window.location.href =
              WIZARD_URL;

            return;
          }


          // Continue ordinary scenes.
          if (
            step <
            scenes.length - 1
          ) {

            step +=
              1;

            renderScene(
              step
            );

            return;
          }


          // Final Guild scene:
          // pass upward; the Wizard handles his own account.
          await testGuildAdmission();
        }

            }
