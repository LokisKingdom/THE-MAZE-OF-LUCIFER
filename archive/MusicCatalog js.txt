export function series(
  folder,
  baseTitle,
  baseFile,
  count,
  includeBase = true,
  start = 1
) {
  const songs = [];

  if (includeBase) {
    songs.push({
      title: baseTitle,
      file: `${folder}/${baseFile}.mp3`
    });
  }

  for (let i = start; i < start + count; i++) {
    songs.push({
      title: `${baseTitle} (${i})`,
      file: `${folder}/${baseFile}%20(${i}).mp3`
    });
  }

  return songs;
}
