Can Android read these images?

Yes.

They're inside the MP3 metadata.

If you're already using Media3/ExoPlayer, Android can extract them.

Typically you'd read the embedded artwork from the file's ID3 tags (the APIC frame). Libraries such as MediaMetadataRetriever or Media3's metadata support can provide the embedded image as a bitmap.

No extra files are needed.

Performance

The only thing I'd avoid is loading every cover at startup.

Instead:

load artwork only for the currently playing song
cache the last few covers
recycle the bitmap when changing songs

That keeps memory usage tiny.

5. Add the artwork functions

Place these functions above setupButtons():

private fun updateArtworkFromController() {
    val artworkData = mediaController
        ?.mediaMetadata
        ?.artworkData

    updateArtwork(artworkData)
}

private fun updateArtwork(artworkData: ByteArray?) {
    if (artworkData == null || artworkData.isEmpty()) {
        currentArtworkBitmap = null
        artworkImage.setImageDrawable(null)
        artworkImage.visibility = View.GONE
        artworkExpandBtn.visibility = View.GONE
        return
    }

    val bitmap = BitmapFactory.decodeByteArray(
        artworkData,
        0,
        artworkData.size
    )

    if (bitmap == null) {
        currentArtworkBitmap = null
        artworkImage.setImageDrawable(null)
        artworkImage.visibility = View.GONE
        artworkExpandBtn.visibility = View.GONE
        return
    }

    currentArtworkBitmap = bitmap

    artworkImage.setImageBitmap(bitmap)
    artworkImage.visibility = View.VISIBLE
    artworkExpandBtn.visibility = View.VISIBLE
}

private fun showFullscreenArtwork() {
    val bitmap = currentArtworkBitmap ?: return

    val imageView = ImageView(this).apply {
        setImageBitmap(bitmap)
        adjustViewBounds = true
        scaleType = ImageView.ScaleType.FIT_CENTER
        setBackgroundColor(Color.BLACK)

        setPadding(
            dpToPx(8),
            dpToPx(8),
            dpToPx(8),
            dpToPx(8)
        )
    }

    val dialog = AlertDialog.Builder(this)
        .setView(imageView)
        .setNegativeButton("Close", null)
        .create()

    dialog.window?.setBackgroundDrawable(
        ColorDrawable(Color.BLACK)
    )

    imageView.setOnClickListener {
        dialog.dismiss()
    }

    dialog.setOnShowListener {
        dialog.window?.setLayout(
            ViewGroup.LayoutParams.MATCH_PARENT,
            ViewGroup.LayoutParams.MATCH_PARENT
        )
    }

    dialog.show()

    dialog.window?.setLayout(
        ViewGroup.LayoutParams.MATCH_PARENT,
        ViewGroup.LayoutParams.MATCH_PARENT
    )
}

private fun dpToPx(dp: Int): Int {
    return (dp * resources.displayMetrics.density).toInt()
}

The main artwork will be large inside the app, and tapping either the artwork or the Expand Artwork button will open the nearly full-screen version. Tapping the full-screen image or pressing Close dismisses it.

private fun updateArtwork(artworkData: ByteArray?) {
    // Media3 may briefly report empty artwork while changing songs.
    // Ignore that temporary state so the WebView does not flash.
    if (artworkData == null || artworkData.isEmpty()) {
        return

        artworkData,

    }
