async function handleChat(request, env) {
  let body;

  try {
    body = await request.json();
  } catch {
    return jsonResponse(
      {
        error: "Invalid request."
      },
      400
    );
  }

  const message =
    typeof body.message === "string"
      ? body.message.trim()
      : "";

  const turnNumber =
    Number.isFinite(Number(body.turnNumber))
      ? Math.max(1, Math.min(Number(body.turnNumber), 100))
      : 1;

  const history =
    Array.isArray(body.history)
      ? body.history
          .filter((item) => {
            return (
              item &&
              (item.role === "user" || item.role === "assistant") &&
              typeof item.content === "string"
            );
          })
          .slice(-MAX_HISTORY_MESSAGES)
          .map((item) => ({
            role: item.role,
            content: item.content.slice(0, MAX_MESSAGE_LENGTH)
          }))
      : [];

  if (!message) {
    return jsonResponse(
      {
        error: "No message was provided."
      },
      400
    );
  }

  if (message.length > MAX_MESSAGE_LENGTH) {
    return jsonResponse(
      {
        error: "The message is too long."
      },
      413
    );
  }

  if (containsForbiddenName(message)) {
    return jsonResponse({
      type: "death",
      text:
        "Sally goes perfectly still. Then she bolts into the passage " +
        "without looking back. You follow before the opening can close. " +
        "The corridor turns twice, narrows, and ends at a door already " +
        "standing open. Something inside has been waiting for a name to " +
        "be spoken."
    });
  }

  if (!env.AI) {
    console.error("The Workers AI binding named AI is missing.");

    return jsonResponse(
      {
        type: "reply",
        text: "The wall trembles, but no voice passes through it."
      },
      500
    );
  }

  const messages = [
    {
      role: "system",
      content:
        SALLY_PROMPT +
        "\n\nCURRENT CONVERSATION TURN:\n" +
        String(turnNumber) +
        "\nUse this only to judge whether Sally is becoming restless. " +
        "Never mention the turn number to the visitor."
    },
    ...history,
    {
      role: "user",
      content: message
    }
  ];

  try {
    const result = await env.AI.run(MODEL, {
      messages,

      response_format: {
        type: "json_schema",

        json_schema: {
          type: "object",

          properties: {
            type: {
              type: "string",
              enum: ["reply", "death"]
            },

            text: {
              type: "string"
            }
          },

          required: ["type", "text"],
          additionalProperties: false
        }
      },

      max_tokens: 300,
      temperature: 0.8,
      repetition_penalty: 1.08
    });

    const parsed = normalizeSallyResponse(result?.response);

    return jsonResponse(parsed);
  } catch (error) {
    console.error("Workers AI error:", error);

    return jsonResponse(
      {
        type: "reply",
        text:
          "The voice falters. For a moment, there is only scratching " +
          "inside the wall."
      },
      502
    );
  }
}

function containsForbiddenName(message) {
  return FORBIDDEN_PATTERNS.some((pattern) => pattern.test(message));
}

function normalizeSallyResponse(response) {
  if (response && typeof response === "object") {
    if (
      response.type === "death" &&
      typeof response.text === "string" &&
      response.text.trim()
    ) {
      return {
        type: "death",
        text: limitText(response.text.trim(), 1200)
      };
    }

    if (
      response.type === "reply" &&
      typeof response.text === "string" &&
      response.text.trim()
    ) {
      return {
        type: "reply",
        text: limitText(response.text.trim(), 1200)
      };
    }
  }

  if (typeof response === "string" && response.trim()) {
    return parseSallyResponse(response.trim());
  }

  return {
    type: "reply",
    text:
      "For a moment, only the sound of Sally breathing comes through the wall."
  };
}

function parseSallyResponse(raw) {
  const candidates = [];

  if (raw) {
    candidates.push(raw);

    const fenced = raw
      .replace(/^```json\s*/i, "")
      .replace(/^```\s*/, "")
      .replace(/\s*```$/, "")
      .trim();

    if (fenced !== raw) {
      candidates.push(fenced);
    }

    const firstBrace = raw.indexOf("{");
    const lastBrace = raw.lastIndexOf("}");

    if (firstBrace !== -1 && lastBrace > firstBrace) {
      candidates.push(raw.slice(firstBrace, lastBrace + 1));
    }
  }

  for (const candidate of candidates) {
    try {
      const parsed = JSON.parse(candidate);

      if (
        parsed.type === "death" &&
        typeof parsed.text === "string" &&
        parsed.text.trim()
      ) {
        return {
          type: "death",
          text: limitText(parsed.text.trim(), 1200)
        };
      }

      if (
        parsed.type === "reply" &&
        typeof parsed.text === "string" &&
        parsed.text.trim()
      ) {
        return {
          type: "reply",
          text: limitText(parsed.text.trim(), 1200)
        };
      }
    } catch {
      // Try the next possible JSON candidate.
    }
  }

  return {
    type: "reply",
    text: limitText(raw, 1200) || "The passage falls silent for a moment."
  };
}

function limitText(value, maximumLength) {
  if (typeof value !== "string") {
    return "";
  }

  return value.slice(0, maximumLength);
}

function jsonResponse(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,

    headers: {
      "content-type": "application/json; charset=UTF-8",
      "cache-control": "no-store"
    }
  });
}

      messagesElement.appendChild(message);

      messagesElement.scrollTop =
        messagesElement.scrollHeight;
    }

    function addSystemMessage(text) {
      const message =
        document.createElement("div");

      message.className =
        "message system";

      message.textContent = text;

      messagesElement.appendChild(message);

      messagesElement.scrollTop =
        messagesElement.scrollHeight;
    }

    function setInputEnabled(enabled) {
      inputElement.disabled = !enabled;
      speakButton.disabled = !enabled;
    }

    function killPlayer(deathDescription) {
      conversationEnded = true;
      conversationHistory = [];

      stopVoice();
      setInputEnabled(false);

      statusElement.textContent =
        "THE PASSAGE HAS GONE SILENT";

      window.setTimeout(() => {
        messagesElement.replaceChildren();

        chatArea.style.display = "none";
        deathText.textContent = deathDescription;
        deathScreen.style.display = "flex";

        speakText(deathDescription);

        inputElement.value = "";
      }, 900);
    }

    async function sendMessage() {
      if (
        conversationEnded ||
        inputElement.disabled
      ) {
        return;
      }

      const message =
        inputElement.value.trim();

      if (!message) {
        return;
      }

      turnNumber += 1;

      addMessage(
        "YOU",
        message,
        "visitor"
      );

      inputElement.value = "";
      setInputEnabled(false);

      statusElement.textContent =
        "SALLY IS LISTENING";

      try {
        const response = await fetch("/chat", {
          method: "POST",

          headers: {
            "content-type": "application/json"
          },

          body: JSON.stringify({
            message,
            history: conversationHistory,
            turnNumber
          })
        });

        let result;

        try {
          result = await response.json();
        } catch {
          throw new Error(
            "The passage returned an unreadable answer."
          );
        }

        if (!response.ok) {
          throw new Error(
            result.error ||
            "The passage failed."
          );
        }

        if (result.type === "death") {
          addSystemMessage(
            "Something has gone wrong in the passage."
          );

          killPlayer(result.text);
          return;
        }

        const reply =
          typeof result.text === "string" &&
          result.text.trim()
            ? result.text.trim()
            : "Sally does not answer.";

        addMessage(
          "SALLY",
          reply,
          "sally"
        );

        speakText(reply);

        conversationHistory.push(
          {
            role: "user",
            content: message
          },
          {
            role: "assistant",
            content: reply
          }
        );

        conversationHistory =
          conversationHistory.slice(-12);

        setInputEnabled(true);

        statusElement.textContent =
          "SOMETHING LISTENS BEHIND THE WALL";

        inputElement.focus();
      } catch (error) {
        console.error(error);

        const failureLine =
          "The wall has gone cold. I cannot hear you.";

        addMessage(
          "SALLY",
          failureLine,
          "sally"
        );

        speakText(failureLine);
        setInputEnabled(true);

        statusElement.textContent =
          "THE PASSAGE IS UNSTABLE";

        inputElement.focus();
      }
    }

    function restartConversation() {
      conversationEnded = false;
      conversationHistory = [];
      turnNumber = 0;

      stopVoice();
      messagesElement.replaceChildren();

      deathScreen.style.display = "none";
      chatArea.style.display = "flex";

      setInputEnabled(true);

      inputElement.value = "";

      statusElement.textContent =
        "SOMETHING LISTENS BEHIND THE WALL";

      const openingLine =
        "...Who speaks through the wall?";

      addMessage(
        "SALLY",
        openingLine,
        "sally"
      );

    window.setTimeout(() => {
        if (browserSpeechSupported) {
            chooseVoice();
        }

        speakText(openingLine);
    }, 500);

      inputElement.focus();
    }

    voiceToggle.addEventListener(
      "click",
      toggleVoice
    );

    if (browserSpeechSupported) {
        chooseVoice();

        window.speechSynthesis.addEventListener(
            "voiceschanged",
            chooseVoice
        );
    }

    updateVoiceToggle();

    speakButton.addEventListener(
      "click",
      sendMessage
    );

    inputElement.addEventListener(
      "keydown",
      (event) => {
        if (event.key === "Enter") {
          event.preventDefault();
          sendMessage();
        }
      }
    );

    restartButton.addEventListener(
      "click",
      restartConversation
    );

    restartConversation();
  </script>
</body>
</html>`;
