// ------------------------------------------------------------
// TITLE MUSIC
// ------------------------------------------------------------

function playGodsDebris () {

	if (gameStarted) {
		return;
	}

	if (Scene1Music) {

		Scene1Music.pause();

		try {
			Scene1Music.currentTime = 0;
		} catch (error) {}

		Scene1Music = null;
	}

	TheBeginningMusic.pause();
	BrandonMusic.pause();
	Scene3Music.pause();

	if (GodsDebrisMusic.paused) {

		GodsDebrisMusic.play()
			.then(() => {
				titleMusicStarted = true;
				console.log('GODS DEBRIS playing from R2.');
			})
			.catch((error) => {
				console.error(
					'GODS DEBRIS playback failed:',
					error
				);
			});
	}
}


// ------------------------------------------------------------
// SCENE 1 MUSIC
// ------------------------------------------------------------

function playScene1Music () {

	gameStarted = true;

	GodsDebrisMusic.pause();
	TheBeginningMusic.pause();
	BrandonMusic.pause();
	Scene3Music.pause();

	try {
		GodsDebrisMusic.currentTime = 0;
	} catch (error) {}

	try {
		TheBeginningMusic.currentTime = 0;
	} catch (error) {}

	try {
		BrandonMusic.currentTime = 0;
	} catch (error) {}

	try {
		Scene3Music.currentTime = 0;
	} catch (error) {}

	if (Scene1Music) {

		Scene1Music.pause();

		try {
			Scene1Music.currentTime = 0;
		} catch (error) {}

		Scene1Music = null;
	}

	let selectedIndex;

	do {
		selectedIndex =
			Math.floor(
				Math.random() * Scene1MusicFiles.length
			);
	} while (
		Scene1MusicFiles.length > 1 &&
		selectedIndex === lastScene1MusicIndex
	);

	lastScene1MusicIndex = selectedIndex;

	const selectedFile =
		Scene1MusicFiles[selectedIndex];

	const musicUrl =
		Scene1MusicBase +
		selectedFile
			.split('/')
			.map(part => encodeURIComponent(part))
			.join('/');

	Scene1Music = new Audio(musicUrl);

	Scene1Music.loop = true;
	Scene1Music.preload = 'auto';
	Scene1Music.volume = 0.35;

	Scene1Music.play()
		.then(() => {
			console.log(
				'SCENE 1 MUSIC:',
				selectedFile
			);
		})
		.catch((error) => {
			console.error(
				'Scene 1 music playback failed:',
				selectedFile,
				error
			);
		});
}

let currentR2Voice = null;


function playR2Voice (soundName) {

	const file = R2Scene1Sounds[soundName];

	if (!file) {
		console.error(
			'Unknown R2 Scene 1 sound:',
			soundName
		);
		return;
	}

	if (currentR2Voice) {
		currentR2Voice.pause();
		currentR2Voice = null;
	}

	const voiceUrl =
		R2Scene1Base +
		file
			.split('/')
			.map(part => encodeURIComponent(part))
			.join('/');

	console.log(
		'Playing R2 voice:',
		voiceUrl
	);

	currentR2Voice =
		new Audio(voiceUrl);

	currentR2Voice.preload = 'auto';
	currentR2Voice.volume = 1;

	currentR2Voice.play()
		.catch((error) => {
			console.error(
				'R2 Scene 1 sound playback failed:',
				soundName,
				voiceUrl,
				error
			);
		});
}


function stopR2Voice () {

	if (!currentR2Voice) {
		return;
	}

	currentR2Voice.pause();

	try {
		currentR2Voice.currentTime = 0;
	} catch (error) {}

	currentR2Voice = null;
}


window.playR2Voice = playR2Voice;
window.stopR2Voice = stopR2Voice;
