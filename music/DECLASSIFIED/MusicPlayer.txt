<div class="music-player" style="margin: 3rem auto; max-width: 900px; padding: 2rem; background: #111; color: #FFD700; border: 2px solid #4B0082; border-radius: 14px;">
  <h2 style="text-align: center; color: #FFD700;">Music Player</h2>

  <p style="text-align: center;">
    Now Playing:
    <strong class="nowPlayingTitle">Ready</strong>
  </p>

  <audio class="audioPlayer" controls style="width: 100%; margin: 1rem 0;"></audio>

  <div style="text-align: center; margin: 1rem 0;">
    <button class="repeatBtn" style="padding: 10px 18px; background: #4B0082; color: white; border: none; border-radius: 8px; font-weight: bold; cursor: pointer;">
      Repeat: All
    </button>
  </div>

  <ul class="playlist" style="list-style: none; padding: 0; margin: 2rem 0 0;"></ul>
</div>

<script define:vars={{ tracks }}>
  const safeTracks = Array.isArray(tracks) ? tracks : [];

  const player = document.querySelector('.music-player');

  if (player) {
    const audio = player.querySelector('.audioPlayer');
    const titleEl = player.querySelector('.nowPlayingTitle');
    const repeatBtn = player.querySelector('.repeatBtn');
    const playlistEl = player.querySelector('.playlist');

    const r2Base =
      'https://pub-4ae7f49887bc46e9a03de6e38886b609.r2.dev/';

    let currentIndex = 0;
    let userStartedPlayback = false;

    // 0 = off
    // 1 = all
    // 2 = one
    let repeatMode = 1;

    function getAudioSrc(file) {
      if (!file) return '';

      if (file.startsWith('http://') || file.startsWith('https://')) {
        return file;
      }

      if (file.startsWith('/')) {
        return file;
      }

      return r2Base + file;
    }

    function getDownloadName(track) {
      if (!track || !track.title) return 'song.mp3';

      const file = track.file || '';
      const cleanFile = file.split('?')[0].split('#')[0];
      const extensionMatch = cleanFile.match(/\.[a-z0-9]+$/i);
      const extension = extensionMatch ? extensionMatch[0] : '.mp3';

      return track.title + extension;
    }

    function updateRepeatMode() {
      if (!audio || !repeatBtn) return;

      audio.loop = repeatMode === 2;

      if (repeatMode === 0) {
        repeatBtn.textContent = 'Repeat: Off';
      } else if (repeatMode === 1) {
        repeatBtn.textContent = 'Repeat: All';
      } else {
        repeatBtn.textContent = 'Repeat: One';
      }
    }

    function updateMediaSession(track) {
      if (!track || !('mediaSession' in navigator)) return;

      navigator.mediaSession.metadata =
        new MediaMetadata({
          title: track.title || 'Unknown Track',
          artist: "Loki's Kingdom",
          album: 'EFSMB',
          artwork: [
            {
              src: '/Loki.jpg',
              sizes: '512x512',
              type: 'image/jpeg'
            }
          ]
        });

      navigator.mediaSession.setActionHandler('nexttrack', () => {
        if (safeTracks.length === 0) return;

        userStartedPlayback = true;
        currentIndex = (currentIndex + 1) % safeTracks.length;
        playTrack(currentIndex, true);
      });

      navigator.mediaSession.setActionHandler('previoustrack', () => {
        if (safeTracks.length === 0) return;

        userStartedPlayback = true;
        currentIndex = (currentIndex - 1 + safeTracks.length) % safeTracks.length;
        playTrack(currentIndex, true);
      });
    }

    function playTrack(index, shouldAutoPlay = true) {
      if (!audio || !titleEl) return;
      if (safeTracks.length === 0) return;
      if (index < 0 || index >= safeTracks.length) return;

      const track = safeTracks[index];

      if (!track || !track.file) {
        console.warn('Missing track or file at index:', index, track);
        return;
      }

      currentIndex = index;
      titleEl.textContent = track.title || 'Unknown Track';

      audio.pause();
      audio.src = getAudioSrc(track.file);
      audio.load();

      updateMediaSession(track);

      if (shouldAutoPlay && userStartedPlayback) {
        audio.play().catch((err) => {
          console.error('Playback failed:', err);
        });
      }
    }

    // BUILD PLAYLIST
    if (playlistEl) {
      playlistEl.innerHTML = '';

      safeTracks.forEach((track, index) => {
        if (!track || !track.file) {
          console.warn('Skipping invalid track:', index, track);
          return;
        }

        const li = document.createElement('li');

        li.style.margin = '12px 0';
        li.style.display = 'flex';
        li.style.alignItems = 'center';
        li.style.gap = '12px';

        const songLink = document.createElement('a');

        songLink.href = '#';
        songLink.textContent = track.title || `Track ${index + 1}`;

        songLink.style.color = 'inherit';
        songLink.style.textDecoration = 'none';
        songLink.style.flex = '1';
        songLink.style.cursor = 'pointer';

        songLink.addEventListener('click', (e) => {
          e.preventDefault();
          userStartedPlayback = true;
          playTrack(index, true);
        });

        const downloadLink = document.createElement('a');

        downloadLink.href = getAudioSrc(track.file);
        downloadLink.download = getDownloadName(track);
        downloadLink.textContent = '⬇ Download';

        downloadLink.style.padding = '9px 18px';
        downloadLink.style.background = '#8B0000';
        downloadLink.style.color = 'white';
        downloadLink.style.borderRadius = '6px';
        downloadLink.style.textDecoration = 'none';
        downloadLink.style.fontWeight = 'bold';
        downloadLink.style.whiteSpace = 'nowrap';
        downloadLink.style.display = 'inline-block';

        li.append(songLink, downloadLink);
        playlistEl.appendChild(li);
      });
    }

    if (audio) {
      audio.addEventListener('play', () => {
        userStartedPlayback = true;
      });

      audio.addEventListener('ended', () => {
        if (repeatMode === 2) return;

        if (repeatMode === 1 && safeTracks.length > 0) {
          currentIndex = (currentIndex + 1) % safeTracks.length;
          playTrack(currentIndex, true);
        }
      });
    }

    if (repeatBtn) {
      repeatBtn.addEventListener('click', () => {
        repeatMode = (repeatMode + 1) % 3;
        updateRepeatMode();
      });
    }

    updateRepeatMode();

    if (safeTracks.length > 0 && titleEl) {
      titleEl.textContent = safeTracks[0]?.title || 'Ready';
    } else if (titleEl) {
      titleEl.textContent = 'No tracks loaded';
    }
  }
</script>
